import Main

/- Build-time verification only; all mathematical declarations are in Main.lean. -/
open Lean in
run_cmd do
  let env ← getEnv
  let mut records : List (String × Json) := []
  let mut count : Nat := 0
  for (n, ci) in env.constants.toList do
    if (`MG).isPrefixOf n || (`formal_oet).isPrefixOf n then
      let axs ← collectAxioms n
      for ax in axs do
        unless #[`propext, `Classical.choice, `Quot.sound].contains ax do
          throwError "Forbidden axiom {ax} in {n}"
      let origin := match env.getModuleIdxFor? n with
        | some idx => env.header.moduleNames[idx]!
        | none => env.mainModule
      let record := Json.mkObj [
        ("module", toJson origin.toString),
        ("axioms", toJson (axs.toList.map Name.toString)),
        ("uses", toJson (ci.getUsedConstantsAsSet.toList.map Name.toString))]
      records := (n.toString, record) :: records
      if (`MG).isPrefixOf n then
        count := count + 1
  IO.FS.createDirAll ".build"
  IO.FS.writeFile ".build/audit.json" (Json.mkObj records).pretty
  logInfo m!"Audited {count} MG declarations and imported frozen declarations; only permitted axioms."
