def rowColEntry {I : Type} (M : ChannelMatrix I) (j i : I) : ENNReal :=
  M i j

/--

将通道矩阵转换为以输出分量为行指标、输入分量为列指标的标准 `Matrix I I ENNReal`。

/-! ## CH02-D01：全域/受限 Lip 矩阵谓词与证书族 -/

/--
OET anchor: oet-af85899693cc8750 (CH02-D01)
**定义（全域 Lipschitz 矩阵谓词）**：
对 $\phi \in \mathcal{X}_{\mathrm{op}}$，矩阵 $\mathbf{L}$ 在 $\mathrm{dom}(\phi)$ 上对所有 $x, y \in \mathrm{dom}(\phi)$ 和 $j \in I$ 满足
$d_j(\phi(x), \phi(y)) \le \sum_{i \in I} L_{i \to j} \cdot d_i(x, y)$。
-/
def IsLipMatrix {X I : Type} {bot : X} [Fintype I]
    (d : I → X → X → ENNReal) (ϕ : Operator X bot) (L : ChannelMatrix I) : Prop :=
  ∀ x ∈ dom ϕ, ∀ y ∈ dom ϕ, ∀ j : I,
    d j (ϕ.func x) (ϕ.func y) ≤ ∑ i : I, L i j * d i x y

/--
OET anchor: oet-af85899693cc8750 (CH02-D01)
**定义（全域 Lip 证书族 $\mathscr{L}(\phi)$）**：算子 $\phi$ 的全体合法全域 Lip 矩阵之集。
-/
def LipCertFamily {X I : Type} {bot : X} [Fintype I]
    (d : I → X → X → ENNReal) (ϕ : Operator X bot) : Set (ChannelMatrix I) :=
  { L | IsLipMatrix d ϕ L }

/--
OET anchor: oet-af85899693cc8750 (CH02-D01)
**定义（给定集合上的逐点 Lip 不等式谓词）**：
对任意 $x, y \in U$ 和 $j \in I$ 满足 $d_j(\phi(x), \phi(y)) \le \sum_{i \in I} L_{i \to j} \cdot d_i(x, y)$。
-/
def IsLipOn {X I : Type} {bot : X} [Fintype I]
    (d : I → X → X → ENNReal) (ϕ : Operator X bot) (U : Set X) (L : ChannelMatrix I) : Prop :=
  ∀ x ∈ U, ∀ y ∈ U, ∀ j : I,
    d j (ϕ.func x) (ϕ.func y) ≤ ∑ i : I, L i j * d i x y

/--
OET anchor: oet-af85899693cc8750 (CH02-D01)
**定义（子集受限 Lip 矩阵谓词）**：
对子集 $U \subseteq \mathrm{dom}(\phi)$，在 $U$ 上对任意 $x, y \in U$ 和 $j \in I$ 均满足
$d_j(\phi(x), \phi(y)) \le \sum_{i \in I} L_{i \to j} \cdot d_i(x, y)$。
-/
def IsRestrictedLipMatrix {X I : Type} {bot : X} [Fintype I]
    (d : I → X → X → ENNReal) (ϕ : Operator X bot) (U : Set X) (L : ChannelMatrix I) : Prop :=
  U ⊆ dom ϕ ∧
    ∀ x ∈ U, ∀ y ∈ U, ∀ j : I,
      d j (ϕ.func x) (ϕ.func y) ≤ ∑ i : I, L i j * d i x y

/--
OET anchor: oet-af85899693cc8750 (CH02-D01)
**定义（受限 Lip 证书族 $\mathscr{L}(\phi)\big|_U$）**：算子 $\phi$ 在子集 $U \subseteq \mathrm{dom}(\phi)$ 上的合法受限 Lip 矩阵之集。
-/
def RestrictedLipCertFamily {X I : Type} {bot : X} [Fintype I]
    (d : I → X → X → ENNReal) (ϕ : Operator X bot) (U : Set X) : Set (ChannelMatrix I) :=
  { L | IsRestrictedLipMatrix d ϕ U L }

/--
OET anchor: oet-af85899693cc8750 (CH02-D01)
当已知 $U \subseteq \mathrm{dom}(\phi)$ 时，`IsRestrictedLipMatrix` 与 `IsLipOn` 等价。
-/
theorem isRestrictedLipMatrix_iff_isLipOn {X I : Type} {bot : X} [Fintype I]
    (d : I → X → X → ENNReal) (ϕ : Operator X bot) {U : Set X}
    (hU : U ⊆ dom ϕ) (L : ChannelMatrix I) :
    IsRestrictedLipMatrix d ϕ U L ↔ IsLipOn d ϕ U L :=
  ⟨fun h => h.2, fun h => ⟨hU, h⟩⟩

/--
OET anchor: oet-af85899693cc8750 (CH02-D01)
当限制域取全域 $U = \mathrm{dom}(\phi)$ 时，受限 Lip 谓词与全域 Lip 谓词等价。
-/
theorem isRestrictedLipMatrix_dom_iff {X I : Type} {bot : X} [Fintype I]
    (d : I → X → X → ENNReal) (ϕ : Operator X bot) (L : ChannelMatrix I) :
    IsRestrictedLipMatrix d ϕ (dom ϕ) L ↔ IsLipMatrix d ϕ L := by

  · exact fun h => h.2
  · exact fun h => ⟨Set.Subset.rfl, h⟩

/--
OET anchor: oet-af85899693cc8750 (CH02-D01)
$\mathscr{L}(\phi)\big|_{\mathrm{dom}(\phi)} = \mathscr{L}(\phi)$。
-/
theorem restrictedLipCertFamily_dom_eq {X I : Type} {bot : X} [Fintype I]
    (d : I → X → X → ENNReal) (ϕ : Operator X bot) :
    RestrictedLipCertFamily d ϕ (dom ϕ) = LipCertFamily d ϕ := by

  exact isRestrictedLipMatrix_dom_iff d ϕ L

/--
OET anchor: oet-4ff5c58d71e1651f (CH02-D01)
**引理 2.1（受限 Lip 矩阵族的单调性）**：
若 $U \subseteq V \subseteq \mathrm{dom}(\phi)$，则 $\mathscr{L}(\phi)\big|_V \subseteq \mathscr{L}(\phi)\big|_U$。
-/

    (d : I → X → X → ENNReal) (ϕ : Operator X bot) {U V : Set X}
    (hUV : U ⊆ V) (hV : V ⊆ dom ϕ) :

/-! ## CH02-P19：单一输入项行化简；包含对角特例 -/

/--
OET anchor: oet-7fa97eb6e24d5527 (CH02-P19)
**单一输入项行求和化简**：
若第 $j$ 行除 $L_{i \to j}$ 外均为零，则 $\sum_{k \in I} L_{k \to j} d_k(x, y) = L_{i \to j} d_i(x, y)$。
-/
theorem singleInputRow_sum_eq {X I : Type} [Fintype I]
    (d : I → X → X → ENNReal) (L : ChannelMatrix I) (i j : I)
    (h_row : IsSingleInputRow L i j) (x y : X) :
    (∑ k : I, L k j * d k x y) = L i j * d i x y := by
  refine Finset.sum_eq_single i ?_ ?_
  · intro k _ hki
    rw [h_row k hki, zero_mul]
  · intro hi_not_mem
    exact False.elim (hi_not_mem (Finset.mem_univ i))

/--
OET anchor: oet-7fa97eb6e24d5527 (CH02-P19)
**单一输入项行 Lip 不等式化简**：
设 $\mathbf{L} \in \mathscr{L}(\phi)\big|_U$。若第 $j$ 行除 $L_{i \to j}$ 外均为零，则对任意 $x, y \in U$，
$d_j(\phi(x), \phi(y)) \le \sum_{k \in I} L_{k \to j} d_k(x, y) = L_{i \to j} d_i(x, y)$。
-/
theorem singleInputRow_lip_bound {X I : Type} {bot : X} [Fintype I]
    (d : I → X → X → ENNReal) (ϕ : Operator X bot) (U : Set X)
    (L : ChannelMatrix I) (i j : I)
    (hL : L ∈ RestrictedLipCertFamily d ϕ U)
    (h_row : IsSingleInputRow L i j) :
    ∀ x ∈ U, ∀ y ∈ U,
      d j (ϕ.func x) (ϕ.func y) ≤ ∑ k : I, L k j * d k x y ∧
      (∑ k : I, L k j * d k x y) = L i j * d i x y ∧
      d j (ϕ.func x) (ϕ.func y) ≤ L i j * d i x y := by
  intro x hx y hy
  have h_sum : (∑ k : I, L k j * d k x y) = L i j * d i x y :=
    singleInputRow_sum_eq d L i j h_row x y
  have h_lip : d j (ϕ.func x) (ϕ.func y) ≤ ∑ k : I, L k j * d k x y :=
    hL.2 x hx y hy j
  exact ⟨h_lip, h_sum, h_sum ▸ h_lip⟩

/--
OET anchor: oet-7fa97eb6e24d5527 (CH02-P19)
**单一输入项行 Lip 标量上界**：
若 $\mathbf{L} \in \mathscr{L}(\phi)\big|_U$ 且第 $j$ 行除 $L_{i \to j}$ 外均为零，则对任意 $x, y \in U$，
$d_j(\phi(x), \phi(y)) \le L_{i \to j} d_i(x, y)$。
-/
theorem singleInputRow_lip_le {X I : Type} {bot : X} [Fintype I]
    (d : I → X → X → ENNReal) (ϕ : Operator X bot) (U : Set X)
    (L : ChannelMatrix I) (i j : I)
    (hL : L ∈ RestrictedLipCertFamily d ϕ U)
    (h_row : IsSingleInputRow L i j) :
    ∀ x ∈ U, ∀ y ∈ U,
      d j (ϕ.func x) (ϕ.func y) ≤ L i j * d i x y :=
  fun x hx y hy => (singleInputRow_lip_bound d ϕ U L i j hL h_row x hx y hy).2.2

/--
OET anchor: oet-7fa97eb6e24d5527 (CH02-P19)
**对角 Lip 矩阵特例**：
若 $\mathbf{L} \in \mathscr{L}(\phi)\big|_U$ 为对角矩阵，则对每个同分量通道 $(i, i)$ 及任意 $x, y \in U$，
$\sum_{k \in I} L_{k \to i} d_k(x, y) = L_{i \to i} d_i(x, y)$ 且
$d_i(\phi(x), \phi(y)) \le L_{i \to i} d_i(x, y)$。
-/
theorem diagonal_lip_bound {X I : Type} {bot : X} [Fintype I]
    (d : I → X → X → ENNReal) (ϕ : Operator X bot) (U : Set X)
    (L : ChannelMatrix I)
    (hL : L ∈ RestrictedLipCertFamily d ϕ U)
    (h_diag : IsDiagonalChannelMatrix L) :
    ∀ i : I, ∀ x ∈ U, ∀ y ∈ U,
      (∑ k : I, L k i * d k x y) = L i i * d i x y ∧
      d i (ϕ.func x) (ϕ.func y) ≤ L i i * d i x y := by
  intro i x hx y hy
  have h_row : IsSingleInputRow L i i :=
    (isDiagonalChannelMatrix_iff_forall_isSingleInputRow L).mp h_diag i
  have h := singleInputRow_lip_bound d ϕ U L i i hL h_row x hx y hy
  exact ⟨h.2.1, h.2.2⟩
