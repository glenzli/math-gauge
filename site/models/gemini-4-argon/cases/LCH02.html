/--
OET anchor: oet-af85899693cc8750 (CH02-D03)
**输入/输出矩阵与指标约定**
以通道方向 `i → j`（输入分量 `i` 到输出分量 `j`）索引的扩展非负实数矩阵：`L i j` 表示 $L_{i \to j} \in \bar{\mathbb{R}}_+$。
-/
abbrev GaugeMatrix (I : Type) : Type := I → I → ENNReal

/--
OET anchor: oet-af85899693cc8750 (CH02-D03)
通道条目读取：`channelEntry L i j` 对应正文中从输入分量 $i$ 到输出分量 $j$ 的矩阵条目 $L_{i \to j}$。
-/
def channelEntry {I : Type} (L : GaugeMatrix I) (i j : I) : ENNReal := L i j

/--
OET anchor: oet-af85899693cc8750 (CH02-D03)
行列视图转换：书写矩阵 $\mathbf{L}$ 时，以输出分量 $j$ 为行指标、输入分量 $i$ 为列指标，
即第 $j$ 行、第 $i$ 列元素满足 $(\mathbf{L})_{j i} = L_{i \to j}$。
-/
def toRowColMatrix {I : Type} (L : GaugeMatrix I) : Matrix I I ENNReal :=
  fun j i => L i j

/--
OET anchor: oet-af85899693cc8750 (CH02-D03)
从以输出为行、输入为列的标准 `Matrix I I ENNReal` 还原通道索引矩阵。
-/
def ofRowColMatrix {I : Type} (A : Matrix I I ENNReal) : GaugeMatrix I :=
  fun i j => A j i

/--
OET anchor: oet-af85899693cc8750 (CH02-D03)
行列指标约定恒等式：$(\mathbf{L})_{j i} = L_{i \to j}$。
-/
@[simp] theorem toRowColMatrix_apply {I : Type} (L : GaugeMatrix I) (j i : I) :
    toRowColMatrix L j i = channelEntry L i j := rfl

/--
OET anchor: oet-af85899693cc8750 (CH02-D03)
**矩阵乘法（按输入/输出通道约定）**
$$(\mathbf{M}^{(2)} \cdot \mathbf{M}^{(1)})_{i \to j} := \sum_{k \in I} M^{(2)}_{k \to j} M^{(1)}_{i \to k}.$$
-/
def matMul {I : Type} [Fintype I] (M₂ M₁ : GaugeMatrix I) : GaugeMatrix I :=
  fun i j => ∑ k : I, M₂ k j * M₁ i k

@[simp] theorem matMul_apply {I : Type} [Fintype I] (M₂ M₁ : GaugeMatrix I) (i j : I) :
    matMul M₂ M₁ i j = ∑ k : I, M₂ k j * M₁ i k := rfl

/--
OET anchor: oet-af85899693cc8750 (CH02-D03)
通道指标下的 `matMul` 与“输出为行、输入为列”约定下的标准代数矩阵乘法严格一致：
$(\mathbf{M}^{(2)} \cdot \mathbf{M}^{(1)})_{j i} = \sum_{k \in I} (\mathbf{M}^{(2)})_{j k} (\mathbf{M}^{(1)})_{k i}$。
-/
theorem toRowColMatrix_matMul {I : Type} [Fintype I] (M₂ M₁ : GaugeMatrix I) :
    toRowColMatrix (matMul M₂ M₁) = toRowColMatrix M₂ * toRowColMatrix M₁ := by
  ext j i
  simp [toRowColMatrix, matMul, Matrix.mul_apply]

OET anchor: oet-af85899693cc8750 (CH02-D03)
扩展非负矩阵乘法的结合律。

theorem matMul_assoc {I : Type} [Fintype I] (M₃ M₂ M₁ : GaugeMatrix I) :
    matMul (matMul M₃ M₂) M₁ = matMul M₃ (matMul M₂ M₁) := by

  simp only [matMul_apply, Finset.sum_mul, Finset.mul_sum, mul_assoc]

**定义（全域 Lipschitz 矩阵谓词）**

  ∀ x ∈ dom ϕ, ∀ y ∈ dom ϕ, ∀ j : I,
    d j (ϕ.func x) (ϕ.func y) ≤ ∑ i : I, L i j * d i x y
