/-- HC02-D03.  A bound matrix is stored input first: `M i j` is the
coefficient from input component `i` to output component `j`. -/
abbrev BoundMatrix (I : Type u) := I → I → ENNReal

/-- HC02-D03.  The usual displayed matrix has output `j` as its row and
input `i` as its column. -/
def displayMatrix {I : Type u} (M : BoundMatrix I) : Matrix I I ENNReal :=
  fun j i => M i j

/-- Convert an ordinarily displayed matrix back to input-first storage. -/
def inputMatrix {I : Type u} (A : Matrix I I ENNReal) : BoundMatrix I :=
  fun i j => A j i

@[simp] theorem inputMatrix_displayMatrix {I : Type u} (M : BoundMatrix I) :
    inputMatrix (displayMatrix M) = M := rfl

@[simp] theorem displayMatrix_inputMatrix {I : Type u} (A : Matrix I I ENNReal) :
    displayMatrix (inputMatrix A) = A := rfl

/-- HC02-D03.  This is `(M₂ M₁) i j = ∑ k, M₂ k j * M₁ i k`. -/
noncomputable def boundMul {I : Type u} [Fintype I] (M₂ M₁ : BoundMatrix I) : BoundMatrix I :=
  fun i j => ∑ k : I, M₂ k j * M₁ i k

@[simp] theorem boundMul_apply {I : Type u} [Fintype I]
    (M₂ M₁ : BoundMatrix I) (i j : I) :
    boundMul M₂ M₁ i j = ∑ k : I, M₂ k j * M₁ i k := rfl

/-- HC02-D03: input-first multiplication is ordinary matrix multiplication
after exchanging rows and columns. -/
theorem displayMatrix_boundMul {I : Type u} [Fintype I] (M₂ M₁ : BoundMatrix I) :
    displayMatrix (boundMul M₂ M₁) = displayMatrix M₂ * displayMatrix M₁ := by
  ext j i
  simp [displayMatrix, boundMul, Matrix.mul_apply]

/-- HC02-D04.  The identity uses the Kronecker delta, with input index first. -/
noncomputable def boundIdentity {I : Type u} : BoundMatrix I := by
  classical
  exact fun i j => if i = j then 1 else 0

/-- HC02-D04.  The ordered product of `A 0, ..., A (n-1)` is
`A (n-1) * ... * A 0`; the empty product is the identity. -/
noncomputable def orderedProduct {I : Type u} [Fintype I] (A : ℕ → BoundMatrix I) :
    ℕ → BoundMatrix I
  | 0 => boundIdentity
  | n + 1 => boundMul (A n) (orderedProduct A n)

@[simp] theorem orderedProduct_zero {I : Type u} [Fintype I] (A : ℕ → BoundMatrix I) :
    orderedProduct A 0 = boundIdentity := rfl

/-- HC02-D01.  Restricted Lipschitz certificates.  Membership includes the
stated restriction-domain condition. -/
def IsRestrictedLip {X : Type} {I : Type u} {bot : X} [Fintype I]
    (d : I → X → X → ENNReal) (ϕ : Operator X bot) (U : Set X)
    (L : BoundMatrix I) : Prop :=
  U ⊆ dom ϕ ∧
    ∀ x ∈ U, ∀ y ∈ U, ∀ j : I,
      d j (ϕ.func x) (ϕ.func y) ≤ ∑ i : I, L i j * d i x y

/-- HC02-D01.  The family of restricted Lipschitz certificates. -/
def restrictedLipCertificates {X : Type} {I : Type u} {bot : X} [Fintype I]
    (d : I → X → X → ENNReal) (ϕ : Operator X bot) (U : Set X) : Set (BoundMatrix I) :=
  {L | IsRestrictedLip d ϕ U L}

/-- HC02-D01.  Global certificates are certificates on `dom ϕ`. -/
def lipCertificates {X : Type} {I : Type u} {bot : X} [Fintype I]
    (d : I → X → X → ENNReal) (ϕ : Operator X bot) : Set (BoundMatrix I) :=
  restrictedLipCertificates d ϕ (dom ϕ)

/-- HC02-D02.  Restricted co-Lipschitz certificates, including nonemptiness
of the restriction domain as part of the definition. -/
def IsRestrictedCoLip {X : Type} {I : Type u} {bot : X} [Fintype I]
    (d : I → X → X → ENNReal) (ϕ : Operator X bot) (U : Set X)
    (K : BoundMatrix I) : Prop :=
  U.Nonempty ∧ U ⊆ dom ϕ ∧
    ∀ x ∈ U, ∀ y ∈ U, ∀ j : I,
      (∑ i : I, K i j * d i x y) ≤ d j (ϕ.func x) (ϕ.func y)

/-- HC02-D02.  The family of restricted co-Lipschitz certificates. -/
def restrictedCoLipCertificates {X : Type} {I : Type u} {bot : X} [Fintype I]
    (d : I → X → X → ENNReal) (ϕ : Operator X bot) (U : Set X) : Set (BoundMatrix I) :=
  {K | IsRestrictedCoLip d ϕ U K}

/-- HC02-D02.  Global co-Lipschitz certificates use `dom ϕ`; in particular,
this family is defined only when that domain is nonempty. -/
def coLipCertificates {X : Type} {I : Type u} {bot : X} [Fintype I]
    (d : I → X → X → ENNReal) (ϕ : Operator X bot) : Set (BoundMatrix I) :=
  restrictedCoLipCertificates d ϕ (dom ϕ)

/-- HC02-P04.  Entrywise infimum over a family of matrix certificates. -/
noncomputable def entryInf {X : Type} {I : Type u} {bot : X} [Fintype I]
    (d : I → X → X → ENNReal) (ϕ : Operator X bot) (U : Set X) (i j : I) : ENNReal :=
  sInf ((fun L : BoundMatrix I => L i j) '' restrictedLipCertificates d ϕ U)

/-- HC02-D07.  A component channel is an input/output pair. -/
abbrev ComponentChannel (I : Type u) := I × I

/-- HC02-D07.  In output row `j`, only input position `i` may be nonzero. -/
def RowSparse {I : Type u} (L : BoundMatrix I) (i j : I) : Prop :=
  ∀ k : I, k ≠ i → L k j = 0

/-- HC02-D07.  Existence of a single supporting input in output row `j`. -/
def RowDecoupled {I : Type u} (L : BoundMatrix I) (j : I) : Prop :=
  ∃ i : I, RowSparse L i j

/-- HC02-D07.  In input column `i`, only output position `j` may be nonzero. -/
def ColumnSparse {I : Type u} (L : BoundMatrix I) (i j : I) : Prop :=
  ∀ k : I, k ≠ j → L i k = 0

/-- HC02-D07.  Existence of a single supporting output in input column `i`. -/
def ColumnDecoupled {I : Type u} (L : BoundMatrix I) (i : I) : Prop :=
  ∃ j : I, ColumnSparse L i j

/-- HC02-D07.  A completely decoupled certificate is diagonal. -/
def Diagonal {I : Type u} (L : BoundMatrix I) : Prop :=
  ∀ i j : I, i ≠ j → L i j = 0

/-- HC02-D08.  Feasible scalar constants for one component channel. -/
def scalarFeasible {X : Type} {I : Type u} {bot : X}
    (d : I → X → X → ENNReal) (ϕ : Operator X bot) (U : Set X) (i j : I) : Set ENNReal :=
  {lam | ∀ x ∈ U, ∀ y ∈ U, d j (ϕ.func x) (ϕ.func y) ≤ lam * d i x y}

/-- HC02-D08.  The optimal single-channel constant.  In `ENNReal`,
`sInf ∅ = ∞`, exactly as required by the text. -/
noncomputable def channelLip {X : Type} {I : Type u} {bot : X}
    (d : I → X → X → ENNReal) (ϕ : Operator X bot) (U : Set X) (i j : I) : ENNReal :=
  sInf (scalarFeasible d ϕ U i j)

/-- HC02-D05.  The actual input domain after zero steps is `U`; after the
next step it is the image under that step. -/
def actualDomain {X : Type} {bot : X} (U : Set X)
    (ϕ : ℕ → Operator X bot) : ℕ → Set X
  | 0 => U
  | n + 1 => (ϕ n).func '' actualDomain U ϕ n

/-- HC02-D04/D05.  The finite operator chain in application order. -/
def operatorChain {X : Type} {bot : X} (ϕ : ℕ → Operator X bot) :
    ℕ → Operator X bot
  | 0 => OperatorChain []
  | n + 1 => OperatorChain [ϕ n, operatorChain ϕ n]

/-- HC02-D05.  Every local Lip bound is imposed on the actual input domain
of its own step; no final-composite bound occurs in this predicate. -/
def LipChainValid {X : Type} {I : Type u} {bot : X} [Fintype I]
    (d : I → X → X → ENNReal) (U : Set X) (ϕ : ℕ → Operator X bot)
    (L : ℕ → BoundMatrix I) : ℕ → Prop
  | 0 => True
  | n + 1 => LipChainValid d U ϕ L n ∧
      L n ∈ restrictedLipCertificates d (ϕ n) (actualDomain U ϕ n)

/-- HC02-D05.  The analogous local co-Lip chain data. -/
def CoLipChainValid {X : Type} {I : Type u} {bot : X} [Fintype I]
    (d : I → X → X → ENNReal) (U : Set X) (ϕ : ℕ → Operator X bot)
    (K : ℕ → BoundMatrix I) : ℕ → Prop
  | 0 => True
  | n + 1 => CoLipChainValid d U ϕ K n ∧
      K n ∈ restrictedCoLipCertificates d (ϕ n) (actualDomain U ϕ n)

/-- HC02-D06.  Complete finite index paths.  `single` is the unique empty
intermediate sequence at length one.  `snoc` appends one final edge; the old
terminal index becomes the last intermediate index. -/
inductive IndexPath (I : Type u) : ℕ → I → I → Type u where
  | single (i j : I) : IndexPath I 1 i j
  | snoc {n : ℕ} {i k : I} (p : IndexPath I n i k) (j : I) : IndexPath I (n + 1) i j

namespace IndexPath

/-- HC02-D06.  The intermediate index sequence `(k₁,...,k_{n-1})`. -/
def intermediate {I : Type u} {n : ℕ} {i j : I} : IndexPath I n i j → List I
  | .single _ _ => []
  | .snoc (k := k) p _ => intermediate p ++ [k]

/-- HC02-D06.  The path weight, namely the product of its edge entries in
application order. -/
noncomputable def weight {I : Type u} (A : ℕ → BoundMatrix I) {n : ℕ} {i j : I} :
    IndexPath I n i j → ENNReal
  | .single i j => A 0 i j
  | .snoc (n := n) (k := k) p j => weight A p * A n k j

/-- HC02-D06.  Every edge on the path is strictly positive. -/
def AllPositive {I : Type u} (A : ℕ → BoundMatrix I) {n : ℕ} {i j : I} :
    IndexPath I n i j → Prop
  | .single i j => 0 < A 0 i j
  | .snoc (n := n) (k := k) p j => AllPositive A p ∧ 0 < A n k j

/-- HC02-D06.  At least one edge on the path is zero. -/
def HasZero {I : Type u} (A : ℕ → BoundMatrix I) {n : ℕ} {i j : I} :
    IndexPath I n i j → Prop
  | .single i j => A 0 i j = 0
  | .snoc (n := n) (k := k) p j => HasZero A p ∨ A n k j = 0

end IndexPath

/-- HC02-D04.  A one-step ordered product is its unique matrix. -/
@[simp] theorem orderedProduct_one {I : Type u} [Fintype I] (A : ℕ → BoundMatrix I) :

/-- HC02-P14.  The identity matrix is a Lip certificate for the empty
operator chain. -/
theorem identity_lip {X I : Type} {bot : X} [Fintype I]

    (hU : U ⊆ dom (OperatorChain ([] : List (Operator X bot)))) :

/-- HC02-P15.  On the same nonempty legal domain, the identity matrix is

theorem identity_coLip {X I : Type} {bot : X} [Fintype I]

/-- HC02-P01.  On a singleton index type the matrix condition is exactly
the displayed scalar inequality. -/
theorem lip_unique {X I : Type} {bot : X} [Fintype I] [Unique I]
    (d : I → X → X → ENNReal) (ϕ : Operator X bot) (U : Set X) (L : BoundMatrix I) :
    L ∈ restrictedLipCertificates d ϕ U ↔
      U ⊆ dom ϕ ∧ ∀ x ∈ U, ∀ y ∈ U,
        d default (ϕ.func x) (ϕ.func y) ≤ L default default * d default x y := by
  constructor
  · intro hL
    refine ⟨hL.1, ?_⟩
    intro x hx y hy
    simpa using hL.2 x hx y hy default
  · rintro ⟨hU, hscalar⟩
    refine ⟨hU, ?_⟩
    intro x hx y hy j
    have hj : j = default := Subsingleton.elim _ _
    subst j
    simpa using hscalar x hx y hy

/-- The same singleton degeneration with an explicitly named unique index. -/
theorem lip_singleton {X I : Type} {bot : X} [Fintype I]
    (d : I → X → X → ENNReal) (ϕ : Operator X bot) (U : Set X) (L : BoundMatrix I)
    (i₀ : I) (hI : ∀ k : I, k = i₀) :
    L ∈ restrictedLipCertificates d ϕ U ↔
      U ⊆ dom ϕ ∧ ∀ x ∈ U, ∀ y ∈ U,
        d i₀ (ϕ.func x) (ϕ.func y) ≤ L i₀ i₀ * d i₀ x y := by
  letI : Unique I := { toInhabited := ⟨i₀⟩, uniq := hI }
  exact lip_unique d ϕ U L

/-- HC02-P02.  A global Lip certificate restricts to every legal subset. -/
theorem lip_global_subset_restricted {X I : Type} {bot : X} [Fintype I]
    (d : I → X → X → ENNReal) (ϕ : Operator X bot) (U : Set X)

    lipCertificates d ϕ ⊆ restrictedLipCertificates d ϕ U := by
  intro L hL
  exact ⟨hU, fun x hx y hy j => hL.2 x (hU hx) y (hU hy) j⟩

/-- HC02-P03.  Shrinking the evaluation domain enlarges the certificate family. -/
theorem restrictedLip_mono {X I : Type} {bot : X} [Fintype I]
    (d : I → X → X → ENNReal) (ϕ : Operator X bot) {U V : Set X}
    (hUV : U ⊆ V) :
    restrictedLipCertificates d ϕ V ⊆ restrictedLipCertificates d ϕ U := by
  intro L hL
  exact ⟨hUV.trans hL.1, fun x hx y hy j => hL.2 x (hUV hx) y (hUV hy) j⟩

/-- HC02-P04.  Entrywise infima do not increase when the evaluation domain

theorem entryInf_mono {X I : Type} {bot : X} [Fintype I]

  exact Set.image_mono (restrictedLip_mono d ϕ hUV)

/-- The extended-nonnegative-real finite-sum calculation shared by the Lip
and co-Lip composition laws. -/
theorem weighted_sum_exchange {I : Type u} [Fintype I]
    (M₂ M₁ : BoundMatrix I) (w : I → ENNReal) (j : I) :
    (∑ k : I, M₂ k j * (∑ i : I, M₁ i k * w i)) =
      ∑ i : I, boundMul M₂ M₁ i j * w i := by
  classical
  simp_rw [boundMul, Finset.mul_sum, Finset.sum_mul, mul_assoc]
  exact Finset.sum_comm

/-- HC02-P05.  Two-step Lipschitz composition on the actual intermediate
image domain. -/
theorem lip_comp_two {X I : Type} {bot : X} [Fintype I]
    (d : I → X → X → ENNReal) (ϕ₁ ϕ₂ : Operator X bot) (U₁ : Set X)
    (L₁ L₂ : BoundMatrix I)
    (hU₁ : U₁ ⊆ dom (OperatorChain [ϕ₂, ϕ₁]))
    (hL₁ : L₁ ∈ restrictedLipCertificates d ϕ₁ U₁)
    (hL₂ : L₂ ∈ restrictedLipCertificates d ϕ₂ (ϕ₁.func '' U₁)) :
    boundMul L₂ L₁ ∈ restrictedLipCertificates d (OperatorChain [ϕ₂, ϕ₁]) U₁ := by
  refine ⟨hU₁, ?_⟩
  intro x hx y hy j
  change d j (ϕ₂.func (ϕ₁.func x)) (ϕ₂.func (ϕ₁.func y)) ≤
    ∑ i : I, boundMul L₂ L₁ i j * d i x y
  calc
    d j (ϕ₂.func (ϕ₁.func x)) (ϕ₂.func (ϕ₁.func y)) ≤
        ∑ k : I, L₂ k j * d k (ϕ₁.func x) (ϕ₁.func y) :=
      hL₂.2 (ϕ₁.func x) ⟨x, hx, rfl⟩ (ϕ₁.func y) ⟨y, hy, rfl⟩ j
    _ ≤ ∑ k : I, L₂ k j * (∑ i : I, L₁ i k * d i x y) := by
      apply Finset.sum_le_sum
      intro k hk
      exact mul_le_mul_left' (hL₁.2 x hx y hy k) _
    _ = ∑ i : I, boundMul L₂ L₁ i j * d i x y :=
      weighted_sum_exchange L₂ L₁ (fun i => d i x y) j

/-- HC02-P07.  Two-step co-Lipschitz composition on the actual intermediate
image domain. -/
theorem coLip_comp_two {X I : Type} {bot : X} [Fintype I]
    (d : I → X → X → ENNReal) (ϕ₁ ϕ₂ : Operator X bot) (U₁ : Set X)
    (K₁ K₂ : BoundMatrix I)
    (hU₁ : U₁ ⊆ dom (OperatorChain [ϕ₂, ϕ₁]))
    (hK₁ : K₁ ∈ restrictedCoLipCertificates d ϕ₁ U₁)
    (hK₂ : K₂ ∈ restrictedCoLipCertificates d ϕ₂ (ϕ₁.func '' U₁)) :
    boundMul K₂ K₁ ∈ restrictedCoLipCertificates d (OperatorChain [ϕ₂, ϕ₁]) U₁ := by
  refine ⟨hK₁.1, hU₁, ?_⟩
  intro x hx y hy j
  change (∑ i : I, boundMul K₂ K₁ i j * d i x y) ≤
    d j (ϕ₂.func (ϕ₁.func x)) (ϕ₂.func (ϕ₁.func y))
  calc
    (∑ i : I, boundMul K₂ K₁ i j * d i x y) =
        ∑ k : I, K₂ k j * (∑ i : I, K₁ i k * d i x y) :=
      (weighted_sum_exchange K₂ K₁ (fun i => d i x y) j).symm
    _ ≤ ∑ k : I, K₂ k j * d k (ϕ₁.func x) (ϕ₁.func y) := by
      apply Finset.sum_le_sum
      intro k hk
      exact mul_le_mul_left' (hK₁.2.2 x hx y hy k) _
    _ ≤ d j (ϕ₂.func (ϕ₁.func x)) (ϕ₂.func (ϕ₁.func y)) :=
      hK₂.2.2 (ϕ₁.func x) ⟨x, hx, rfl⟩ (ϕ₁.func y) ⟨y, hy, rfl⟩ j

/-- HC02-P06.  A zero output distance together with positive input distance
forces the corresponding global co-Lip entry to vanish. -/
theorem coLip_entry_eq_zero_of_output_zero {X I : Type} {bot : X} [Fintype I]
    (d : I → X → X → ENNReal) (ϕ : Operator X bot) (K : BoundMatrix I)
    (hK : K ∈ coLipCertificates d ϕ) (x y : X) (hx : x ∈ dom ϕ) (hy : y ∈ dom ϕ)
    (i j : I) (hout : d j (ϕ.func x) (ϕ.func y) = 0) (hin : 0 < d i x y) :
    K i j = 0 := by
  have hsum : (∑ k : I, K k j * d k x y) = 0 := by
    apply le_antisymm
    · simpa [hout] using hK.2.2 x hx y hy j
    · exact zero_le _
  have hterm_le : K i j * d i x y ≤ ∑ k : I, K k j * d k x y := by
    exact Finset.single_le_sum (fun k _ => zero_le (K k j * d k x y)) (Finset.mem_univ i)
  have hterm : K i j * d i x y = 0 :=
    le_antisymm (hterm_le.trans_eq hsum) (zero_le _)
  by_contra hKne
  have hpos : 0 < K i j * d i x y := ENNReal.mul_pos hKne hin.ne'
  rw [hterm] at hpos
  exact (lt_irrefl 0) hpos

/-- HC02-P08.  Forward point-pair image-distance perturbation. -/
theorem perturbation_forward {X I : Type} {bot : X} [Fintype I]
    (d : I → X → X → ENNReal)
    (hd : ∀ i : I, ExtendedPseudoMetric X bot (d i))
    (ϕ₁ ϕ₂ : Operator X bot) (U : Set X) (δ : I → ENNReal)
    (hδ : ∀ z ∈ U, ∀ j : I, d j (ϕ₁.func z) (ϕ₂.func z) ≤ δ j)
    (x y : X) (hx : x ∈ U) (hy : y ∈ U) (j : I) :
    d j (ϕ₂.func x) (ϕ₂.func y) ≤
      d j (ϕ₁.func x) (ϕ₁.func y) + δ j + δ j := by
  have hδrev : ∀ z ∈ U, ∀ j : I, d j (ϕ₂.func z) (ϕ₁.func z) ≤ δ j := by
    intro z hz k
    rw [(hd k).symm (ϕ₂.func z) (ϕ₁.func z)]
    exact hδ z hz k
  calc
    d j (ϕ₂.func x) (ϕ₂.func y) ≤
        d j (ϕ₂.func x) (ϕ₁.func x) + d j (ϕ₁.func x) (ϕ₂.func y) :=
      (hd j).triangle _ _ _
    _ ≤ d j (ϕ₂.func x) (ϕ₁.func x) +
        (d j (ϕ₁.func x) (ϕ₁.func y) + d j (ϕ₁.func y) (ϕ₂.func y)) :=
      add_le_add (le_refl _) ((hd j).triangle _ _ _)
    _ ≤ δ j + (d j (ϕ₁.func x) (ϕ₁.func y) + δ j) :=
      add_le_add (hδrev x hx j) (add_le_add (le_refl _) (hδ y hy j))
    _ = d j (ϕ₁.func x) (ϕ₁.func y) + δ j + δ j := by
      rw [← add_assoc, add_comm (δ j) (d j (ϕ₁.func x) (ϕ₁.func y)), add_assoc]

/-- HC02-P09.  The reverse point-pair perturbation follows by symmetry. -/
theorem perturbation_reverse {X I : Type} {bot : X} [Fintype I]
    (d : I → X → X → ENNReal)
    (hd : ∀ i : I, ExtendedPseudoMetric X bot (d i))
    (ϕ₁ ϕ₂ : Operator X bot) (U : Set X) (δ : I → ENNReal)
    (hδ : ∀ z ∈ U, ∀ j : I, d j (ϕ₁.func z) (ϕ₂.func z) ≤ δ j)
    (x y : X) (hx : x ∈ U) (hy : y ∈ U) (j : I) :
    d j (ϕ₁.func x) (ϕ₁.func y) ≤
      d j (ϕ₂.func x) (ϕ₂.func y) + δ j + δ j := by
  apply perturbation_forward d hd ϕ₂ ϕ₁ U δ
  · intro z hz k
    rw [(hd k).symm (ϕ₂.func z) (ϕ₁.func z)]
    exact hδ z hz k
  · exact hx
  · exact hy

/-- HC02-P10.  Additive perturbation of a Lip upper bound. -/
theorem perturbation_lip {X I : Type} {bot : X} [Fintype I]
    (d : I → X → X → ENNReal)
    (hd : ∀ i : I, ExtendedPseudoMetric X bot (d i))
    (ϕ₁ ϕ₂ : Operator X bot) (U : Set X) (δ : I → ENNReal) (L₁ : BoundMatrix I)
    (hδ : ∀ z ∈ U, ∀ j : I, d j (ϕ₁.func z) (ϕ₂.func z) ≤ δ j)
    (hL₁ : L₁ ∈ restrictedLipCertificates d ϕ₁ U)
    (x y : X) (hx : x ∈ U) (hy : y ∈ U) (j : I) :
    d j (ϕ₂.func x) (ϕ₂.func y) ≤
      (∑ i : I, L₁ i j * d i x y) + 2 * δ j := by
  calc
    d j (ϕ₂.func x) (ϕ₂.func y) ≤
        d j (ϕ₁.func x) (ϕ₁.func y) + δ j + δ j :=
      perturbation_forward d hd ϕ₁ ϕ₂ U δ hδ x y hx hy j
    _ ≤ (∑ i : I, L₁ i j * d i x y) + δ j + δ j := by
      exact add_le_add (add_le_add (hL₁.2 x hx y hy j) (le_refl _)) (le_refl _)
    _ = (∑ i : I, L₁ i j * d i x y) + 2 * δ j := by rw [two_mul]; ac_rfl

/-- HC02-P11.  Additive perturbation of a co-Lip lower bound. -/
theorem perturbation_coLip {X I : Type} {bot : X} [Fintype I]
    (d : I → X → X → ENNReal)
    (hd : ∀ i : I, ExtendedPseudoMetric X bot (d i))
    (ϕ₁ ϕ₂ : Operator X bot) (U : Set X) (δ : I → ENNReal) (K₁ : BoundMatrix I)
    (hδ : ∀ z ∈ U, ∀ j : I, d j (ϕ₁.func z) (ϕ₂.func z) ≤ δ j)
    (hK₁ : K₁ ∈ restrictedCoLipCertificates d ϕ₁ U)
    (x y : X) (hx : x ∈ U) (hy : y ∈ U) (j : I) :
    (∑ i : I, K₁ i j * d i x y) ≤
      d j (ϕ₂.func x) (ϕ₂.func y) + 2 * δ j := by
  calc
    (∑ i : I, K₁ i j * d i x y) ≤ d j (ϕ₁.func x) (ϕ₁.func y) :=
      hK₁.2.2 x hx y hy j
    _ ≤ d j (ϕ₂.func x) (ϕ₂.func y) + δ j + δ j :=
      perturbation_reverse d hd ϕ₁ ϕ₂ U δ hδ x y hx hy j
    _ = d j (ϕ₂.func x) (ϕ₂.func y) + 2 * δ j := by rw [two_mul]; ac_rfl

/-- HC02-P25.  A zero matrix coefficient removes its own summand, including
when the corresponding distance is infinite. -/
theorem zero_coefficient_term {I X : Type} (L : BoundMatrix I)

/-- HC02-P25.  The zero input position can be deleted from the finite sum. -/
theorem sum_erase_of_coefficient_zero {I X : Type} [Fintype I]

/-- HC02-P26.  A single supported input position simplifies an output row. -/
theorem lip_row_sparse {X I : Type} {bot : X} [Fintype I]

/-- HC02-P28.  A diagonal Lip certificate gives scalar bounds on every
component. -/
theorem diagonal_lip_bounds {X I : Type} {bot : X} [Fintype I]
    (d : I → X → X → ENNReal) (ϕ : Operator X bot) (U : Set X) (L : BoundMatrix I)
    (hL : L ∈ restrictedLipCertificates d ϕ U) (hdiag : Diagonal L)

    d j (ϕ.func x) (ϕ.func y) ≤ L j j * d j x y := by

/-- HC02-P29.  Only row sparsity of the Lip certificate is needed for the
scalar two-sided bound; no sparsity assumption is made on `K`. -/
theorem diagonal_input_two_sided {X I : Type} {bot : X} [Fintype I]
    (d : I → X → X → ENNReal) (ϕ : Operator X bot) (U : Set X)
    (L K : BoundMatrix I)
    (hL : L ∈ restrictedLipCertificates d ϕ U)
    (hK : K ∈ restrictedCoLipCertificates d ϕ U) {j : I}
    (hrow : ∀ k : I, k ≠ j → L k j = 0)
    (x y : X) (hx : x ∈ U) (hy : y ∈ U) :
    K j j * d j x y ≤ d j (ϕ.func x) (ϕ.func y) ∧
      d j (ϕ.func x) (ϕ.func y) ≤ L j j * d j x y := by
  constructor
  · calc
      K j j * d j x y ≤ ∑ k : I, K k j * d k x y := by
        exact Finset.single_le_sum (fun k _ => zero_le (K k j * d k x y)) (Finset.mem_univ j)
      _ ≤ d j (ϕ.func x) (ϕ.func y) := hK.2.2 x hx y hy j
  · exact lip_row_sparse d ϕ U L hL hrow x y hx hy

/-- HC02-P30.  Componentwise two-sided bounds, including the finite-diagonal
coefficient branch. -/
theorem all_components_two_sided_finite {X I : Type} {bot : X} [Fintype I]
    (d : I → X → X → ENNReal) (ϕ : Operator X bot) (U : Set X)
    (L K : BoundMatrix I)
    (hL : L ∈ restrictedLipCertificates d ϕ U)
    (hK : K ∈ restrictedCoLipCertificates d ϕ U)
    (hrows : ∀ j k : I, k ≠ j → L k j = 0)
    (hfinite : ∀ j : I, L j j < ∞ ∧ K j j < ∞) :
    ∀ j : I, K j j < ∞ ∧ L j j < ∞ ∧
      ∀ x ∈ U, ∀ y ∈ U,
        K j j * d j x y ≤ d j (ϕ.func x) (ϕ.func y) ∧
          d j (ϕ.func x) (ϕ.func y) ≤ L j j * d j x y := by

  exact ⟨(hfinite j).2, (hfinite j).1,

/-- HC02-P31.  A positive co-Lip channel preserves distinguishability. -/
theorem positive_coLip_channel {X I : Type} {bot : X} [Fintype I]
    (d : I → X → X → ENNReal) (ϕ : Operator X bot) (U : Set X) (K : BoundMatrix I)
    (hK : K ∈ restrictedCoLipCertificates d ϕ U) {i j : I}
    (hKpos : 0 < K i j) (x y : X) (hx : x ∈ U) (hy : y ∈ U)
    (hdist : 0 < d i x y) :
    0 < d j (ϕ.func x) (ϕ.func y) := by
  have hterm : K i j * d i x y ≤ ∑ k : I, K k j * d k x y := by
    exact Finset.single_le_sum (fun k _ => zero_le (K k j * d k x y)) (Finset.mem_univ i)
  have hpos : 0 < K i j * d i x y := ENNReal.mul_pos hKpos.ne' hdist.ne'
  exact hpos.trans_le (hterm.trans (hK.2.2 x hx y hy j))

/-- HC02-P32.  Positive diagonal co-Lip channels reflect the family zero
relation. -/
theorem positive_diagonal_coLip_reflects {X I : Type} {bot : X} [Fintype I]
    (d : I → X → X → ENNReal) (ϕ : Operator X bot) (U : Set X) (K : BoundMatrix I)
    (hK : K ∈ restrictedCoLipCertificates d ϕ U)
    (hdiag : ∀ j : I, 0 < K j j) (x y : X) (hx : x ∈ U) (hy : y ∈ U)
    (hzero : GaugeZeroRelated d (ϕ.func x) (ϕ.func y)) :
    GaugeZeroRelated d x y := by
  intro j
  by_contra hj
  have hpos : 0 < d j x y := pos_iff_ne_zero.mpr hj
  have hout := positive_coLip_channel d ϕ U K hK (hdiag j) x y hx hy hpos
  exact hout.ne' (hzero j)

/-- HC02-D05.  The image of the whole composite is exactly the recursively
computed actual domain. -/
theorem operatorChain_image {X : Type} {bot : X} (U : Set X)
    (ϕ : ℕ → Operator X bot) (n : ℕ) :
    (operatorChain ϕ n).func '' U = actualDomain U ϕ n := by
  induction n with
  | zero =>
      simp [operatorChain, actualDomain, OperatorChain]
  | succ n ih =>
      calc
        (operatorChain ϕ (n + 1)).func '' U =
            ((ϕ n).func ∘ (operatorChain ϕ n).func) '' U := rfl
        _ = (ϕ n).func '' ((operatorChain ϕ n).func '' U) := Set.image_comp _ _ _
        _ = actualDomain U ϕ (n + 1) := by rw [ih]; rfl

/-- A positive ordered-product entry supplies a complete all-positive path. -/
theorem exists_allPositive_path_of_orderedProduct_pos {I : Type} [Fintype I]
    (A : ℕ → BoundMatrix I) (n : ℕ) (i j : I)
    (h : 0 < orderedProduct A (n + 1) i j) :
    ∃ p : IndexPath I (n + 1) i j, p.AllPositive A := by
  induction n generalizing i j with
  | zero =>
      exact ⟨IndexPath.single i j, by simpa using h⟩
  | succ n ih =>
      change 0 < ∑ k : I, A (n + 1) k j * orderedProduct A (n + 1) i k at h
      have hex : ∃ k : I, 0 < A (n + 1) k j * orderedProduct A (n + 1) i k := by
        simpa using (Finset.sum_pos_iff.mp h)
      obtain ⟨k, hk⟩ := hex
      have hparts := ENNReal.mul_pos_iff.mp hk
      obtain ⟨p, hp⟩ := ih i k hparts.2
      exact ⟨IndexPath.snoc p j, hp, hparts.1⟩

/-- Every all-positive path forces the corresponding ordered-product entry
to be positive. -/
theorem orderedProduct_pos_of_allPositive_path {I : Type} [Fintype I]
    (A : ℕ → BoundMatrix I) {n : ℕ} {i j : I}
    (p : IndexPath I n i j) (hp : p.AllPositive A) :
    0 < orderedProduct A n i j := by
  induction p with
  | single i j => simpa using hp
  | @snoc n i k p j ih =>
      change 0 < ∑ l : I, A n l j * orderedProduct A n i l
      apply Finset.sum_pos_iff.mpr
      refine ⟨k, Finset.mem_univ k, ?_⟩
      exact ENNReal.mul_pos_iff.mpr ⟨hp.2, ih hp.1⟩

/-- HC02-P17, algebraic form.  Positivity of an ordered-product entry is
equivalent to existence of one all-positive complete path. -/
theorem orderedProduct_pos_iff {I : Type} [Fintype I]
    (A : ℕ → BoundMatrix I) (n : ℕ) (i j : I) :
    0 < orderedProduct A (n + 1) i j ↔
      ∃ p : IndexPath I (n + 1) i j, p.AllPositive A := by
  constructor
  · exact exists_allPositive_path_of_orderedProduct_pos A n i j
  · rintro ⟨p, hp⟩
    exact orderedProduct_pos_of_allPositive_path A p hp

/-- HC02-P16, algebraic form.  Vanishing is equivalent to every complete
path containing at least one zero edge. -/
theorem orderedProduct_zero_iff {I : Type} [Fintype I]
    (A : ℕ → BoundMatrix I) (n : ℕ) (i j : I) :
    orderedProduct A (n + 1) i j = 0 ↔
      ∀ p : IndexPath I (n + 1) i j, p.HasZero A := by
  constructor
  · intro h p
    apply (IndexPath.hasZero_iff_not_allPositive A p).mpr
    intro hp
    have hpos := (orderedProduct_pos_iff A n i j).mpr ⟨p, hp⟩
    rw [h] at hpos
    exact (lt_irrefl 0) hpos
  · intro h
    by_contra hne
    obtain ⟨p, hp⟩ := (orderedProduct_pos_iff A n i j).mp (pos_iff_ne_zero.mpr hne)
    have hz := h p
    exact (IndexPath.hasZero_iff_not_allPositive A p).mp hz hp

  · rw [hBtop, ENNReal.mul_top ha.ne']

  · have hBpos : 0 < d i x y := pos_iff_ne_zero.mpr hBzero
    by_contra hnot

    have hadiv : channelLip d ϕ U i j < d j (ϕ.func x) (ϕ.func y) / d i x y :=

/-! ### The zero-infimum counterexample -/

/-- HC02-P23.  Two nonabsorbing points and one absorbing point. -/
abbrev CounterPoint := Option Bool

/-- The absorbing point of the counterexample. -/
def counterBot : CounterPoint := none

/-- Distinct points are at infinite extended distance. -/
def counterDist : CounterPoint → CounterPoint → ENNReal :=
  fun x y => if x = y then 0 else ∞

/-- HC02-P23.  This is a legal extended pseudometric with the required
absorbing-point isolation. -/
theorem counterDist_extendedPseudoMetric :
    ExtendedPseudoMetric CounterPoint counterBot counterDist := by
  refine
    { nonabsorbing := ⟨some false, by simp [counterBot]⟩
      dist_self := ?_
      symm := ?_
      triangle := ?_
      isolate_bot := ?_ }
  · intro x
    simp [counterDist]
  · intro x y
    fin_cases x <;> fin_cases y <;> simp [counterDist]
  · intro x y z
    fin_cases x <;> fin_cases y <;> fin_cases z <;> simp [counterDist]
  · intro x hx
    fin_cases x <;> simp_all [counterDist, counterBot]

/-- The identity operator on the counterexample space. -/
def counterOperator : Operator CounterPoint counterBot := ⟨id, rfl⟩

/-- A nonempty legal evaluation domain containing both nonabsorbing points. -/
def counterU : Set CounterPoint := {some false, some true}

theorem counterU_nonempty : counterU.Nonempty := by
  exact ⟨some false, by simp [counterU]⟩

theorem counterU_subset_dom : counterU ⊆ dom counterOperator := by
  intro x hx
  rcases (by simpa [counterU] using hx : x = some false ∨ x = some true) with rfl | rfl <;>
    simp [dom, counterOperator, counterBot]

/-- HC02-P23.  For the distinguished channel, the feasible scalars are
exactly the nonzero extended nonnegative reals. -/
theorem counter_scalarFeasible (lam : ENNReal) :
    lam ∈ scalarFeasible (fun _ : Unit => counterDist) counterOperator counterU () () ↔
      lam ≠ 0 := by
  constructor
  · intro hlam
    intro hzero
    subst lam
    have hbound := hlam (some false) (by simp [counterU])
      (some true) (by simp [counterU])
    simp [counterOperator, counterDist] at hbound
  · intro hne x hx y hy
    have hx' : x = some false ∨ x = some true := by simpa [counterU] using hx
    have hy' : y = some false ∨ y = some true := by simpa [counterU] using hy
    rcases hx' with rfl | rfl <;> rcases hy' with rfl | rfl <;>
      simp [counterOperator, counterDist, hne]

/-- The feasible set is precisely `(0, +∞]` in `ENNReal`. -/

    scalarFeasible (fun _ : Unit => counterDist) counterOperator counterU () () = Set.Ioi 0 := by
  ext lam
  simp [counter_scalarFeasible, pos_iff_ne_zero]
