/-! ### General helpers about extended pseudometrics and operators -/

theorem epm_zero_trans {d : X → X → ENNReal} (hd : ExtendedPseudoMetric X bot d)
    {x y z : X} (h1 : d x y = 0) (h2 : d y z = 0) : d x z = 0 :=
  le_antisymm (by simpa [h1, h2] using hd.triangle x y z) (zero_le _)

theorem epm_eq_of_zero_left {d : X → X → ENNReal} (hd : ExtendedPseudoMetric X bot d)
    {x x' : X} (h : d x x' = 0) (y : X) : d x y = d x' y := by
  apply le_antisymm
  · calc d x y ≤ d x x' + d x' y := hd.triangle _ _ _
      _ = d x' y := by rw [h, zero_add]
  · calc d x' y ≤ d x' x + d x y := hd.triangle _ _ _
      _ = d x y := by rw [hd.symm, h, zero_add]

theorem epm_eq_of_zero_right {d : X → X → ENNReal} (hd : ExtendedPseudoMetric X bot d)
    {y y' : X} (h : d y y' = 0) (x : X) : d x y = d x y' := by
  rw [hd.symm x y, hd.symm x y']; exact epm_eq_of_zero_left hd h x

@[simp] theorem mul_func (ϕ₂ ϕ₁ : Operator X bot) (x : X) :
    (ϕ₂ * ϕ₁).func x = ϕ₂.func (ϕ₁.func x) := rfl

@[simp] theorem one_func (x : X) : (1 : Operator X bot).func x = x := rfl

/-- 复合定义直接保证 -/
theorem mem_dom_mul {ϕ₂ ϕ₁ : Operator X bot} {x : X} (hx : x ∈ dom (ϕ₂ * ϕ₁)) :
    x ∈ dom ϕ₁ ∧ ϕ₁.func x ∈ dom ϕ₂ := by
  refine ⟨fun h => hx ?_, hx⟩
  change ϕ₂.func (ϕ₁.func x) = bot
  rw [h, ϕ₂.preserves_bot]

theorem image_subset_dom_of_subset_dom_mul {ϕ₂ ϕ₁ : Operator X bot} {U : Set X}
    (hU : U ⊆ dom (ϕ₂ * ϕ₁)) : ϕ₁.func '' U ⊆ dom ϕ₂ := by
  rintro _ ⟨x, hx, rfl⟩; exact (mem_dom_mul (hU hx)).2

/-- 子族复合伪度量为零当且仅当子族每个分量为零。 -/
theorem subfamily_pseudometric_eq_zero_iff (d : I → X → X → ENNReal) (J : Set I) (a b : X) :
    subfamily_pseudometric d J a b = 0 ↔ ∀ j ∈ J, d j a b = 0 := by
  simp [subfamily_pseudometric, ENNReal.iSup_eq_zero]

theorem subfamily_pseudometric_self (d : I → X → X → ENNReal)
    (hd : ∀ i, ExtendedPseudoMetric X bot (d i)) (J : Set I) (a : X) :
    subfamily_pseudometric d J a a = 0 :=
  (subfamily_pseudometric_eq_zero_iff d J a a).2 fun j _ => (hd j).dist_self a

theorem subfamily_pseudometric_comm (d : I → X → X → ENNReal)
    (hd : ∀ i, ExtendedPseudoMetric X bot (d i)) (J : Set I) (a b : X) :
    subfamily_pseudometric d J a b = subfamily_pseudometric d J b a := by
  simp only [subfamily_pseudometric]; congr with j; exact (hd j.1).symm a b

theorem le_subfamily_pseudometric (d : I → X → X → ENNReal) {J : Set I} {j : I} (hj : j ∈ J)
    (a b : X) : d j a b ≤ subfamily_pseudometric d J a b :=
  le_iSup (fun k : {k // k ∈ J} => d k.1 a b) ⟨j, hj⟩

/-- 有限非空子族上 `d_J = max_{j∈J} d_j`：冻结的上确界实现由某个 `j ∈ J` 取到。 -/
theorem subfamily_pseudometric_eq_max [Finite I] (d : I → X → X → ENNReal) (J : Set I)
    (hJ : J.Nonempty) (a b : X) :
    ∃ j ∈ J, subfamily_pseudometric d J a b = d j a b ∧ ∀ j' ∈ J, d j' a b ≤ d j a b := by
  obtain ⟨j, hj, hmax⟩ := Set.exists_max_image J (fun j => d j a b) (Set.toFinite J) hJ
  refine ⟨j, hj, le_antisymm ?_ (le_subfamily_pseudometric d hj a b), hmax⟩
  show (⨆ k : {k // k ∈ J}, d k.1 a b) ≤ d j a b
  exact iSup_le fun k => hmax k.1 k.2

/-! ### CH03-D01 分量分划 -/

/-- 常值分量 `I_const(ϕ|U)`。 -/
def Iconst (d : I → X → X → ENNReal) (ϕ : Operator X bot) (U : Set X) : Set I :=
  {i | ∀ x ∈ U, ∀ y ∈ U, d i (ϕ.func x) (ϕ.func y) = 0}

/-- 恒等分量 `I_id(ϕ|U)`：不在常值分量中（常值优先），且逐点输出与输入零距。 -/
def Iid (d : I → X → X → ENNReal) (ϕ : Operator X bot) (U : Set X) : Set I :=
  {i | i ∉ Iconst d ϕ U ∧ ∀ x ∈ U, d i (ϕ.func x) x = 0}

/-- 有效分量 `I_act(ϕ|U) = I \ (I_const ∪ I_id)`（`I` 由全集 `univ` 表示）。 -/
def Iact (d : I → X → X → ENNReal) (ϕ : Operator X bot) (U : Set X) : Set I :=
  univ \ (Iconst d ϕ U ∪ Iid d ϕ U)

/-- 全域分划（`U = dom ϕ`，省略限制后缀）。 -/
abbrev IconstG (d : I → X → X → ENNReal) (ϕ : Operator X bot) : Set I := Iconst d ϕ (dom ϕ)
abbrev IidG (d : I → X → X → ENNReal) (ϕ : Operator X bot) : Set I := Iid d ϕ (dom ϕ)
abbrev IactG (d : I → X → X → ENNReal) (ϕ : Operator X bot) : Set I := Iact d ϕ (dom ϕ)

theorem mem_Iact_iff {d : I → X → X → ENNReal} {ϕ : Operator X bot} {U : Set X} {i : I} :
    i ∈ Iact d ϕ U ↔ i ∉ Iconst d ϕ U ∧ ¬ ∀ x ∈ U, d i (ϕ.func x) x = 0 := by
  simp only [Iact, Iid, mem_diff, mem_univ, mem_union, mem_setOf_eq, true_and]; tauto

theorem not_mem_Iact_iff {d : I → X → X → ENNReal} {ϕ : Operator X bot} {U : Set X} {i : I} :
    i ∉ Iact d ϕ U ↔ i ∈ Iconst d ϕ U ∨ ∀ x ∈ U, d i (ϕ.func x) x = 0 := by
  rw [mem_Iact_iff]; tauto

/-- 三类分量两两不交且覆盖 `I`（分划的表示对应）。 -/
theorem component_partition (d : I → X → X → ENNReal) (ϕ : Operator X bot) (U : Set X) :
    Iconst d ϕ U ∪ Iid d ϕ U ∪ Iact d ϕ U = univ ∧
    Disjoint (Iconst d ϕ U) (Iid d ϕ U) ∧ Disjoint (Iconst d ϕ U) (Iact d ϕ U) ∧
    Disjoint (Iid d ϕ U) (Iact d ϕ U) := by
  refine ⟨?_, ?_, ?_, ?_⟩
  · ext i; simp only [Iact, mem_union, mem_diff, mem_univ, true_and, iff_true]; tauto
  · rw [Set.disjoint_left]; intro i h1 h2; exact h2.1 h1
  · rw [Set.disjoint_left]; intro i h1 h2; exact h2.2 (Or.inl h1)
  · rw [Set.disjoint_left]; intro i h1 h2; exact h2.2 (Or.inr h1)

/-! ### CH03-D02 计算纤维 -/

/-- 计算纤维 `Fib_{ϕ,J}(y|U) = {x ∈ U | d_J(ϕ x, y) = 0}`，`d_J` 为第 1 章子族复合伪度量。 -/
noncomputable def Fib (d : I → X → X → ENNReal) (ϕ : Operator X bot) (J : Set I) (y : X)
    (U : Set X) : Set X :=
  {x | x ∈ U ∧ subfamily_pseudometric d J (ϕ.func x) y = 0}

/-- 单分量计算纤维 `Fib_{ϕ,j}(y|U) = {x ∈ U | d_j(ϕ x, y) = 0}`。 -/
def Fib1 (d : I → X → X → ENNReal) (ϕ : Operator X bot) (j : I) (y : X) (U : Set X) : Set X :=
  {x | x ∈ U ∧ d j (ϕ.func x) y = 0}

/-- 全域纤维 `Fib_{ϕ,J}(y)`（`U = dom ϕ`）。 -/
noncomputable abbrev FibG (d : I → X → X → ENNReal) (ϕ : Operator X bot) (J : Set I) (y : X) :
    Set X := Fib d ϕ J y (dom ϕ)

theorem mem_Fib_iff {d : I → X → X → ENNReal} {ϕ : Operator X bot} {J : Set I} {y x : X}
    {U : Set X} : x ∈ Fib d ϕ J y U ↔ x ∈ U ∧ ∀ j ∈ J, d j (ϕ.func x) y = 0 := by
  simp [Fib, subfamily_pseudometric_eq_zero_iff]

/-- 单元素子族的纤维即单分量纤维（命题 3.5(v) 的单分量等式）。 -/
theorem Fib_singleton (d : I → X → X → ENNReal) (ϕ : Operator X bot) (j : I) (y : X)
    (U : Set X) : Fib d ϕ {j} y U = Fib1 d ϕ j y U := by
  ext x; simp [mem_Fib_iff, Fib1]

/-- 目标与 `ϕ(U)` 中每一点都有正 `d_J`-距离时，相应纤维为空。 -/
theorem Fib_eq_empty_of_pos (d : I → X → X → ENNReal) (ϕ : Operator X bot) (J : Set I) (y : X)
    (U : Set X) (h : ∀ x ∈ U, 0 < subfamily_pseudometric d J (ϕ.func x) y) :
    Fib d ϕ J y U = ∅ := by
  rw [eq_empty_iff_forall_notMem]
  rintro x ⟨hx, h0⟩
  exact (h x hx).ne' h0

/-! ### CH03-D03 点值/零距点对关系、像商、纤维预序 -/

/-- `R_=(ϕ|U) = {(x,y) ∈ U × U | ϕ x = ϕ y}`。 -/
def Req (ϕ : Operator X bot) (U : Set X) : Set (X × X) :=
  {p | p.1 ∈ U ∧ p.2 ∈ U ∧ ϕ.func p.1 = ϕ.func p.2}

/-- `R_∼(ϕ|U) = {(x,y) ∈ U × U | ϕ x ∼_G ϕ y}`。 -/
def Rsim (d : I → X → X → ENNReal) (ϕ : Operator X bot) (U : Set X) : Set (X × X) :=
  {p | p.1 ∈ U ∧ p.2 ∈ U ∧ GaugeZeroRelated d (ϕ.func p.1) (ϕ.func p.2)}

/-- 第二类点对集是计算纤维的关系表达 -/
theorem mem_Rsim_iff_mem_Fib (d : I → X → X → ENNReal) (hd : ∀ i, ExtendedPseudoMetric X bot (d i))
    (ϕ : Operator X bot) (U : Set X) {x y : X} (hx : x ∈ U) (hy : y ∈ U) :
    (x, y) ∈ Rsim d ϕ U ↔ y ∈ Fib d ϕ univ (ϕ.func x) U := by
  simp only [Rsim, mem_setOf_eq, mem_Fib_iff, mem_univ, true_implies, GaugeZeroRelated]
  constructor
  · rintro ⟨-, -, h⟩; exact ⟨hy, fun j => by rw [(hd j).symm]; exact h j⟩
  · rintro ⟨-, h⟩; exact ⟨hx, hy, fun j => by rw [(hd j).symm]; exact h j⟩

theorem Req_subset_Rsim (d : I → X → X → ENNReal) (hd : ∀ i, ExtendedPseudoMetric X bot (d i))
    (ϕ : Operator X bot) (U : Set X) : Req ϕ U ⊆ Rsim d ϕ U := by
  rintro ⟨x, y⟩ ⟨hx, hy, h⟩
  exact ⟨hx, hy, fun i => by simp only at h ⊢; rw [h]; exact (hd i).dist_self _⟩

/-- 族零距关系的 setoid（由第 1 章引理 1.1 给出等价性）。 -/
def gaugeSetoid (d : I → X → X → ENNReal) (hd : ∀ i, ExtendedPseudoMetric X bot (d i)) :
    Setoid X :=
  ⟨GaugeZeroRelated d, gauge_zero_related_equivalence d (fun i => (hd i).dist_self)
    (fun i => (hd i).symm) (fun i => (hd i).triangle)⟩

/-- 像商 `ϕ(U)/∼_G`：实际像集（子类型）按族零距关系取商。 -/
abbrev ImageQuot (d : I → X → X → ENNReal) (hd : ∀ i, ExtendedPseudoMetric X bot (d i))
    (ϕ : Operator X bot) (U : Set X) : Type :=
  Quotient ((gaugeSetoid d hd).comap (Subtype.val : ↥(ϕ.func '' U) → X))

/-- 像类 `[ϕ(x)] ∈ ϕ(U)/∼_G`（`x ∈ U`）。 -/
def imgCls (d : I → X → X → ENNReal) (hd : ∀ i, ExtendedPseudoMetric X bot (d i))
    (ϕ : Operator X bot) (U : Set X) (x : X) (hx : x ∈ U) : ImageQuot d hd ϕ U :=
  Quotient.mk _ ⟨ϕ.func x, mem_image_of_mem _ hx⟩

theorem imgCls_eq_iff (d : I → X → X → ENNReal) (hd : ∀ i, ExtendedPseudoMetric X bot (d i))
    (ϕ : Operator X bot) (U : Set X) {x y : X} (hx : x ∈ U) (hy : y ∈ U) :
    imgCls d hd ϕ U x hx = imgCls d hd ϕ U y hy ↔ GaugeZeroRelated d (ϕ.func x) (ϕ.func y) :=
  Quotient.eq (r := (gaugeSetoid d hd).comap (Subtype.val : ↥(ϕ.func '' U) → X))

/-- 零测距纤维预序 `ϕA ⪰_{fiber,U} ϕB :⇔ R_∼(ϕA|U) ⊆ R_∼(ϕB|U)`。 -/
def fiberGE (d : I → X → X → ENNReal) (U : Set X) (ϕA ϕB : Operator X bot) : Prop :=
  Rsim d ϕA U ⊆ Rsim d ϕB U

/-! ### CH03-D04 点态零化内切半径 -/

/-- 合法有限正半径集 `{r ∈ (0,+∞) | B_{d_in}(y,r) ∩ U ⊆ {x ∈ U | d_out(ϕ x, ϕ y) = 0}}`。 -/
def admRadii (din dout : X → X → ENNReal) (ϕ : Operator X bot) (y : X) (U : Set X) :
    Set ENNReal :=
  {r | 0 < r ∧ r < ⊤ ∧ ball din y r ∩ U ⊆ {x | x ∈ U ∧ dout (ϕ.func x) (ϕ.func y) = 0}}

/-- 零化内切半径 `r^N_{ϕ,d_in,d_out}(y|U) = sup(admRadii)`，`sup ∅ = 0`，可取 `+∞`。 -/
noncomputable def nullRadius (din dout : X → X → ENNReal) (ϕ : Operator X bot) (y : X)
    (U : Set X) : ENNReal :=
  sSup (admRadii din dout ϕ y U)

/-- 简写 `r^N_{ϕ,i,J}(y|U) := r^N_{ϕ,d_i,d_J}(y|U)`。 -/
noncomputable abbrev rNJ (d : I → X → X → ENNReal) (ϕ : Operator X bot) (i : I) (J : Set I)
    (y : X) (U : Set X) : ENNReal :=
  nullRadius (d i) (subfamily_pseudometric d J) ϕ y U

/-- 简写 `r^N_{ϕ,i,j}(y|U) := r^N_{ϕ,d_i,d_j}(y|U)`。 -/
noncomputable abbrev rN (d : I → X → X → ENNReal) (ϕ : Operator X bot) (i j : I) (y : X)
    (U : Set X) : ENNReal :=
  nullRadius (d i) (d j) ϕ y U

/-- 全域简写 `r^N_{ϕ,d_in,d_out}(y) := r^N(y | dom ϕ)`。 -/
noncomputable abbrev nullRadiusG (din dout : X → X → ENNReal) (ϕ : Operator X bot) (y : X) :
    ENNReal :=
  nullRadius din dout ϕ y (dom ϕ)

theorem rNJ_singleton (d : I → X → X → ENNReal) (ϕ : Operator X bot) (i j : I) (y : X)
    (U : Set X) : rNJ d ϕ i {j} y U = rN d ϕ i j y U := by
  simp only [rNJ, rN, subfamily_pseudometric_singleton]

/-- 输出取 `d_J` 时，输出零测距集合就是计算纤维 `Fib_{ϕ,J}(ϕ y | U)`。 -/
theorem zeroSet_eq_Fib (d : I → X → X → ENNReal) (ϕ : Operator X bot) (J : Set I) (y : X)
    (U : Set X) :
    {x | x ∈ U ∧ subfamily_pseudometric d J (ϕ.func x) (ϕ.func y) = 0} = Fib d ϕ J (ϕ.func y) U :=
  rfl

theorem zeroSet_eq_Fib1 (d : I → X → X → ENNReal) (ϕ : Operator X bot) (j : I) (y : X)
    (U : Set X) :
    {x | x ∈ U ∧ d j (ϕ.func x) (ϕ.func y) = 0} = Fib1 d ϕ j (ϕ.func y) U := rfl

section NullRadiusCore
variable {din dout : X → X → ENNReal} {ϕ : Operator X bot} {y : X} {U : Set X}

theorem admRadii_mono {r r' : ENNReal} (hr : 0 < r) (hrr' : r ≤ r')
    (h : r' ∈ admRadii din dout ϕ y U) : r ∈ admRadii din dout ϕ y U := by
  refine ⟨hr, lt_of_le_of_lt hrr' h.2.1, fun x hx => h.2.2 ⟨lt_of_lt_of_le hx.1 hrr', hx.2⟩⟩

theorem le_nullRadius {r : ENNReal} (h : r ∈ admRadii din dout ϕ y U) :
    r ≤ nullRadius din dout ϕ y U := le_sSup h

/-- 约定 `sup ∅ = 0`：不存在合法有限正半径时零化半径为零。 -/
theorem nullRadius_of_admRadii_empty (h : admRadii din dout ϕ y U = ∅) :
    nullRadius din dout ϕ y U = 0 := by
  simp [nullRadius, h]

/-- 严格小于点态上确界时，可提取合法有限正半径。 -/
theorem exists_adm_of_lt_nullRadius {a : ENNReal} (h : a < nullRadius din dout ϕ y U) :
    ∃ ρ ∈ admRadii din dout ϕ y U, a < ρ := by
  simpa using (lt_sSup_iff.1 h)

/-- 核心判据 -/
theorem zero_of_lt_nullRadius {x : X} (hx : x ∈ U) (h : din x y < nullRadius din dout ϕ y U) :
    dout (ϕ.func x) (ϕ.func y) = 0 := by
  obtain ⟨ρ, hρ, hlt⟩ := exists_adm_of_lt_nullRadius h
  exact (hρ.2.2 ⟨hlt, hx⟩).2

/-- 若所有严格小于 `M` 的有限正半径都合法，则 `M ≤ r^N`。 -/
theorem le_nullRadius_of_forall {M : ENNReal}
    (h : ∀ r : ENNReal, 0 < r → r < ⊤ → r < M →
      ball din y r ∩ U ⊆ {x | x ∈ U ∧ dout (ϕ.func x) (ϕ.func y) = 0}) :
    M ≤ nullRadius din dout ϕ y U := by
  apply le_of_forall_lt
  intro c hc
  obtain ⟨c', hcc', hc'M⟩ := exists_between hc
  have hpos : 0 < c' := lt_of_le_of_lt (zero_le _) hcc'
  have hfin : c' < ⊤ := lt_of_lt_of_le hc'M le_top
  exact lt_of_lt_of_le hcc' (le_nullRadius ⟨hpos, hfin, h c' hpos hfin hc'M⟩)

/-- 所有有限正半径均合法时 `r^N = +∞`。 -/
theorem nullRadius_eq_top_of_forall
    (h : ∀ r : ENNReal, 0 < r → r < ⊤ →
      ball din y r ∩ U ⊆ {x | x ∈ U ∧ dout (ϕ.func x) (ϕ.func y) = 0}) :
    nullRadius din dout ϕ y U = ⊤ :=
  top_unique (le_nullRadius_of_forall fun r h0 h1 _ => h r h0 h1)

end NullRadiusCore

/-! ### CH03-D05 统一零化半径与零化矩阵 -/

/-- 统一零化半径 `r̲^N_{ϕ,d_in,d_out}(A|U) = inf_{y∈A} r^N_{ϕ,d_in,d_out}(y|U)`。 -/
noncomputable def uniformNullRadius (din dout : X → X → ENNReal) (ϕ : Operator X bot)
    (A U : Set X) : ENNReal :=
  ⨅ y ∈ A, nullRadius din dout ϕ y U

/-- 简写 `r̲^N_{ϕ,d_in,d_out}|_U := r̲^N(U|U)`。 -/
noncomputable abbrev uNR (din dout : X → X → ENNReal) (ϕ : Operator X bot) (U : Set X) :
    ENNReal := uniformNullRadius din dout ϕ U U

/-- 全域简写 `r̲^N_{ϕ,d_in,d_out} := r̲^N(dom ϕ | dom ϕ)`。 -/
noncomputable abbrev uNRG (din dout : X → X → ENNReal) (ϕ : Operator X bot) : ENNReal :=
  uniformNullRadius din dout ϕ (dom ϕ) (dom ϕ)

/-- 零化矩阵 `[N(ϕ)|_U]_{ji} := r̲^N_{ϕ,i,j}|_U`：第一个参数 `j` 为输出（行），第二个 `i` 为输入（列）。 -/
noncomputable def nullMatrix (d : I → X → X → ENNReal) (ϕ : Operator X bot) (U : Set X) :
    I → I → ENNReal :=
  fun j i => uNR (d i) (d j) ϕ U

/-- 全域零化矩阵 `N(ϕ)`。 -/
noncomputable abbrev nullMatrixG (d : I → X → X → ENNReal) (ϕ : Operator X bot) : I → I → ENNReal :=
  nullMatrix d ϕ (dom ϕ)

theorem uniform_le_nullRadius {din dout : X → X → ENNReal} {ϕ : Operator X bot} {A U : Set X}
    {y : X} (hy : y ∈ A) : uniformNullRadius din dout ϕ A U ≤ nullRadius din dout ϕ y U :=
  iInf₂_le y hy

theorem le_uniform {din dout : X → X → ENNReal} {ϕ : Operator X bot} {A U : Set X} {M : ENNReal}
    (h : ∀ y ∈ A, M ≤ nullRadius din dout ϕ y U) : M ≤ uniformNullRadius din dout ϕ A U :=
  le_iInf₂ h

/-! ### CH03-D06 输入尺度函数族与阈值分类 -/

/-- 输入尺度函数族：`σ_i` 在非空子集上关于包含单调（空集上的取值不被使用）。 -/
structure InputScaleFamily (X I : Type) where
  σ : I → Set X → ENNReal
  mono : ∀ (i : I) (A₁ A₂ : Set X), A₁.Nonempty → A₁ ⊆ A₂ → σ i A₁ ≤ σ i A₂

open Classical in
/-- 表示对应：仅定义在非空子集上的单调映射族 `τ_i : {A | A ≠ ∅} → R̄₊` 给出输入尺度函数族
（空集上取 0，该值不进入任何陈述）。 -/
noncomputable def InputScaleFamily.ofNonempty (τ : I → {A : Set X // A.Nonempty} → ENNReal)
    (hmono : ∀ (i : I) (A₁ A₂ : {A : Set X // A.Nonempty}), A₁.1 ⊆ A₂.1 → τ i A₁ ≤ τ i A₂) :
    InputScaleFamily X I where
  σ i A := if h : A.Nonempty then τ i ⟨A, h⟩ else 0
  mono i A₁ A₂ h₁ h := by
    have h₂ : A₂.Nonempty := h₁.mono h
    show (if h : A₁.Nonempty then τ i ⟨A₁, h⟩ else 0) ≤ (if h : A₂.Nonempty then τ i ⟨A₂, h⟩ else 0)
    rw [dif_pos h₁, dif_pos h₂]
    exact hmono i ⟨A₁, h₁⟩ ⟨A₂, h₂⟩ h

open Classical in
/-- `ofNonempty` 在每个非空子集上取回原映射。 -/
theorem InputScaleFamily.ofNonempty_eq (τ : I → {A : Set X // A.Nonempty} → ENNReal)
    (hmono : ∀ (i : I) (A₁ A₂ : {A : Set X // A.Nonempty}), A₁.1 ⊆ A₂.1 → τ i A₁ ≤ τ i A₂)
    (i : I) (A : {A : Set X // A.Nonempty}) :
    (InputScaleFamily.ofNonempty τ hmono).σ i A.1 = τ i A := by
  show (if h : A.1.Nonempty then τ i ⟨A.1, h⟩ else 0) = τ i A
  rw [dif_pos A.2]

/-- 反向：任一输入尺度函数族限制到非空子集上即为原文意义的单调映射族。 -/
theorem InputScaleFamily.restrict_mono (σ : InputScaleFamily X I) (i : I)
    (A₁ A₂ : {A : Set X // A.Nonempty}) (h : A₁.1 ⊆ A₂.1) : σ.σ i A₁.1 ≤ σ.σ i A₂.1 :=
  σ.mono i A₁.1 A₂.1 A₁.2 h

/-- 直径 `diam_d(A) = sup_{x,y∈A} d(x,y)`。 -/
noncomputable def diam (d : X → X → ENNReal) (A : Set X) : ENNReal := ⨆ x ∈ A, ⨆ y ∈ A, d x y

theorem diam_mono (d : X → X → ENNReal) {A B : Set X} (h : A ⊆ B) : diam d A ≤ diam d B :=
  iSup₂_le fun x hx => iSup₂_le fun y hy => le_iSup₂_of_le x (h hx) (le_iSup₂_of_le y (h hy) le_rfl)

theorem le_diam (d : X → X → ENNReal) {A : Set X} {x y : X} (hx : x ∈ A) (hy : y ∈ A) :
    d x y ≤ diam d A := le_iSup₂_of_le x hx (le_iSup₂_of_le y hy le_rfl)

/-- 默认直径尺度 `σ_i = diam_{d_i}`，它是输入尺度函数族。 -/
noncomputable def diamScale (d : I → X → X → ENNReal) : InputScaleFamily X I where
  σ i A := diam (d i) A
  mono i _ _ _ h := diam_mono (d i) h

/-- 阈值分类 `Type_{f,A}(i,j) = (1{σ_i(A) < τσ}, 1{r̲^N_{f,i,j}|_A ≥ τN}) ∈ {0,1}²`。 -/
noncomputable def thresholdType (σ : InputScaleFamily X I) (τσ τN : ENNReal)

/-- 全有界：`∀ r ∈ (0,+∞), N(r, U) < +∞`。 -/
def TotallyBoundedIn (d : X → X → ENNReal) (U : Set X) : Prop :=
  ∀ r : ENNReal, 0 < r → r < ⊤ → covering_number d r U < ⊤

theorem log2_ereal_mono {a b : ENNReal} (h : a ≤ b) : log2_ereal a ≤ log2_ereal b := by
  unfold log2_ereal
  by_cases ha0 : a = 0
  · simp [ha0]
  by_cases hbt : b = ⊤
  · simp [hbt]
  have hb0 : b ≠ 0 := fun hb => ha0 (le_antisymm (hb ▸ h) (zero_le _))
  have hat : a ≠ ⊤ := fun ha => hbt (top_unique (ha ▸ h))
  simp only [ha0, hat, hb0, hbt, if_false]
  exact EReal.coe_le_coe_iff.2 (Real.logb_le_logb_of_le one_lt_two
    (ENNReal.toReal_pos ha0 hat) ((ENNReal.toReal_le_toReal hat hbt).2 h))

theorem log2_ereal_natCast (n : ℕ) (hn : n ≠ 0) :
    log2_ereal (n : ENNReal) = ((Real.logb 2 n : ℝ) : EReal) := by
  simp [log2_ereal, hn]

theorem log2_ereal_one : log2_ereal 1 = 0 := by
  simp [log2_ereal]

theorem log2_ereal_lt_top {a : ENNReal} (h : a < ⊤) : log2_ereal a < ⊤ := by
  unfold log2_ereal
  split_ifs with h0
  · exact bot_lt_top
  · exact absurd ‹a = ⊤› h.ne
  · exact EReal.coe_lt_top _

theorem log2_ereal_eq_top_iff {a : ENNReal} : log2_ereal a = ⊤ ↔ a = ⊤ := by
  constructor
  · intro h; by_contra ha; exact (log2_ereal_lt_top (lt_top_iff_ne_top.2 ha)).ne h
  · intro h; simp [log2_ereal, h]

/-- 正有限覆盖数的熵是有限实数。 -/
theorem log2_ereal_real_of_pos_fin {a : ENNReal} (h0 : a ≠ 0) (ht : a ≠ ⊤) :
    log2_ereal a = ((Real.logb 2 a.toReal : ℝ) : EReal) := by
  simp [log2_ereal, h0, ht]

/-- D07 表示对应 -/
theorem covering_entropy_interface (d : I → X → X → ENNReal) (i : I) (ε : ENNReal) (U : Set X) :
    ((∃ S : Finset X, InternalEpsilonNet (d i) ε U S) →
        ∃ S : Finset X, InternalEpsilonNet (d i) ε U S ∧ covering_number (d i) ε U = S.card ∧
          ∀ T : Finset X, InternalEpsilonNet (d i) ε U T → S.card ≤ T.card) ∧
    (covering_number (d i) ε U = ⊤ ↔ ¬ ∃ S : Finset X, InternalEpsilonNet (d i) ε U S) ∧
    covering_number (d i) ε (∅ : Set X) = 0 ∧
    (U.Nonempty → 1 ≤ covering_number (d i) ε U) ∧
    metric_entropy (d i) ε U = log2_ereal (covering_number (d i) ε U) ∧
    (U.Nonempty → 0 ≤ metric_entropy (d i) ε U) ∧
    metric_entropy (d i) ε (∅ : Set X) = ⊥ ∧
    subfamily_metric_entropy d {i} ε U = metric_entropy (d i) ε U ∧
    covering_number (subfamily_pseudometric d {i}) ε U = covering_number (d i) ε U := by
  refine ⟨covering_number_attained (d i) ε U, covering_number_eq_top_iff (d i) ε U,
    covering_number_empty (d i) ε, one_le_covering_number_of_nonempty (d i) ε U, rfl,
    metric_entropy_nonneg (d i) ε U, metric_entropy_empty (d i) ε,
    subfamily_metric_entropy_singleton d i ε U, by rw [subfamily_pseudometric_singleton]⟩

theorem covering_number_le_card {d : X → X → ENNReal} {ε : ENNReal} {U : Set X} {S : Finset X}
    (h : InternalEpsilonNet d ε U S) : covering_number d ε U ≤ S.card :=
  iInf_le_of_le S (iInf_le_of_le h le_rfl)

theorem covering_number_ne_zero {d : X → X → ENNReal} {ε : ENNReal} {U : Set X}
    (hU : U.Nonempty) : covering_number d ε U ≠ 0 :=

    metric_entropy d ε U ≤ metric_entropy d' ε' U' := log2_ereal_mono h

/-! ### CH03-D08 映射熵差 -/

/-- 映射熵差的定义域：非空 `U ⊆ dom ϕ`、非空两端子族、有限正精度，且两端覆盖数至少一个有限
（从而排除 `+∞ - +∞`）。 -/
structure EntropyDiffDomain (d : I → X → X → ENNReal) (Jin Jout : Set I) (ε : ENNReal)
    (ϕ : Operator X bot) (U : Set X) : Prop where
  U_nonempty : U.Nonempty
  U_sub_dom : U ⊆ dom ϕ
  Jin_nonempty : Jin.Nonempty
  Jout_nonempty : Jout.Nonempty
  ε_pos : 0 < ε
  ε_lt_top : ε < ⊤
  finite_side : covering_number (subfamily_pseudometric d Jin) ε U < ⊤ ∨
    covering_number (subfamily_pseudometric d Jout) ε (ϕ.func '' U) < ⊤

/-- 映射熵差 `ΔE_{ε,J_in,J_out}(ϕ|U) = E_{ε,J_in}(U) - E_{ε,J_out}(ϕ(U)) ∈ [-∞,+∞]`，
仅在定义域证明 `h` 给出时定义。 -/
noncomputable def mapEntropyDiff (d : I → X → X → ENNReal) (Jin Jout : Set I) (ε : ENNReal)
    (ϕ : Operator X bot) (U : Set X) (_h : EntropyDiffDomain d Jin Jout ε ϕ U) : EReal :=
  subfamily_metric_entropy d Jin ε U - subfamily_metric_entropy d Jout ε (ϕ.func '' U)

/-- 定义域内两端熵均非 `-∞`（非空），至多一个为 `+∞`；并验证 `[-∞,+∞]` 算术的三种情形。 -/
theorem mapEntropyDiff_cases (d : I → X → X → ENNReal) (Jin Jout : Set I) (ε : ENNReal)
    (ϕ : Operator X bot) (U : Set X) (h : EntropyDiffDomain d Jin Jout ε ϕ U) :
    let Ein := subfamily_metric_entropy d Jin ε U
    let Eout := subfamily_metric_entropy d Jout ε (ϕ.func '' U)
    0 ≤ Ein ∧ 0 ≤ Eout ∧ ¬ (Ein = ⊤ ∧ Eout = ⊤) ∧
    (Ein = ⊤ → Eout < ⊤ → mapEntropyDiff d Jin Jout ε ϕ U h = ⊤) ∧
    (Ein < ⊤ → Eout = ⊤ → mapEntropyDiff d Jin Jout ε ϕ U h = ⊥) ∧
    (∀ a b : ℝ, Ein = a → Eout = b → mapEntropyDiff d Jin Jout ε ϕ U h = ((a - b : ℝ) : EReal)) := by
  intro Ein Eout
  have h0in : 0 ≤ Ein := metric_entropy_nonneg _ ε U h.U_nonempty
  have h0out : 0 ≤ Eout := metric_entropy_nonneg _ ε _ (h.U_nonempty.image _)
  refine ⟨h0in, h0out, ?_, ?_, ?_, ?_⟩
  · rintro ⟨h1, h2⟩
    rcases h.finite_side with hf | hf
    · exact (log2_ereal_lt_top hf).ne h1
    · exact (log2_ereal_lt_top hf).ne h2
  · intro h1 h2
    show Ein - Eout = ⊤
    rw [h1]
    have hb : Eout ≠ ⊥ := ne_bot_of_le_ne_bot (by simp) h0out
    lift Eout to ℝ using ⟨h2.ne, hb⟩ with b
    simp
  · intro h1 h2
    show Ein - Eout = ⊥
    rw [h2]
    have hb : Ein ≠ ⊥ := ne_bot_of_le_ne_bot (by simp) h0in
    lift Ein to ℝ using ⟨h1.ne, hb⟩ with a
    simp
  · intro a b ha hb
    show Ein - Eout = _
    rw [ha, hb, EReal.coe_sub]

/-- 简写 `ΔE_{ε,i,j} := ΔE_{ε,{i},{j}}`，其两端熵即单分量熵。 -/
theorem mapEntropyDiff_single (d : I → X → X → ENNReal) (i j : I) (ε : ENNReal)
    (ϕ : Operator X bot) (U : Set X) (h : EntropyDiffDomain d {i} {j} ε ϕ U) :
    mapEntropyDiff d {i} {j} ε ϕ U h = metric_entropy (d i) ε U - metric_entropy (d j) ε (ϕ.func '' U) := by
  simp [mapEntropyDiff, subfamily_metric_entropy_singleton]

/-- 单分量情形的定义域构造。 -/
theorem entropyDiffDomain_single (d : I → X → X → ENNReal) (i j : I) (ε : ENNReal)
    (ϕ : Operator X bot) (U : Set X) (hU : U.Nonempty) (hUd : U ⊆ dom ϕ) (hε : 0 < ε)
    (hεt : ε < ⊤)
    (hfin : covering_number (d i) ε U < ⊤ ∨ covering_number (d j) ε (ϕ.func '' U) < ⊤) :
    EntropyDiffDomain d {i} {j} ε ϕ U :=
  ⟨hU, hUd, singleton_nonempty i, singleton_nonempty j, hε, hεt, by
    simpa [subfamily_pseudometric_singleton] using hfin⟩

/-! ### 有限复合链 -/

/-- `c_n = ϕ_{n-1} * ⋯ * ϕ_0`（0 起标号：`ϕ m` 是第 `m+1` 步），`c_0 = id`。 -/
def chainComp (ϕ : ℕ → Operator X bot) : ℕ → Operator X bot
  | 0 => 1
  | n + 1 => ϕ n * chainComp ϕ n

/-- 第 `m` 步（0 起标号）的实际输入域：`stepDom 0 = U`，`stepDom (m+1) = ϕ m (stepDom m)`。 -/
def stepDom (ϕ : ℕ → Operator X bot) (U : Set X) : ℕ → Set X
  | 0 => U
  | m + 1 => (ϕ m).func '' stepDom ϕ U m

theorem stepDom_eq_image (ϕ : ℕ → Operator X bot) (U : Set X) (n : ℕ) :
    stepDom ϕ U n = (chainComp ϕ n).func '' U := by
  induction n with
  | zero => simp [stepDom, chainComp]
  | succ n ih =>
    simp only [stepDom, chainComp, ih, image_image]; rfl

/-- 表示对应：`chainComp ϕ n` 是按函数复合顺序取的幺半群乘积 `ϕ (n-1) * ⋯ * ϕ 0`，
即第 1 章约定的有限复合 `c_n = φ_n ∘ ⋯ ∘ φ_1`（`n = 0` 时为 `id`）。 -/
theorem chainComp_eq_prod (ϕ : ℕ → Operator X bot) (n : ℕ) :
    chainComp ϕ n = ((List.range n).map ϕ).reverse.prod := by
  induction n with
  | zero => simp [chainComp]
  | succ n ih => simp [chainComp, List.range_succ, ih]

/-- 与冻结的有限复合 `OperatorChain`（列表首项最后作用）一致。 -/
theorem chainComp_eq_OperatorChain (ϕ : ℕ → Operator X bot) (n : ℕ) :
    chainComp ϕ n = OperatorChain ((List.range n).map ϕ).reverse := by
  rw [chainComp_eq_prod, operator_chain_eq_prod]

/-- 有限族 `φ : Fin n → Operator`（`φ ⟨m, _⟩` 为第 `m+1` 步）延拓为序列，越界项取 `id`。 -/
def finSeq {n : ℕ} (φ : Fin n → Operator X bot) : ℕ → Operator X bot :=
  fun m => if h : m < n then φ ⟨m, h⟩ else 1

theorem finSeq_of_lt {n : ℕ} (φ : Fin n → Operator X bot) {m : ℕ} (h : m < n) :
    finSeq φ m = φ ⟨m, h⟩ := dif_pos h

/-- 有限族的复合：`chainComp (finSeq φ) n = φ_n ∘ ⋯ ∘ φ_1`（`List.ofFn φ` 逆序之积）。 -/
theorem chainComp_finSeq {n : ℕ} (φ : Fin n → Operator X bot) :
    chainComp (finSeq φ) n = (List.ofFn φ).reverse.prod := by
  rw [chainComp_eq_prod]
  congr 2
  apply List.ext_getElem (by simp)
  intro k h1 h2
  simp [finSeq_of_lt φ (by simpa using h1)]

/-- 像值变换 `T : ϕA(U) → X` 满足 `T ∘ ϕA = ϕB`（在 `U` 上）。 -/
def IsImageFactor (ϕA ϕB : Operator X bot) (U : Set X) (T : ↥(ϕA.func '' U) → X) : Prop :=
  ∀ x (hx : x ∈ U), T ⟨ϕA.func x, mem_image_of_mem _ hx⟩ = ϕB.func x

/-- 命题 3.6(i) 必要性。 -/
theorem prop36_nec (ϕA ϕB : Operator X bot) (U : Set X) (T : ↥(ϕA.func '' U) → X)
    (hT : IsImageFactor ϕA ϕB U T) : Req ϕA U ⊆ Req ϕB U := by
  rintro ⟨x, y⟩ ⟨hx, hy, hxy⟩
  refine ⟨hx, hy, ?_⟩
  simp only at hxy ⊢
  rw [← hT x hx, ← hT y hy]
  congr 1
  exact Subtype.ext hxy

/-- 显式构造：对 `y ∈ ϕA(U)` 选代表元 `x ∈ U`，令 `T(y) = ϕB(x)`。 -/
noncomputable def factorMap (ϕA ϕB : Operator X bot) (U : Set X) : ↥(ϕA.func '' U) → X :=
  fun y => ϕB.func (Classical.choose y.2)

/-- 命题 3.6(i) 充分性 -/
theorem prop36_suff (ϕA ϕB : Operator X bot) (U : Set X) (h : Req ϕA U ⊆ Req ϕB U) :
    IsImageFactor ϕA ϕB U (factorMap ϕA ϕB U) := by
  intro x hx
  have hc := Classical.choose_spec (mem_image_of_mem ϕA.func hx)
  exact (h (show (Classical.choose (mem_image_of_mem ϕA.func hx), x) ∈ Req ϕA U from ⟨hc.1, hx, hc.2⟩)).2.2

/-- 命题 3.6(i) 充分性的空集边界 -/
theorem prop36_suff_empty (ϕA ϕB : Operator X bot) :
    Req ϕA ∅ ⊆ Req ϕB ∅ ∧ IsImageFactor ϕA ϕB ∅ (factorMap ϕA ϕB ∅) ∧ IsEmpty ↥(ϕA.func '' ∅) := by
  refine ⟨fun p hp => absurd hp.1 (notMem_empty _), fun x hx => absurd hx (notMem_empty _), ?_⟩
  rw [image_empty]; exact Set.isEmpty_coe_sort.2 rfl

theorem prop36_iii (δ : X → X → ENNReal) (ϕA ϕB : Operator X bot) (U : Set X)

    _ = LB / KA * δ (ϕA.func x) (ϕA.func x') := by
      simp only [div_eq_mul_inv]; ring

/-- 商映射 `T̄ : ϕA(U)/∼ → ϕB(U)/∼` 满足规定赋值公式 `T̄[ϕA x] = [ϕB x]`。 -/
def IsQuotFactor (ϕA ϕB : Operator X bot) (U : Set X)
    (Tb : ImageQuot d hd ϕA U → ImageQuot d hd ϕB U) : Prop :=
  ∀ x (hx : x ∈ U), Tb (imgCls d hd ϕA U x hx) = imgCls d hd ϕB U x hx

/-- 代表元无关的商映射构造（需 `R_∼(ϕA|U) ⊆ R_∼(ϕB|U)`）。 -/
noncomputable def quotFactorMap (ϕA ϕB : Operator X bot) (U : Set X)
    (h : Rsim d ϕA U ⊆ Rsim d ϕB U) : ImageQuot d hd ϕA U → ImageQuot d hd ϕB U :=
  Quotient.lift (fun y : ↥(ϕA.func '' U) =>
      imgCls d hd ϕB U (Classical.choose y.2) (Classical.choose_spec y.2).1)
    (by
      rintro ⟨y, hy⟩ ⟨y', hy'⟩ hyy'
      apply (imgCls_eq_iff d hd ϕB U _ _).2
      have h1 := Classical.choose_spec hy
      have h2 := Classical.choose_spec hy'
      have hmem : (Classical.choose hy, Classical.choose hy') ∈ Rsim d ϕA U := by
        refine ⟨h1.1, h2.1, ?_⟩
        simp only
        rw [h1.2, h2.2]
        exact hyy'
      exact (h hmem).2.2)

theorem quotFactorMap_spec (ϕA ϕB : Operator X bot) (U : Set X)
    (h : Rsim d ϕA U ⊆ Rsim d ϕB U) :
    IsQuotFactor d hd ϕA ϕB U (quotFactorMap d hd ϕA ϕB U h) := by
  intro x hx
  have hx' : ϕA.func x ∈ ϕA.func '' U := mem_image_of_mem _ hx
  have hc := Classical.choose_spec hx'
  show imgCls d hd ϕB U (Classical.choose hx') hc.1 = imgCls d hd ϕB U x hx
  apply (imgCls_eq_iff d hd ϕB U _ _).2
  have : (Classical.choose hx', x) ∈ Rsim d ϕA U :=
    Req_subset_Rsim d hd ϕA U ⟨hc.1, hx, hc.2⟩
  exact (h this).2.2

theorem prop37_one (ϕA ϕB : Operator X bot) (U : Set X) (hU : U ⊆ dom ϕA ∩ dom ϕB) :

  constructor
  · rintro ⟨Tb, hTb⟩ ⟨x, y⟩ ⟨hx, hy, hxy⟩

    have e2 := congrArg Tb e1
    rw [hTb x hx, hTb y hy] at e2
    exact (imgCls_eq_iff d hd ϕB U hx hy).1 e2
  · intro h; exact ⟨_, quotFactorMap_spec d hd ϕA ϕB U h⟩

theorem fiberGE_trans (U : Set X) {ϕ ψ χ : Operator X bot} (h1 : fiberGE d U ϕ ψ)
    (h2 : fiberGE d U ψ χ) : fiberGE d U ϕ χ := subset_trans h1 h2

/-- 固定公共域时 `⪰_{fiber,U}` 是预序（自反、传递）。 -/
instance fiberGE_isPreorder (U : Set X) : IsPreorder (Operator X bot) (fiberGE d U) where
  refl := fiberGE_refl d U
  trans _ _ _ := fiberGE_trans d U

/-- 有限链 `ϕ_0, …, ϕ_m` 的逐步包含推出任意两层 `a < b ≤ m` 的包含。 -/
theorem fiber_chain (U : Set X) (ϕ : ℕ → Operator X bot) (m : ℕ)
    (h : ∀ r, 1 ≤ r → r ≤ m → fiberGE d U (ϕ (r - 1)) (ϕ r)) :
    ∀ a b, a < b → b ≤ m → fiberGE d U (ϕ a) (ϕ b) := by
  have key : ∀ a b, a ≤ b → b ≤ m → fiberGE d U (ϕ a) (ϕ b) := by
    intro a b hab
    induction b, hab using Nat.le_induction with
    | base => intro _; exact fiberGE_refl d U _
    | succ b hab ih =>
      intro hb
      exact fiberGE_trans d U (ih (by omega)) (by simpa using h (b + 1) (by omega) hb)
  exact fun a b hab hbm => key a b hab.le hbm

/-- 链中某一步真包含推出端点真包含，并给出差集见证 `R_r \ R_{r-1} ⊆ R_m \ R_0`。 -/
theorem fiber_chain_strict (U : Set X) (ϕ : ℕ → Operator X bot) (m : ℕ)
    (h : ∀ r, 1 ≤ r → r ≤ m → fiberGE d U (ϕ (r - 1)) (ϕ r)) (r : ℕ) (hr1 : 1 ≤ r)
    (hrm : r ≤ m) (hs : Rsim d (ϕ (r - 1)) U ⊂ Rsim d (ϕ r) U) :
    (Rsim d (ϕ r) U \ Rsim d (ϕ (r - 1)) U ⊆ Rsim d (ϕ m) U \ Rsim d (ϕ 0) U) ∧
    (Rsim d (ϕ r) U \ Rsim d (ϕ (r - 1)) U).Nonempty ∧
    Rsim d (ϕ 0) U ⊂ Rsim d (ϕ m) U := by
  have hle : ∀ a b, a ≤ b → b ≤ m → fiberGE d U (ϕ a) (ϕ b) := by
    intro a b hab hbm
    rcases hab.lt_or_eq with hlt | heq
    · exact fiber_chain d U ϕ m h a b hlt hbm
    · subst heq; exact fiberGE_refl d U _
  have h0 : Rsim d (ϕ 0) U ⊆ Rsim d (ϕ (r - 1)) U := hle 0 (r - 1) (Nat.zero_le _) (by omega)
  have hm : Rsim d (ϕ r) U ⊆ Rsim d (ϕ m) U := hle r m hrm le_rfl
  have hdiff : Rsim d (ϕ r) U \ Rsim d (ϕ (r - 1)) U ⊆ Rsim d (ϕ m) U \ Rsim d (ϕ 0) U :=
    fun p hp => ⟨hm hp.1, fun h0p => hp.2 (h0 h0p)⟩
  have hne : (Rsim d (ϕ r) U \ Rsim d (ϕ (r - 1)) U).Nonempty := by
    obtain ⟨p, hp1, hp2⟩ := Set.exists_of_ssubset hs
    exact ⟨p, hp1, hp2⟩
  refine ⟨hdiff, hne, ?_⟩
  obtain ⟨p, hp⟩ := hne
  exact ssubset_of_subset_of_ne (hle 0 m (Nat.zero_le _) (by omega))
    (fun heq => (hdiff hp).2 (heq ▸ (hdiff hp).1))

/-- `ϕA ⪰_{fiber,U} ϕB` 当且仅当 `ϕA` 的每个全族纤维包含于 `ϕB` 在同一输入处的全族纤维，
即 `ϕA` 的零测距纤维分划细于或等于 `ϕB` 的分划。 -/
theorem fiberGE_iff_Fib_subset (ϕA ϕB : Operator X bot) (U : Set X) :
    fiberGE d U ϕA ϕB ↔ ∀ x ∈ U, Fib d ϕA univ (ϕA.func x) U ⊆ Fib d ϕB univ (ϕB.func x) U := by
  constructor
  · intro h x hx y hy
    have hyU : y ∈ U := hy.1
    exact (mem_Rsim_iff_mem_Fib d hd ϕB U hx hyU).1 (h ((mem_Rsim_iff_mem_Fib d hd ϕA U hx hyU).2 hy))
  · rintro h ⟨x, y⟩ hp
    have hx : x ∈ U := hp.1
    have hy : y ∈ U := hp.2.1
    exact (mem_Rsim_iff_mem_Fib d hd ϕB U hx hy).2 (h x hx ((mem_Rsim_iff_mem_Fib d hd ϕA U hx hy).1 hp))

/-- 有限链 `φ : Fin (m+1) → Operator`（即 `φ_0, …, φ_m`）形式：逐步包含推出任意两层 `a ≤ b` 的包含。 -/
theorem fiber_chain_fin (U : Set X) {m : ℕ} (φ : Fin (m + 1) → Operator X bot)
    (h : ∀ r : Fin m, fiberGE d U (φ r.castSucc) (φ r.succ)) :
    ∀ a b : Fin (m + 1), a ≤ b → fiberGE d U (φ a) (φ b) :=
  fun a b hab => (Fin.monotone_iff_le_succ (f := fun k => Rsim d (φ k) U)).2 h hab

/-- 有限链形式：某一步 `R_r ⊊ R_{r+1}` 推出差集见证 `R_{r+1} \ R_r ⊆ R_m \ R_0` 与 `R_0 ⊊ R_m`。 -/
theorem fiber_chain_fin_strict (U : Set X) {m : ℕ} (φ : Fin (m + 1) → Operator X bot)
    (h : ∀ r : Fin m, fiberGE d U (φ r.castSucc) (φ r.succ)) (r : Fin m)
    (hs : Rsim d (φ r.castSucc) U ⊂ Rsim d (φ r.succ) U) :
    (Rsim d (φ r.succ) U \ Rsim d (φ r.castSucc) U ⊆ Rsim d (φ (Fin.last m)) U \ Rsim d (φ 0) U) ∧
    (Rsim d (φ r.succ) U \ Rsim d (φ r.castSucc) U).Nonempty ∧
    Rsim d (φ 0) U ⊂ Rsim d (φ (Fin.last m)) U := by
  have h0 : Rsim d (φ 0) U ⊆ Rsim d (φ r.castSucc) U := fiber_chain_fin d U φ h _ _ (Fin.zero_le _)
  have hm : Rsim d (φ r.succ) U ⊆ Rsim d (φ (Fin.last m)) U := fiber_chain_fin d U φ h _ _ (Fin.le_last _)
  refine ⟨fun p hp => ⟨hm hp.1, fun h0p => hp.2 (h0 h0p)⟩, ?_, ?_⟩
  · obtain ⟨p, hp1, hp2⟩ := Set.exists_of_ssubset hs
    exact ⟨p, hp1, hp2⟩
  · exact (h0.trans_ssubset hs).trans_subset hm

/-- 目标子族合法当且仅当对每个分量合法（由分量交集律）。 -/
theorem admRadii_subfamily (ϕ : Operator X bot) (U : Set X) (i : I) (J : Set I)
    (hJ : J.Nonempty) (y : X) (r : ENNReal) :
    r ∈ admRadii (d i) (subfamily_pseudometric d J) ϕ y U ↔
      0 < r ∧ r < ⊤ ∧ ∀ j ∈ J, r ∈ admRadii (d i) (d j) ϕ y U := by
  simp only [admRadii, mem_setOf_eq]
  constructor
  · rintro ⟨h0, ht, h⟩
    refine ⟨h0, ht, fun j hj => ⟨h0, ht, fun x hx => ?_⟩⟩
    have := h hx
    exact ⟨this.1, (subfamily_pseudometric_eq_zero_iff d J _ _).1 this.2 j hj⟩
  · rintro ⟨h0, ht, h⟩
    refine ⟨h0, ht, fun x hx => ⟨hx.2, (subfamily_pseudometric_eq_zero_iff d J _ _).2
      fun j hj => ((h j hj).2.2 hx).2⟩⟩

theorem prop310_pointwise (ϕ : Operator X bot) (U : Set X) (i : I) (J : Set I)
    (hJ : J.Nonempty) (y : X) :
    rNJ d ϕ i J y U = ⨅ j ∈ J, rN d ϕ i j y U := by
  apply le_antisymm
  · refine le_iInf₂ fun j hj => sSup_le_sSup fun r hr => ?_
    exact ((admRadii_subfamily d ϕ U i J hJ y r).1 hr).2.2 j hj
  · apply le_of_forall_lt
    intro c hc
    obtain ⟨c', hcc', hc'⟩ := exists_between hc
    have hpos : 0 < c' := lt_of_le_of_lt (zero_le _) hcc'
    have hfin : c' < ⊤ := lt_of_lt_of_le hc' le_top
    have hadm : c' ∈ admRadii (d i) (subfamily_pseudometric d J) ϕ y U := by
      refine (admRadii_subfamily d ϕ U i J hJ y c').2 ⟨hpos, hfin, fun j hj => ?_⟩
      obtain ⟨ρ, hρ, hlt⟩ := exists_adm_of_lt_nullRadius (lt_of_lt_of_le hc' (iInf₂_le j hj))

  obtain ⟨j, hj, hmin⟩ := Set.exists_min_image J (fun j => rN d ϕ i j y U) (Set.toFinite J) hJ

  intro r _ _ hrM x' hx'
  have h1 : d i x' x * L < nullMatrix d ϕ₂ (ϕ₁.func '' U) j l := by

  have h2 : d l (ϕ₁.func x') (ϕ₁.func x) < nullRadius (d l) (d j) ϕ₂ (ϕ₁.func x) (ϕ₁.func '' U) :=
    calc d l (ϕ₁.func x') (ϕ₁.func x) ≤ L * d i x' x := hLip x' hx'.2 x hx

      _ < _ := h1

  have h3 := zero_of_lt_nullRadius (mem_image_of_mem ϕ₁.func hx'.2) h2

    thresholdType σ τσ τN d f A i j ∈ ({(1, 1), (1, 0), (0, 1), (0, 0)} : Set (Fin 2 × Fin 2)) := by
  unfold thresholdType
  by_cases h1 : σ.σ i A < τσ <;> by_cases h2 : τN ≤ uNR (d i) (d j) f A <;>
    simp [h1, h2, not_lt.1, not_le.1]

theorem lip_push_net (ϕ : Operator X bot) (U : Set X) (i j : I) (ε L : ENNReal)
    (hL0 : 0 < L) (hLt : L < ⊤)
    (hLip : ∀ x ∈ U, ∀ y ∈ U, d j (ϕ.func x) (ϕ.func y) ≤ L * d i x y)
    (S : Finset X) (hS : InternalEpsilonNet (d i) (ε / L) U S) :
    InternalEpsilonNet (d j) ε (ϕ.func '' U) (S.image ϕ.func) ∧ (S.image ϕ.func).card ≤ S.card := by

  refine ⟨⟨?_, ?_⟩, Finset.card_image_le⟩

      _ < ε := hds

theorem colip_pull_net [Fintype I] (ϕ : Operator X bot) (U : Set X) (i j : I) (ε : ENNReal)
    (K : I → I → ENNReal) (hK : IsCoLipMatrix d ϕ U K) (hK0 : 0 < K i j) (hKt : K i j < ⊤)
    (T : Finset X) (hT : InternalEpsilonNet (d j) ε (ϕ.func '' U) T) :
    InternalEpsilonNet (d i) (ε / K i j) U (preCenters ϕ U T) ∧
      (preCenters ϕ U T).card ≤ T.card := by

  refine ⟨⟨hsub, fun x hx => ?_⟩, hcard⟩

  calc d i x c * K i j = K i j * d i x c := mul_comm _ _
    _ ≤ ∑ k : I, K k j * d k x c :=

    _ ≤ d j (ϕ.func x) (ϕ.func c) := hK.2.2 x hx c (hsub hc) j
    _ < ε := hdt

section ClsDist
variable {Y α : Type} [DecidableEq Y] [DecidableEq α]

/-- 类距离：`b` 与自身距 0，与其余点距 `+∞`；非吸收点按类函数 `c` 同类距 0，异类距 `v`。 -/
noncomputable def clsDist (b : Y) (c : Y → α) (v : ENNReal) : Y → Y → ENNReal :=
  fun x y => if x = b ∧ y = b then 0 else if x = b ∨ y = b then ⊤ else if c x = c y then 0 else v

/-- 类距离满足扩展伪度量四公理（含吸收元隔离）。 -/
theorem clsDist_epm (b : Y) (c : Y → α) (v : ENNReal) (hne : ∃ y : Y, y ≠ b) :
    ExtendedPseudoMetric Y b (clsDist b c v) where
  nonabsorbing := hne
  dist_self x := by unfold clsDist; by_cases hx : x = b <;> simp [hx]
  symm x y := by
    unfold clsDist
    by_cases hx : x = b <;> by_cases hy : y = b <;> simp [hx, hy, eq_comm]
  triangle x y z := by
    unfold clsDist
    by_cases hx : x = b <;> by_cases hy : y = b <;> by_cases hz : z = b <;> simp [hx, hy, hz]
    split_ifs <;> simp_all
  isolate_bot x hx := by unfold clsDist; simp [hx]

end ClsDist

inductive P3 | bot | a | b
  deriving DecidableEq

noncomputable def dSwap : Unit → P3 → P3 → ENNReal := fun _ => clsDist P3.bot id 1

theorem dSwap_epm : ∀ i, ExtendedPseudoMetric P3 P3.bot (dSwap i) :=
  fun _ => clsDist_epm _ _ _ ⟨P3.a, by decide⟩

/-- 两点换位算子（保持吸收元）。 -/
def swap3 : Operator P3 P3.bot where
  func := fun x => match x with
    | .bot => .bot
    | .a => .b
    | .b => .a
  preserves_bot := rfl

def U3 : Set P3 := {P3.a, P3.b}

theorem swap3_mul : swap3 * swap3 = 1 := by
  apply Operator.ext; funext x; cases x <;> rfl

theorem swap3_image : swap3.func '' U3 = U3 := by
  ext x; cases x <;> simp [U3, swap3]

theorem U3_sub_dom_swap1 : U3 ⊆ dom swap3 := by
  intro x hx; simp only [U3, mem_insert_iff, mem_singleton_iff] at hx
  rcases hx with rfl | rfl <;> simp [dom, swap3]

theorem U3_sub_dom_swap : U3 ⊆ dom (swap3 * swap3) := by
  rw [swap3_mul]; intro x hx; simp only [U3, mem_insert_iff, mem_singleton_iff] at hx
  rcases hx with rfl | rfl <;> simp [dom]

theorem swap_not_const : () ∉ Iconst dSwap swap3 U3 :=
  (not_mem_Iconst_iff dSwap).2 ⟨.a, by simp [U3], .b, by simp [U3], by simp [dSwap, clsDist, swap3]⟩

theorem swap_act : Iact dSwap swap3 U3 = {()} ∧ Iid dSwap swap3 U3 = ∅ := by

  · ext i; cases i; simp only [mem_singleton_iff, iff_true]

  · ext i; cases i; simp only [mem_empty_iff_false, iff_false]
    exact fun h => hnid h.2

  · rw [swap3_image, h2, h3]; simp

/-- 两点无公共内部中心时覆盖数至少为 2。 -/
theorem two_le_covering_number {δ : X → X → ENNReal} {ε : ENNReal} {U : Set X} {x y : X}
    (hx : x ∈ U) (hy : y ∈ U) (h : ∀ c ∈ U, δ x c < ε → δ y c < ε → False) :
    2 ≤ covering_number δ ε U := by
  apply le_iInf₂
  intro S hS
  obtain ⟨cx, hcx, hdx⟩ := hS.2 x hx
  obtain ⟨cy, hcy, hdy⟩ := hS.2 y hy
  have hne : cx ≠ cy := by rintro rfl; exact h cx (hS.1 hcx) hdx hdy
  have : 2 ≤ S.card := Finset.one_lt_card.2 ⟨cx, hcx, cy, hcy, hne⟩
  exact_mod_cast this

/-- 若 `T` 中每点只能被自身作为内部中心覆盖，则覆盖数至少为 `|T|`。 -/
theorem card_le_covering_number_of_isolated {δ : X → X → ENNReal} {ε : ENNReal} {U : Set X}
    (T : Finset X) (hT : (↑T : Set X) ⊆ U) (h : ∀ x ∈ T, ∀ c ∈ U, δ x c < ε → c = x) :
    (T.card : ENNReal) ≤ covering_number δ ε U := by
  apply le_iInf₂
  intro S hS
  have hsub : T ⊆ S := by
    intro x hx
    obtain ⟨c, hc, hd⟩ := hS.2 x (hT hx)
    rw [← h x hx c (hS.1 hc) hd]; exact hc
  exact_mod_cast Finset.card_le_card hsub

theorem covering_number_le_two {δ : X → X → ENNReal} {ε : ENNReal} {U : Set X} [DecidableEq X]
    (a b : X) (h : InternalEpsilonNet δ ε U {a, b}) : covering_number δ ε U ≤ 2 :=
  le_trans (covering_number_le_card h) (by exact_mod_cast Finset.card_le_two)

theorem covering_number_le_one {δ : X → X → ENNReal} {ε : ENNReal} {U : Set X}
    (a : X) (h : InternalEpsilonNet δ ε U {a}) : covering_number δ ε U ≤ 1 :=
  le_trans (covering_number_le_card h) (by simp)

/-! ### 带吸收元的实直线 `ℝ ⊔ {⊥}` -/

/-- `ℝ ⊔ {⊥}`：`none` 为吸收元，`some x` 为实数点。 -/
abbrev RB := Option ℝ

/-- 把实数部分的距离按吸收元约定延拓：`d(⊥,⊥) = 0`，`⊥` 与实数点距离 `+∞`。 -/
noncomputable def liftDist (ρ : ℝ → ℝ → ENNReal) : RB → RB → ENNReal
  | none, none => 0
  | some _, none => ⊤
  | none, some _ => ⊤
  | some x, some y => ρ x y

@[simp] theorem liftDist_some (ρ : ℝ → ℝ → ENNReal) (x y : ℝ) :
    liftDist ρ (some x) (some y) = ρ x y := rfl

theorem liftDist_epm (ρ : ℝ → ℝ → ENNReal) (h0 : ∀ x, ρ x x = 0) (hs : ∀ x y, ρ x y = ρ y x)
    (ht : ∀ x y z, ρ x z ≤ ρ x y + ρ y z) : ExtendedPseudoMetric RB none (liftDist ρ) where
  nonabsorbing := ⟨some 0, by simp⟩
  dist_self x := by cases x <;> simp [liftDist, h0]
  symm x y := by cases x <;> cases y <;> simp [liftDist, hs]
  triangle x y z := by cases x <;> cases y <;> cases z <;> simp [liftDist, ht]
  isolate_bot x hx := by cases x <;> simp_all [liftDist]

/-- 通常距离 `|x - y|`。 -/
noncomputable def dR : RB → RB → ENNReal := liftDist (fun x y => ENNReal.ofReal |x - y|)

/-- 绝对值读数距离 `||x| - |y||`。 -/
noncomputable def dAbs : RB → RB → ENNReal := liftDist (fun x y => ENNReal.ofReal |(|x| - |y|)|)

@[simp] theorem dR_some (x y : ℝ) : dR (some x) (some y) = ENNReal.ofReal |x - y| := rfl
@[simp] theorem dAbs_some (x y : ℝ) : dAbs (some x) (some y) = ENNReal.ofReal |(|x| - |y|)| := rfl

theorem dR_epm : ExtendedPseudoMetric RB none dR :=
  liftDist_epm _ (fun x => by simp) (fun x y => by rw [abs_sub_comm])
    (fun x y z => (ENNReal.ofReal_le_ofReal (abs_sub_le x y z)).trans ENNReal.ofReal_add_le)

theorem dAbs_epm : ExtendedPseudoMetric RB none dAbs :=
  liftDist_epm _ (fun x => by simp) (fun x y => by rw [abs_sub_comm])
    (fun x y z => (ENNReal.ofReal_le_ofReal (abs_sub_le |x| |y| |z|)).trans ENNReal.ofReal_add_le)

/-- 单分量规范族（指标集 `Unit`）。 -/
noncomputable def gR : Unit → RB → RB → ENNReal := fun _ => dR
noncomputable def gAbs : Unit → RB → RB → ENNReal := fun _ => dAbs

theorem gR_epm : ∀ i, ExtendedPseudoMetric RB none (gR i) := fun _ => dR_epm
theorem gAbs_epm : ∀ i, ExtendedPseudoMetric RB none (gAbs i) := fun _ => dAbs_epm

/-- 实值全映射提升为保持吸收元的算子。 -/
def realOp (h : ℝ → ℝ) : Operator RB none := ⟨fun z => z.map h, rfl⟩

/-- 偏映射（`none` 表示未定义）提升为保持吸收元的算子。 -/
def partialOp (g : ℝ → Option ℝ) : Operator RB none := ⟨fun z => z.bind g, rfl⟩

@[simp] theorem realOp_some (h : ℝ → ℝ) (x : ℝ) : (realOp h).func (some x) = some (h x) := rfl
@[simp] theorem partialOp_some (g : ℝ → Option ℝ) (x : ℝ) :
    (partialOp g).func (some x) = g x := rfl

theorem some_mem_dom_realOp (h : ℝ → ℝ) (x : ℝ) : some x ∈ dom (realOp h) := by simp [dom]

theorem image_some_sub_dom_realOp (h : ℝ → ℝ) (S : Set ℝ) : some '' S ⊆ dom (realOp h) := by
  rintro _ ⟨x, -, rfl⟩; exact some_mem_dom_realOp h x

theorem ofReal_eq_toReal {ε : ENNReal} (h0 : 0 < ε) (ht : ε < ⊤) :
    ∃ e : ℝ, 0 < e ∧ ε = ENNReal.ofReal e :=
  ⟨ε.toReal, ENNReal.toReal_pos h0.ne' ht.ne, (ENNReal.ofReal_toReal ht.ne).symm⟩

/-! ### CH03-P10 反例 3.1：阶跃算子 -/

/-- 阶跃函数 `f(x) = 0 (x < 0)`，`1 (x ≥ 0)`。 -/
noncomputable def stepF (x : ℝ) : ℝ := if x < 0 then 0 else 1

noncomputable def stepOp : Operator RB none := realOp stepF

def stepSet : Set ℝ := Ico (-1) 0 ∪ Ioc 0 1

/-- `U = [-1,0) ∪ (0,1] ⊆ ℝ ⊔ {⊥}`。 -/
def Ustep : Set RB := some '' stepSet

/-- 反例 3.1 的结构合法性 -/
theorem ex_step_structure :
    (∀ i, ExtendedPseudoMetric RB none (gR i)) ∧ stepOp.func none = none ∧
    Ustep ⊆ dom stepOp ∧ Ustep.Nonempty ∧ none ∉ Ustep :=
  ⟨gR_epm, rfl, image_some_sub_dom_realOp _ _,
    ⟨some 1, mem_image_of_mem _ (Or.inr ⟨one_pos, le_rfl⟩)⟩, by simp [Ustep]⟩

theorem stepF_eq_of_close {x y : ℝ} (hy : y ≠ 0) (h : |x - y| < |y|) : stepF x = stepF y := by
  rcases lt_or_gt_of_ne hy with hy | hy
  · rw [abs_of_neg hy] at h
    have : x < 0 := by linarith [le_abs_self (x - y)]
    simp [stepF, this, hy]
  · rw [abs_of_pos hy] at h
    have : 0 < x := by linarith [neg_abs_le (x - y)]
    simp [stepF, not_lt.2 this.le, not_lt.2 hy.le]

theorem stepSet_ne_zero {y : ℝ} (hy : y ∈ stepSet) : y ≠ 0 := by
  rcases hy with ⟨-, h⟩ | ⟨h, -⟩ <;> [exact h.ne; exact h.ne']

/-- 半径超过 `|y|` 时，存在 `U` 中异号点落入开球。 -/
theorem step_witness {y : ℝ} (hy : y ∈ stepSet) {r : ℝ} (hr : |y| < r) :
    ∃ x ∈ stepSet, |x - y| < r ∧ stepF x ≠ stepF y := by
  obtain ⟨m, hm0, hm1, hm2⟩ : ∃ m : ℝ, 0 < m ∧ m ≤ 1 ∧ m ≤ r - |y| :=
    ⟨min (r - |y|) 1, lt_min (by linarith) one_pos, min_le_right _ _, min_le_left _ _⟩
  rcases hy with ⟨hy1, hy2⟩ | ⟨hy1, hy2⟩
  · rw [abs_of_neg hy2] at hm2
    refine ⟨m / 2, Or.inr ⟨by linarith, by linarith⟩, ?_, ?_⟩
    · rw [abs_of_pos (by linarith)]; linarith
    · simp [stepF, hy2, not_lt.2 (by linarith : (0:ℝ) ≤ m / 2)]
  · rw [abs_of_pos hy1] at hm2
    refine ⟨-(m / 2), Or.inl ⟨by linarith, by linarith⟩, ?_, ?_⟩
    · rw [abs_of_neg (by linarith)]; linarith
    · simp [stepF, not_lt.2 hy1.le, (by linarith : -(m / 2) < 0)]

/-- 反例 3.1 -/
theorem ex_step_adm (y : ℝ) (hy : y ∈ stepSet) :
    admRadii dR dR stepOp (some y) Ustep = Ioc 0 (ENNReal.ofReal |y|) := by
  ext r
  constructor
  · rintro ⟨h0, ht, hsub⟩
    refine ⟨h0, ?_⟩
    by_contra hlt
    rw [not_le] at hlt
    obtain ⟨e, he0, rfl⟩ := ofReal_eq_toReal h0 ht
    have hye : |y| < e := ((ENNReal.ofReal_lt_ofReal_iff he0).1 hlt)
    obtain ⟨x, hx, hxy, hne⟩ := step_witness hy hye
    have hmem : some x ∈ ball dR (some y) (ENNReal.ofReal e) ∩ Ustep :=
      ⟨by simpa [ball] using (ENNReal.ofReal_lt_ofReal_iff he0).2 hxy, mem_image_of_mem _ hx⟩
    have h0' := (hsub hmem).2
    simp only [stepOp, realOp_some, dR_some, ENNReal.ofReal_eq_zero] at h0'
    exact hne (by linarith [abs_nonneg (stepF x - stepF y), abs_le.1 (le_of_le_of_eq h0' rfl)])
  · rintro ⟨h0, hle⟩
    refine ⟨h0, lt_of_le_of_lt hle ENNReal.ofReal_lt_top, ?_⟩
    rintro _ ⟨hb, ⟨x, hx, rfl⟩⟩
    refine ⟨mem_image_of_mem _ hx, ?_⟩
    have hb' : ENNReal.ofReal |x - y| < ENNReal.ofReal |y| := lt_of_lt_of_le hb hle
    have := stepF_eq_of_close (stepSet_ne_zero hy) (ENNReal.ofReal_lt_ofReal_iff'.1 hb').1
    simp [stepOp, this]

/-- 反例 3.1 -/
theorem ex_step_radius (y : ℝ) (hy : y ∈ stepSet) :
    nullRadius dR dR stepOp (some y) Ustep = ENNReal.ofReal |y| ∧
    0 < nullRadius dR dR stepOp (some y) Ustep := by
  have hpos : 0 < ENNReal.ofReal |y| :=
    ENNReal.ofReal_pos.2 (abs_pos.2 (stepSet_ne_zero hy))
  have : nullRadius dR dR stepOp (some y) Ustep = ENNReal.ofReal |y| := by
    rw [nullRadius, ex_step_adm y hy, csSup_Ioc hpos]
  exact ⟨this, this ▸ hpos⟩

/-- 反例 3.1 -/
theorem ex_step_uniform : uNR dR dR stepOp Ustep = 0 ∧ nullMatrix gR stepOp Ustep () () = 0 := by
  have h : uNR dR dR stepOp Ustep = 0 := by
    apply le_antisymm _ (zero_le _)
    apply ENNReal.le_of_forall_pos_le_add
    intro ε hε _
    rw [zero_add]
    have hε' : (0 : ℝ) < ε := NNReal.coe_pos.2 hε
    have hy : min 1 (ε : ℝ) ∈ stepSet := Or.inr ⟨lt_min one_pos hε', min_le_left _ _⟩
    calc uNR dR dR stepOp Ustep ≤ nullRadius dR dR stepOp (some (min 1 (ε : ℝ))) Ustep :=
          uniform_le_nullRadius (mem_image_of_mem _ hy)
      _ = ENNReal.ofReal |min 1 (ε : ℝ)| := (ex_step_radius _ hy).1
      _ ≤ ENNReal.ofReal ε := ENNReal.ofReal_le_ofReal
          (by rw [abs_of_pos (lt_min one_pos hε')]; exact min_le_right _ _)
      _ = ε := ENNReal.ofReal_coe_nnreal
  exact ⟨h, h⟩

/-- 命题 3.16 后正文反例（全称形式）：对任意有限正 `τ` 与任意满足 `0 < a ≤ 1`、`2a < τ` 的 `a`，

    · exact mem_image_of_mem _ (Or.inl ⟨by linarith, by linarith⟩)
    · exact mem_image_of_mem _ (Or.inr ⟨ha0, ha1⟩)
  · apply le_antisymm
    · apply covering_number_le_one (some a)
      refine ⟨by simp, fun x hx => ⟨some a, by simp, ?_⟩⟩
      simp only [mem_insert_iff, mem_singleton_iff] at hx
      rcases hx with rfl | rfl
      · simp only [dR_some]
        rw [show -a - a = -(2 * a) by ring, abs_neg, abs_of_pos (by linarith)]
        exact h2a

/-- 命题 3.16 后正文反例：对任意有限正 `τ`，满足 `0 < a ≤ 1`、`2a < τ` 的 `a` 存在（取 `min(1, τ/4)`）。 -/
theorem remark316 (τ : ENNReal) (hτ0 : 0 < τ) (hτt : τ < ⊤) :
    ∃ a : ℝ, 0 < a ∧ a ≤ 1 ∧ ENNReal.ofReal (2 * a) < τ ∧
      ({some (-a), some a} : Set RB) ⊆ Ustep ∧
      covering_number dR τ {some (-a), some a} = 1 ∧
      stepOp.func '' {some (-a), some a} = {some 0, some 1} ∧
      covering_number dR (1 / 2) (stepOp.func '' {some (-a), some a}) = 2 := by
  obtain ⟨e, he0, rfl⟩ := ofReal_eq_toReal hτ0 hτt
  have ha0 : 0 < min 1 (e / 4) := lt_min one_pos (by linarith)
  have h2a : ENNReal.ofReal (2 * min 1 (e / 4)) < ENNReal.ofReal e :=
    (ENNReal.ofReal_lt_ofReal_iff he0).2 (by linarith [min_le_right (1 : ℝ) (e / 4)])
  exact ⟨_, ha0, min_le_left _ _, h2a, remark316_forall _ hτ0 hτt _ ha0 (min_le_left _ _) h2a⟩

/-- 点态正半径不能代替统一阈值 -/

  refine ⟨_, hB, ?_⟩
  rw [h1, h2]; norm_num

  rintro ⟨τ, hτ0, hτt, H⟩

  exact absurd (H B hB (1 / 2) (ENNReal.half_pos one_ne_zero)
    (ENNReal.div_lt_top ENNReal.one_ne_top two_ne_zero)) (not_le.2 hlt)

        _ = 4 := by rw [U4_finset_card]; rfl

  · apply le_antisymm
