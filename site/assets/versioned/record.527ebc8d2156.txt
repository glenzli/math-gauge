  match x, y with
  | none, none => 0
  | none, some _ => ⊤
  | some _, none => ⊤
  | some u, some v => ENNReal.ofReal (| |u| - |v| |)

    rename_i u v w

  · intro x hx; cases x <;> [contradiction; rfl]
