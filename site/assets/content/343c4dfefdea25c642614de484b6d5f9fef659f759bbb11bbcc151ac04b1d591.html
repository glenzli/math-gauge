/-- Zero distance lets one replace the left argument. -/
theorem dist_congr_left {e : X → X → ENNReal} (h : ExtendedPseudoMetric X bot e)
    {a b : X} (hab : e a b = 0) (c : X) : e a c = e b c := by
  apply le_antisymm
  · calc e a c ≤ e a b + e b c := h.triangle a b c
      _ = e b c := by rw [hab, zero_add]
  · calc e b c ≤ e b a + e a c := h.triangle b a c
      _ = e a c := by rw [h.symm b a, hab, zero_add]

/-- Zero distance lets one replace the right argument. -/
theorem dist_congr_right {e : X → X → ENNReal} (h : ExtendedPseudoMetric X bot e)
    {a b : X} (hab : e a b = 0) (c : X) : e c a = e c b := by
  rw [h.symm c a, h.symm c b]
  exact dist_congr_left h hab c

section Partition
variable {X I : Type} {bot : X} (d : I → X → X → ENNReal)

/-- 常值分量 `I_const(ϕ | U)`. -/
def Iconst (ϕ : Operator X bot) (U : Set X) : Set I :=
  {i | ∀ x ∈ U, ∀ y ∈ U, d i (ϕ.func x) (ϕ.func y) = 0}

/-- 恒等分量 `I_id(ϕ | U)`（常值优先）. -/
def Iid (ϕ : Operator X bot) (U : Set X) : Set I :=
  {i | i ∉ Iconst d ϕ U ∧ ∀ x ∈ U, d i (ϕ.func x) x = 0}

/-- 有效分量 `I_act(ϕ | U) = I \ (I_const ∪ I_id)`. -/
def Iact (ϕ : Operator X bot) (U : Set X) : Set I :=
  (Iconst d ϕ U ∪ Iid d ϕ U)ᶜ

/-- 全域分划：`U = dom ϕ`. -/
abbrev IconstDom (ϕ : Operator X bot) : Set I := Iconst d ϕ (dom ϕ)
abbrev IidDom (ϕ : Operator X bot) : Set I := Iid d ϕ (dom ϕ)
abbrev IactDom (ϕ : Operator X bot) : Set I := Iact d ϕ (dom ϕ)

/-- 三类分量两两不交且覆盖 `I`。 -/
theorem partition_spec (ϕ : Operator X bot) (U : Set X) :
    Iconst d ϕ U ∪ Iid d ϕ U ∪ Iact d ϕ U = Set.univ ∧
    Disjoint (Iconst d ϕ U) (Iid d ϕ U) ∧
    Disjoint (Iconst d ϕ U) (Iact d ϕ U) ∧
    Disjoint (Iid d ϕ U) (Iact d ϕ U) := by
  refine ⟨?_, ?_, ?_, ?_⟩
  · ext i; simp only [Iact, Set.mem_union, Set.mem_compl_iff, Set.mem_univ, iff_true]; tauto
  · rw [Set.disjoint_left]; intro i h1 h2; exact h2.1 h1
  · rw [Set.disjoint_left]; intro i h1 h2; exact h2 (Or.inl h1)
  · rw [Set.disjoint_left]; intro i h1 h2; exact h2 (Or.inr h1)

theorem mem_Iid_iff (ϕ : Operator X bot) (U : Set X) (i : I) :
    i ∈ Iid d ϕ U ↔ i ∉ Iconst d ϕ U ∧ ∀ x ∈ U, d i (ϕ.func x) x = 0 := Iff.rfl

theorem not_mem_Iact_iff (ϕ : Operator X bot) (U : Set X) (i : I) :
    i ∉ Iact d ϕ U ↔ i ∈ Iconst d ϕ U ∨ ∀ x ∈ U, d i (ϕ.func x) x = 0 := by
  simp only [Iact, Set.mem_compl_iff, not_not, Set.mem_union, mem_Iid_iff]
  tauto

theorem prop32_i_core (hd : ∀ i, ExtendedPseudoMetric X bot (d i))
    (ϕ1 ϕ2 : Operator X bot) (U : Set X) :
    Iact d (ϕ2 * ϕ1) U ⊆ Iact d ϕ1 U ∪ Iact d ϕ2 (ϕ1.func '' U) := by
  intro i hi
  by_contra hn
  simp only [Set.mem_union, not_or] at hn
  obtain ⟨h1, h2⟩ := hn
  rw [not_mem_Iact_iff] at h1 h2
  have : i ∉ Iact d (ϕ2 * ϕ1) U := by
    rw [not_mem_Iact_iff]
    rcases h2 with h2 | h2
    · left
      intro x hx y hy
      exact h2 _ (Set.mem_image_of_mem _ hx) _ (Set.mem_image_of_mem _ hy)
    · rcases h1 with h1 | h1
      · left
        intro x hx y hy
        simp only [mul_func]
        rw [dist_congr_left (hd i) (h2 _ (Set.mem_image_of_mem _ hx)),
          dist_congr_right (hd i) (h2 _ (Set.mem_image_of_mem _ hy))]
        exact h1 x hx y hy
      · right
        intro x hx
        simp only [mul_func]
        rw [dist_congr_left (hd i) (h2 _ (Set.mem_image_of_mem _ hx))]
        exact h1 x hx
  exact this hi

theorem prop32_i (hd : ∀ i, ExtendedPseudoMetric X bot (d i))

/-- 有限复合 `c_n = ϕ_{n-1} ∘ ⋯ ∘ ϕ_0`（0 起标号；`c_0 = id`）。 -/
def chainOp (ϕ : ℕ → Operator X bot) : ℕ → Operator X bot
  | 0 => 1
  | n + 1 => ϕ n * chainOp ϕ n

/-- 实际中间域：`U_0 = U`，`U_{m+1} = ϕ_m(U_m)`。 -/
def chainDom (ϕ : ℕ → Operator X bot) (U : Set X) : ℕ → Set X
  | 0 => U
  | n + 1 => (ϕ n).func '' chainDom ϕ U n

theorem chainDom_eq_image (ϕ : ℕ → Operator X bot) (U : Set X) (n : ℕ) :
    chainDom ϕ U n = (chainOp ϕ n).func '' U := by
  induction n with
  | zero => simp [chainDom, chainOp]
  | succ n ih =>
    simp only [chainDom, chainOp, ih, Set.image_image]
    rfl

/-- 零步复合（恒等算子）没有有效分量。 -/
theorem Iact_one (hd : ∀ i, ExtendedPseudoMetric X bot (d i)) (U : Set X) :
    Iact d (1 : Operator X bot) U = ∅ := by
  ext i
  simp only [Set.mem_empty_iff_false, iff_false]
  rw [not_mem_Iact_iff]
  right
  intro x _
  exact (hd i).dist_self x

theorem prop32_ii_core (hd : ∀ i, ExtendedPseudoMetric X bot (d i))
    (ϕ : ℕ → Operator X bot) (U : Set X) (n : ℕ) :
    Iact d (chainOp ϕ n) U ⊆ ⋃ m ∈ Finset.range n, Iact d (ϕ m) (chainDom ϕ U m) := by
  induction n with
  | zero => simp [chainOp, Iact_one d hd]
  | succ n ih =>
    intro i hi
    have h := prop32_i_core d hd (chainOp ϕ n) (ϕ n) U hi
    rw [← chainDom_eq_image] at h
    simp only [Set.mem_iUnion, Finset.mem_range, exists_prop]
    rcases h with h | h
    · obtain ⟨m, hm, hm'⟩ := Set.mem_iUnion₂.mp (ih h)
      exact ⟨m, by simp at hm; omega, hm'⟩
    · exact ⟨n, by omega, h⟩

/-- 命题 3.2 (ii)：任意有限长度链（含零步基例）。 -/
theorem prop32_ii (hd : ∀ i, ExtendedPseudoMetric X bot (d i))
    (ϕ : ℕ → Operator X bot) (U : Set X) (n : ℕ) (hU : U ⊆ dom (chainOp ϕ n)) :
    Iact d (chainOp ϕ n) U ⊆ ⋃ m ∈ Finset.range n, Iact d (ϕ m) (chainDom ϕ U m) :=

theorem prop32_iii (hd : ∀ i, ExtendedPseudoMetric X bot (d i))
    (ϕ1 ϕ2 : Operator X bot) (U : Set X) (hU : U ⊆ dom (ϕ2 * ϕ1)) :
    Iid d ϕ1 U ∩ Iid d ϕ2 (ϕ1.func '' U) ⊆ Iid d (ϕ2 * ϕ1) U := by

section Fiber
variable {X I : Type} {bot : X} (d : I → X → X → ENNReal)

/-- 计算纤维 `Fib_{ϕ,J}(y | U) = {x ∈ U | d_J(ϕ x, y) = 0}`，`d_J` 为子族复合伪度量。 -/
def fiber (J : Set I) (ϕ : Operator X bot) (U : Set X) (y : X) : Set X :=
  {x | x ∈ U ∧ subfamily_pseudometric d J (ϕ.func x) y = 0}

/-- 全域纤维 `Fib_{ϕ,J}(y)`。 -/
abbrev fiberDom (J : Set I) (ϕ : Operator X bot) (y : X) : Set X := fiber d J ϕ (dom ϕ) y

theorem subfamily_eq_zero_iff (J : Set I) (a b : X) :
    subfamily_pseudometric d J a b = 0 ↔ ∀ j ∈ J, d j a b = 0 := by
  simp [subfamily_pseudometric, ENNReal.iSup_eq_zero]

theorem mem_fiber_iff (J : Set I) (ϕ : Operator X bot) (U : Set X) (y x : X) :
    x ∈ fiber d J ϕ U y ↔ x ∈ U ∧ ∀ j ∈ J, d j (ϕ.func x) y = 0 := by
  simp [fiber, subfamily_eq_zero_iff]

theorem mem_fiber_singleton (j : I) (ϕ : Operator X bot) (U : Set X) (y x : X) :
    x ∈ fiber d {j} ϕ U y ↔ x ∈ U ∧ d j (ϕ.func x) y = 0 := by
  simp [mem_fiber_iff]

theorem gaugeZero_iff_subfamily_univ (a b : X) :
    GaugeZeroRelated d a b ↔ subfamily_pseudometric d Set.univ a b = 0 := by
  simp [subfamily_eq_zero_iff, GaugeZeroRelated]

/-- 命题 3.5 (i)：目标零距不变性（任意目标）。 -/
theorem prop35_i (hd : ∀ i, ExtendedPseudoMetric X bot (d i)) (J : Set I)
    (ϕ : Operator X bot) (U : Set X) (hU : U ⊆ dom ϕ) (y y' : X)
    (hyy : subfamily_pseudometric d J y y' = 0) :
    fiber d J ϕ U y = fiber d J ϕ U y' := by

  ext x
  simp only [mem_fiber_iff]
  constructor
  · rintro ⟨hx, h⟩; exact ⟨hx, fun j hj => by rw [← dist_congr_right (hd j) (hyy j hj)]; exact h j hj⟩
  · rintro ⟨hx, h⟩; exact ⟨hx, fun j hj => by rw [dist_congr_right (hd j) (hyy j hj)]; exact h j hj⟩

/-- 命题 3.5 (ii-a)：以像点为目标的纤维非空、含于 `U` 且覆盖 `U`。 -/
theorem prop35_iia (hd : ∀ i, ExtendedPseudoMetric X bot (d i)) (J : Set I)
    (ϕ : Operator X bot) (U : Set X) (hU : U ⊆ dom ϕ) :
    (∀ x ∈ U, x ∈ fiber d J ϕ U (ϕ.func x)) ∧
    (∀ y ∈ ϕ.func '' U, (fiber d J ϕ U y).Nonempty ∧ fiber d J ϕ U y ⊆ U) ∧
    (⋃ y ∈ ϕ.func '' U, fiber d J ϕ U y) = U := by
  have hself : ∀ x ∈ U, x ∈ fiber d J ϕ U (ϕ.func x) := by
    intro x hx
    rw [mem_fiber_iff]
    exact ⟨hx, fun j _ => (hd j).dist_self _⟩
  refine ⟨hself, ?_, ?_⟩
  · rintro _ ⟨x, hx, rfl⟩
    exact ⟨⟨x, hself x hx⟩, fun z hz => hz.1⟩
  · ext z
    simp only [Set.mem_iUnion, exists_prop]
    constructor
    · rintro ⟨y, _, hz⟩; exact hz.1
    · intro hz; exact ⟨ϕ.func z, Set.mem_image_of_mem _ hz, hself z hz⟩

/-- 命题 3.5 (ii-b)：像目标纤维相等或不交；去重后的纤维族构成 `U` 的分划。 -/
theorem prop35_iib (hd : ∀ i, ExtendedPseudoMetric X bot (d i)) (J : Set I)
    (ϕ : Operator X bot) (U : Set X) (hU : U ⊆ dom ϕ) :
    (∀ y1 ∈ ϕ.func '' U, ∀ y2 ∈ ϕ.func '' U,
      fiber d J ϕ U y1 = fiber d J ϕ U y2 ∨ Disjoint (fiber d J ϕ U y1) (fiber d J ϕ U y2)) ∧
    (let F : Set (Set X) := {S | ∃ y ∈ ϕ.func '' U, S = fiber d J ϕ U y}
     (∀ S ∈ F, S.Nonempty ∧ S ⊆ U) ∧ ∀ x ∈ U, ∃! S, S ∈ F ∧ x ∈ S) := by
  have heq : ∀ y1 y2 : X, (fiber d J ϕ U y1 ∩ fiber d J ϕ U y2).Nonempty →
      fiber d J ϕ U y1 = fiber d J ϕ U y2 := by
    rintro y1 y2 ⟨x, hx1, hx2⟩
    apply prop35_i d hd J ϕ U hU
    rw [subfamily_eq_zero_iff]
    intro j hj
    rw [mem_fiber_iff] at hx1 hx2
    rw [dist_congr_left (hd j) (show d j y1 (ϕ.func x) = 0 by rw [(hd j).symm]; exact hx1.2 j hj)]
    exact hx2.2 j hj
  obtain ⟨hself, hne, -⟩ := prop35_iia d hd J ϕ U hU
  refine ⟨?_, ?_⟩
  · intro y1 _ y2 _
    by_cases h : (fiber d J ϕ U y1 ∩ fiber d J ϕ U y2).Nonempty
    · exact Or.inl (heq y1 y2 h)
    · right; rw [Set.disjoint_iff_inter_eq_empty]; exact Set.not_nonempty_iff_eq_empty.mp h
  · refine ⟨?_, ?_⟩
    · rintro S ⟨y, hy, rfl⟩; exact hne y hy
    · intro x hx
      refine ⟨fiber d J ϕ U (ϕ.func x), ⟨⟨ϕ.func x, Set.mem_image_of_mem _ hx, rfl⟩, hself x hx⟩, ?_⟩
      rintro S ⟨⟨y, hy, rfl⟩, hxS⟩
      exact heq _ _ ⟨x, hxS, hself x hx⟩

/-- 命题 3.5 (iii)：有限非空子族的分量交集律。 -/
theorem prop35_iii (J : Set I) (hJ : J.Nonempty) (ϕ : Operator X bot) (U : Set X)
    (hU : U ⊆ dom ϕ) (y : X) :
    fiber d J ϕ U y = ⋂ j ∈ J, fiber d {j} ϕ U y := by
  obtain ⟨j0, hj0⟩ := hJ
  ext x
  simp only [Set.mem_iInter, mem_fiber_iff, Set.mem_singleton_iff, forall_eq]
  constructor
  · rintro ⟨hx, h⟩ j hj; exact ⟨hx, h j hj⟩
  · intro h; exact ⟨(h j0 hj0).1, fun j hj => (h j hj).2⟩

/-- 命题 3.5 (iv)：子族扩大时纤维缩小。 -/
theorem prop35_iv (J K : Set I) (hJK : J ⊆ K) (ϕ : Operator X bot) (U : Set X) (y : X) :
    fiber d K ϕ U y ⊆ fiber d J ϕ U y := by
  intro x hx
  rw [mem_fiber_iff] at hx ⊢
  exact ⟨hx.1, fun j hj => hx.2 j (hJK hj)⟩

/-- 命题 3.5 (v)：单分量等式与全子族向每个分量的包含。 -/
theorem prop35_v (ϕ : Operator X bot) (U : Set X) (y : X) (i : I) :
    fiber d {i} ϕ U y = {x | x ∈ U ∧ d i (ϕ.func x) y = 0} ∧
    fiber d Set.univ ϕ U y ⊆ fiber d {i} ϕ U y := by
  refine ⟨?_, prop35_iv d {i} Set.univ (Set.subset_univ _) ϕ U y⟩
  ext x; exact mem_fiber_singleton d i ϕ U y x

/-- 命题 3.5 (vi-a)：恒等分量的任意目标纤维等于输入零距类。 -/
theorem prop35_via (hd : ∀ i, ExtendedPseudoMetric X bot (d i)) (ϕ : Operator X bot)
    (U : Set X) (hU : U ⊆ dom ϕ) (j : I) (hj : j ∈ Iid d ϕ U) (y : X) :
    fiber d {j} ϕ U y = {x | x ∈ U ∧ d j x y = 0} := by
  ext x
  simp only [mem_fiber_singleton, Set.mem_setOf_eq]
  constructor
  · rintro ⟨hx, h⟩; exact ⟨hx, by rwa [← dist_congr_left (hd j) (hj.2 x hx)]⟩
  · rintro ⟨hx, h⟩; exact ⟨hx, by rwa [dist_congr_left (hd j) (hj.2 x hx)]⟩

/-- 命题 3.5 (vi-b)：常值分量在像目标的纤维为 `U`，正距离目标的纤维为空。 -/
theorem prop35_vib (hd : ∀ i, ExtendedPseudoMetric X bot (d i)) (ϕ : Operator X bot)
    (U : Set X) (hU : U ⊆ dom ϕ) (j : I) (hj : j ∈ Iconst d ϕ U) (y0 : X)
    (hy0 : y0 ∈ ϕ.func '' U) :
    fiber d {j} ϕ U y0 = U ∧ ∀ y, 0 < d j y y0 → fiber d {j} ϕ U y = ∅ := by
  obtain ⟨x0, hx0, rfl⟩ := hy0
  have hall : ∀ x ∈ U, d j (ϕ.func x) (ϕ.func x0) = 0 := fun x hx => hj x hx x0 hx0
  refine ⟨?_, ?_⟩
  · ext x; rw [mem_fiber_singleton]; exact ⟨fun h => h.1, fun hx => ⟨hx, hall x hx⟩⟩
  · intro y hy
    ext x
    simp only [mem_fiber_singleton, Set.mem_empty_iff_false, iff_false, not_and]
    intro hx hxy
    have : d j y (ϕ.func x0) = 0 := by
      rw [dist_congr_left (hd j) (show d j y (ϕ.func x) = 0 by rw [(hd j).symm]; exact hxy)]
      exact hall x hx
    rw [this] at hy
    exact lt_irrefl _ hy

end Fiber

section Relations
variable {X I : Type} {bot : X} (d : I → X → X → ENNReal)

/-- `R_=(ϕ | U)`. -/
def Req (ϕ : Operator X bot) (U : Set X) : Set (X × X) :=
  {p | p.1 ∈ U ∧ p.2 ∈ U ∧ ϕ.func p.1 = ϕ.func p.2}

/-- `R_∼(ϕ | U)`. -/
def Rsim (ϕ : Operator X bot) (U : Set X) : Set (X × X) :=
  {p | p.1 ∈ U ∧ p.2 ∈ U ∧ GaugeZeroRelated d (ϕ.func p.1) (ϕ.func p.2)}

/-- `R_∼` 是计算纤维的关系表达。 -/
theorem Rsim_iff_fiber (hd : ∀ i, ExtendedPseudoMetric X bot (d i)) (ϕ : Operator X bot)
    (U : Set X) (x y : X) (hx : x ∈ U) (hy : y ∈ U) :
    (x, y) ∈ Rsim d ϕ U ↔ y ∈ fiber d Set.univ ϕ U (ϕ.func x) := by
  rw [mem_fiber_iff]
  simp only [Rsim, Set.mem_setOf_eq, GaugeZeroRelated, Set.mem_univ, true_implies]
  constructor
  · rintro ⟨-, -, h⟩; exact ⟨hy, fun j => by rw [(hd j).symm]; exact h j⟩
  · rintro ⟨-, h⟩; exact ⟨hx, hy, fun j => by rw [(hd j).symm]; exact h j⟩

theorem Req_subset_Rsim (hd : ∀ i, ExtendedPseudoMetric X bot (d i)) (ϕ : Operator X bot)
    (U : Set X) : Req ϕ U ⊆ Rsim d ϕ U := by
  rintro ⟨x, y⟩ ⟨hx, hy, h⟩
  exact ⟨hx, hy, fun i => by simp only at h ⊢; rw [h]; exact (hd i).dist_self _⟩

/-- 族零距关系作为 `X` 上的 setoid（引理 1.1）。 -/
def gaugeSetoid (hd : ∀ i, ExtendedPseudoMetric X bot (d i)) : Setoid X :=
  ⟨GaugeZeroRelated d, gauge_zero_related_equivalence d (fun i => (hd i).dist_self)
    (fun i => (hd i).symm) (fun i => (hd i).triangle)⟩

/-- 像集 `ϕ(U)` 上限制的族零距关系。 -/
def imageSetoid (hd : ∀ i, ExtendedPseudoMetric X bot (d i)) (ϕ : Operator X bot)
    (U : Set X) : Setoid (ϕ.func '' U) :=
  ⟨fun a b => GaugeZeroRelated d a.1 b.1,
    ⟨fun a => (gaugeSetoid d hd).iseqv.refl a.1,
     fun h => (gaugeSetoid d hd).iseqv.symm h,
     fun h1 h2 => (gaugeSetoid d hd).iseqv.trans h1 h2⟩⟩

/-- 像商 `ϕ(U)/∼_G`。 -/
abbrev ImageQuot (hd : ∀ i, ExtendedPseudoMetric X bot (d i)) (ϕ : Operator X bot)
    (U : Set X) := Quotient (imageSetoid d hd ϕ U)

/-- 代表元 `[ϕ x]`。 -/
def qmk (hd : ∀ i, ExtendedPseudoMetric X bot (d i)) (ϕ : Operator X bot) (U : Set X)
    (x : X) (hx : x ∈ U) : ImageQuot d hd ϕ U :=
  Quotient.mk (imageSetoid d hd ϕ U) ⟨ϕ.func x, Set.mem_image_of_mem _ hx⟩

/-- 零测距纤维预序 `ϕA ≽_{fiber,U} ϕB`。 -/
def fiberGe (ϕA ϕB : Operator X bot) (U : Set X) : Prop :=
  Rsim d ϕA U ⊆ Rsim d ϕB U

/-! ### 命题 3.6 -/

/-- `T : ϕA(U) → X` 满足 `T ∘ ϕA = ϕB`（在 `U` 上）。 -/
def IsImageFactor (ϕA ϕB : Operator X bot) (U : Set X) (T : ϕA.func '' U → X) : Prop :=
  ∀ x (hx : x ∈ U), T ⟨ϕA.func x, Set.mem_image_of_mem _ hx⟩ = ϕB.func x

/-- 命题 3.6 (i-必要)。 -/
theorem prop36_i_nec (ϕA ϕB : Operator X bot) (U : Set X) (hU : U ⊆ dom ϕA ∩ dom ϕB)
    (h : ∃ T, IsImageFactor ϕA ϕB U T) : Req ϕA U ⊆ Req ϕB U := by
  obtain ⟨T, hT⟩ := h
  rintro ⟨x, y⟩ ⟨hx, hy, hxy⟩
  refine ⟨hx, hy, ?_⟩
  simp only at hxy ⊢
  rw [← hT x hx, ← hT y hy]
  congr 1
  exact Subtype.ext hxy

/-- 命题 3.6 (i-充分)：由关系包含实际构造良定义映射（含 `U = ∅`）。 -/
theorem prop36_i_suff (ϕA ϕB : Operator X bot) (U : Set X) (hU : U ⊆ dom ϕA ∩ dom ϕB)
    (h : Req ϕA U ⊆ Req ϕB U) :
    IsImageFactor ϕA ϕB U (fun z => ϕB.func (preim ϕA U z)) := by
  intro x hx
  have hs := preim_spec ϕA U ⟨ϕA.func x, Set.mem_image_of_mem _ hx⟩
  have := h (show (preim ϕA U ⟨ϕA.func x, Set.mem_image_of_mem _ hx⟩, x) ∈ Req ϕA U from
    ⟨hs.1, hx, hs.2⟩)
  exact this.2.2

/-- 命题 3.6 (i)。 -/
theorem prop36_i (ϕA ϕB : Operator X bot) (U : Set X) (hU : U ⊆ dom ϕA ∩ dom ϕB) :
    (∃ T, IsImageFactor ϕA ϕB U T) ↔ Req ϕA U ⊆ Req ϕB U :=
  ⟨prop36_i_nec ϕA ϕB U hU, fun h => ⟨_, prop36_i_suff ϕA ϕB U hU h⟩⟩

/-- 命题 3.6 (i)，`U = ∅` 边界：像值映射总存在。 -/
theorem prop36_empty (ϕA ϕB : Operator X bot) :
    ∃ T, IsImageFactor ϕA ϕB (∅ : Set X) T :=
  (prop36_i ϕA ϕB ∅ (Set.empty_subset _)).2 (by intro p hp; exact absurd hp.1 (Set.notMem_empty _))

/-- 命题 3.6 (ii)。 -/
theorem prop36_ii (ϕA ϕB : Operator X bot) (U : Set X) (hU : U ⊆ dom ϕA ∩ dom ϕB) :
    ((∃ T, IsImageFactor ϕA ϕB U T) ∧ (∃ S, IsImageFactor ϕB ϕA U S)) ↔
      Req ϕA U = Req ϕB U := by
  rw [prop36_i ϕA ϕB U hU, prop36_i ϕB ϕA U (by rwa [Set.inter_comm]),
    Set.Subset.antisymm_iff]

/-- 命题 3.6 (iii)：单分量情形下的标量 Lip 界 `L_B / K_A`（在实际像域 `ϕA(U)` 上）。 -/
theorem prop36_iii (e : X → X → ENNReal) (ϕA ϕB : Operator X bot) (U : Set X)
    (hU : U ⊆ dom ϕA ∩ dom ϕB) (T : ϕA.func '' U → X) (hT : IsImageFactor ϕA ϕB U T)
    (KA LB : ENNReal) (hK0 : 0 < KA) (hKtop : KA < ⊤) (hLtop : LB < ⊤)
    (hcoA : ∀ x ∈ U, ∀ x' ∈ U, KA * e x x' ≤ e (ϕA.func x) (ϕA.func x'))
    (hlipB : ∀ x ∈ U, ∀ x' ∈ U, e (ϕB.func x) (ϕB.func x') ≤ LB * e x x') :
    ∀ y y' : ϕA.func '' U, e (T y) (T y') ≤ (LB / KA) * e y y' := by
  intro y y'
  obtain ⟨hx, hxy⟩ := preim_spec ϕA U y
  obtain ⟨hx', hxy'⟩ := preim_spec ϕA U y'
  set x := preim ϕA U y
  set x' := preim ϕA U y'
  have hy : y = ⟨ϕA.func x, Set.mem_image_of_mem _ hx⟩ := Subtype.ext hxy.symm
  have hy' : y' = ⟨ϕA.func x', Set.mem_image_of_mem _ hx'⟩ := Subtype.ext hxy'.symm
  rw [hy, hy', hT x hx, hT x' hx']
  simp only
  have h1 : e x x' ≤ e (ϕA.func x) (ϕA.func x') / KA := by
    rw [ENNReal.le_div_iff_mul_le (Or.inl hK0.ne') (Or.inl hKtop.ne), mul_comm]
    exact hcoA x hx x' hx'
  calc e (ϕB.func x) (ϕB.func x') ≤ LB * e x x' := hlipB x hx x' hx'
    _ ≤ LB * (e (ϕA.func x) (ϕA.func x') / KA) := mul_le_mul_right h1 _
    _ = LB / KA * e (ϕA.func x) (ϕA.func x') := by
      rw [div_eq_mul_inv, div_eq_mul_inv]; ring

/-- 单元素指标集上的有限和退化为单项。 -/
theorem sum_subsingleton_eq {I : Type} [Fintype I] [Subsingleton I] (i : I) (f : I → ENNReal) :
    ∑ k : I, f k = f i := by
  apply Finset.sum_eq_single i
  · intro k _ hk; exact absurd (Subsingleton.elim k i) hk
  · intro h; exact absurd (Finset.mem_univ i) h

/-- 命题 3.6 (iii)（矩阵证书形式）：`I = {i}` 时，由 `ϕA` 的 co-Lip 矩阵与 `ϕB` 的 Lip 矩阵
（`0 < K_{i→i} < ⊤`，`L_{i→i} < ⊤`）得到 `T` 在 `ϕA(U)` 上的 `(L_{i→i}/K_{i→i})`-Lip 界。 -/
theorem prop36_iii_matrix [Fintype I] [Subsingleton I] (i : I) (ϕA ϕB : Operator X bot)
    (U : Set X) (hU : U ⊆ dom ϕA ∩ dom ϕB) (T : ϕA.func '' U → X) (hT : IsImageFactor ϕA ϕB U T)
    (K L : I → I → ENNReal) (hK : IsCoLipMatrix d ϕA U K) (hL : IsLipMatrix d ϕB U L)
    (hK0 : 0 < K i i) (hKt : K i i < ⊤) (hLt : L i i < ⊤) :
    ∀ y y' : ϕA.func '' U, d i (T y) (T y') ≤ (L i i / K i i) * d i y y' := by
  apply prop36_iii (d i) ϕA ϕB U hU T hT (K i i) (L i i) hK0 hKt hLt
  · intro x hx x' hx'
    have h := hK.2.2 x hx x' hx' i
    rw [sum_subsingleton_eq i] at h
    exact h
  · exact single_component_lip_matrix_ineq d ϕB U L i hL

/-- 命题 3.7（单向）：满足规定公式的商映射存在 ⟺ 零距关系包含。 -/
theorem prop37_one (hd : ∀ i, ExtendedPseudoMetric X bot (d i)) (ϕA ϕB : Operator X bot)
    (U : Set X) (hU : U ⊆ dom ϕA ∩ dom ϕB) :
    (∃ T : ImageQuot d hd ϕA U → ImageQuot d hd ϕB U,
        ∀ x (hx : x ∈ U), T (qmk d hd ϕA U x hx) = qmk d hd ϕB U x hx) ↔
      Rsim d ϕA U ⊆ Rsim d ϕB U := by
  constructor
  · rintro ⟨T, hT⟩ ⟨x, y⟩ ⟨hx, hy, hrel⟩
    refine ⟨hx, hy, ?_⟩
    have h1 : qmk d hd ϕA U x hx = qmk d hd ϕA U y hy := Quotient.sound hrel
    have h2 := congrArg T h1
    rw [hT x hx, hT y hy] at h2
    exact Quotient.exact h2
  · intro h
    refine ⟨Quotient.lift (fun z => Quotient.mk (imageSetoid d hd ϕB U)
        ⟨ϕB.func (preim ϕA U z), Set.mem_image_of_mem _ (preim_spec ϕA U z).1⟩) ?_, ?_⟩
    · intro z w hzw
      apply Quotient.sound
      have hz := preim_spec ϕA U z
      have hw := preim_spec ϕA U w
      have hmem : (preim ϕA U z, preim ϕA U w) ∈ Rsim d ϕA U := by
        refine ⟨hz.1, hw.1, ?_⟩
        simp only
        rw [hz.2, hw.2]
        exact hzw
      exact (h hmem).2.2
    · intro x hx
      show Quotient.mk _ _ = Quotient.mk _ _
      apply Quotient.sound
      have hs := preim_spec ϕA U ⟨ϕA.func x, Set.mem_image_of_mem _ hx⟩
      have hmem : (preim ϕA U ⟨ϕA.func x, Set.mem_image_of_mem _ hx⟩, x) ∈ Rsim d ϕA U := by
        refine ⟨hs.1, hx, ?_⟩
        simp only
        rw [hs.2]
        exact (gaugeSetoid d hd).iseqv.refl _
      exact (h hmem).2.2

/-- 命题 3.7（双向）：双向商映射存在 ⟺ 零距关系相等。 -/
theorem prop37_two (hd : ∀ i, ExtendedPseudoMetric X bot (d i)) (ϕA ϕB : Operator X bot)
    (U : Set X) (hU : U ⊆ dom ϕA ∩ dom ϕB) :
    ((∃ T : ImageQuot d hd ϕA U → ImageQuot d hd ϕB U,
        ∀ x (hx : x ∈ U), T (qmk d hd ϕA U x hx) = qmk d hd ϕB U x hx) ∧
     (∃ S : ImageQuot d hd ϕB U → ImageQuot d hd ϕA U,
        ∀ x (hx : x ∈ U), S (qmk d hd ϕB U x hx) = qmk d hd ϕA U x hx)) ↔
      Rsim d ϕA U = Rsim d ϕB U := by
  rw [prop37_one d hd ϕA ϕB U hU, prop37_one d hd ϕB ϕA U (by rwa [Set.inter_comm]),
    Set.Subset.antisymm_iff]

/-- 规范族在集合 `S` 内分离。 -/
def SeparatesOn (S : Set X) : Prop :=
  ∀ z ∈ S, ∀ w ∈ S, GaugeZeroRelated d z w → z = w

theorem Req_eq_Rsim_of_sep (hd : ∀ i, ExtendedPseudoMetric X bot (d i))
    (ϕ : Operator X bot) (U : Set X) (hsep : SeparatesOn d (ϕ.func '' U)) :
    Req ϕ U = Rsim d ϕ U := by
  apply Set.Subset.antisymm (Req_subset_Rsim d hd ϕ U)
  rintro ⟨x, y⟩ ⟨hx, hy, h⟩
  exact ⟨hx, hy, hsep _ (Set.mem_image_of_mem _ hx) _ (Set.mem_image_of_mem _ hy) h⟩

/-- 命题 3.7（分离情形）：两个实际像集各自分离时，点值判据与商判据一致。 -/
theorem prop37_sep (hd : ∀ i, ExtendedPseudoMetric X bot (d i)) (ϕA ϕB : Operator X bot)
    (U : Set X) (hU : U ⊆ dom ϕA ∩ dom ϕB)
    (hsepA : SeparatesOn d (ϕA.func '' U)) (hsepB : SeparatesOn d (ϕB.func '' U)) :
    Req ϕA U = Rsim d ϕA U ∧ Req ϕB U = Rsim d ϕB U ∧
    (Req ϕA U ⊆ Req ϕB U ↔ Rsim d ϕA U ⊆ Rsim d ϕB U) ∧
    (Req ϕA U = Req ϕB U ↔ Rsim d ϕA U = Rsim d ϕB U) ∧
    ((∃ T, IsImageFactor ϕA ϕB U T) ↔
      ∃ T : ImageQuot d hd ϕA U → ImageQuot d hd ϕB U,
        ∀ x (hx : x ∈ U), T (qmk d hd ϕA U x hx) = qmk d hd ϕB U x hx) ∧
    (((∃ T, IsImageFactor ϕA ϕB U T) ∧ (∃ S, IsImageFactor ϕB ϕA U S)) ↔
      ((∃ T : ImageQuot d hd ϕA U → ImageQuot d hd ϕB U,
        ∀ x (hx : x ∈ U), T (qmk d hd ϕA U x hx) = qmk d hd ϕB U x hx) ∧
       (∃ S : ImageQuot d hd ϕB U → ImageQuot d hd ϕA U,
        ∀ x (hx : x ∈ U), S (qmk d hd ϕB U x hx) = qmk d hd ϕA U x hx))) := by
  have hA := Req_eq_Rsim_of_sep d hd ϕA U hsepA
  have hB := Req_eq_Rsim_of_sep d hd ϕB U hsepB
  refine ⟨hA, hB, by rw [hA, hB], by rw [hA, hB], ?_, ?_⟩
  · rw [prop36_i ϕA ϕB U hU, prop37_one d hd ϕA ϕB U hU, hA, hB]
  · rw [prop36_ii ϕA ϕB U hU, prop37_two d hd ϕA ϕB U hU, hA, hB]

/-! ### 注（纤维包含的传递） -/

theorem fiberGe_refl (ϕ : Operator X bot) (U : Set X) : fiberGe d ϕ ϕ U :=
  Set.Subset.refl _

theorem fiberGe_trans {ϕa ϕb ϕc : Operator X bot} {U : Set X}
    (h1 : fiberGe d ϕa ϕb U) (h2 : fiberGe d ϕb ϕc U) : fiberGe d ϕa ϕc U :=
  Set.Subset.trans h1 h2

/-- 纤维预序在固定公共域上是预序。 -/
theorem fiberGe_isPreorder (U : Set X) :
    (∀ ϕ : Operator X bot, fiberGe d ϕ ϕ U) ∧
    (∀ ϕa ϕb ϕc : Operator X bot, fiberGe d ϕa ϕb U → fiberGe d ϕb ϕc U → fiberGe d ϕa ϕc U) :=
  ⟨fun ϕ => fiberGe_refl d ϕ U, fun _ _ _ h1 h2 => fiberGe_trans d h1 h2⟩

theorem fiberGe_chain_le (ϕ : ℕ → Operator X bot) (U : Set X) (m : ℕ)
    (hstep : ∀ r < m, fiberGe d (ϕ r) (ϕ (r + 1)) U) :
    ∀ a b, a ≤ b → b ≤ m → fiberGe d (ϕ a) (ϕ b) U := by
  intro a b hab
  induction b, hab using Nat.le_induction with
  | base => intro _; exact fiberGe_refl d _ U
  | succ b hab ih =>
    intro hb
    exact fiberGe_trans d (ih (by omega)) (hstep b (by omega))

/-- 注：有限链任意两层 `a < b` 的关系包含。 -/
theorem fiberGe_chain (ϕ : ℕ → Operator X bot) (U : Set X) (m : ℕ)
    (hU : ∀ k ≤ m, U ⊆ dom (ϕ k))
    (hstep : ∀ r < m, fiberGe d (ϕ r) (ϕ (r + 1)) U) :
    ∀ a b, a < b → b ≤ m → fiberGe d (ϕ a) (ϕ b) U :=
  fun a b hab hb => fiberGe_chain_le d ϕ U m hstep a b hab.le hb

/-- 注：链中某一步严格包含推出端点严格包含，并给出差集见证。 -/
theorem fiberGe_chain_strict (ϕ : ℕ → Operator X bot) (U : Set X) (m : ℕ)
    (hU : ∀ k ≤ m, U ⊆ dom (ϕ k))
    (hstep : ∀ r < m, fiberGe d (ϕ r) (ϕ (r + 1)) U)
    (r : ℕ) (hr : r < m) (hstrict : Rsim d (ϕ r) U ⊂ Rsim d (ϕ (r + 1)) U) :
    (∃ p, p ∈ Rsim d (ϕ (r + 1)) U \ Rsim d (ϕ r) U ∧
        p ∈ Rsim d (ϕ m) U \ Rsim d (ϕ 0) U) ∧
    Rsim d (ϕ 0) U ⊂ Rsim d (ϕ m) U := by

  have h0m : Rsim d (ϕ 0) U ⊆ Rsim d (ϕ m) U := fiberGe_chain_le d ϕ U m hstep 0 m (by omega) le_rfl

  refine ⟨⟨p, ⟨hp1, hp2⟩, hw⟩, ?_⟩
  exact Set.ssubset_iff_subset_ne.mpr ⟨h0m, fun h => hw.2 (h ▸ hw.1)⟩

section Radius
variable {X : Type} {bot : X}

/-- 有限正半径 `r` 对基点 `y` 合法：`B_{din}(y,r) ∩ U ⊆ {x ∈ U | dout(ϕ x, ϕ y) = 0}`。 -/
def IsNullRad (din dout : X → X → ENNReal) (ϕ : Operator X bot) (U : Set X) (y : X)
    (r : ENNReal) : Prop :=
  0 < r ∧ r < ⊤ ∧ ball din y r ∩ U ⊆ {x | x ∈ U ∧ dout (ϕ.func x) (ϕ.func y) = 0}

/-- 零化内切半径 `r^N_{ϕ,din,dout}(y | U)`（`sSup ∅ = 0`）。 -/
noncomputable def nullRadius (din dout : X → X → ENNReal) (ϕ : Operator X bot) (U : Set X)
    (y : X) : ENNReal :=
  sSup {r | IsNullRad din dout ϕ U y r}

/-- 全域简写 `r^N_{ϕ,din,dout}(y)`。 -/
noncomputable abbrev nullRadiusDom (din dout : X → X → ENNReal) (ϕ : Operator X bot) (y : X) : ENNReal :=
  nullRadius din dout ϕ (dom ϕ) y

/-! ## D05 统一零化半径与零化矩阵 -/

/-- 统一零化半径 `r̲^N_{ϕ,din,dout}(A | U) = inf_{y∈A} r^N(y | U)`。 -/
noncomputable def uniformNullRadius (din dout : X → X → ENNReal) (ϕ : Operator X bot)
    (A U : Set X) : ENNReal :=
  ⨅ y ∈ A, nullRadius din dout ϕ U y

/-- 零化矩阵：`nullMatrix d ϕ U i j = [N(ϕ)|_U]_{ji} = r̲^N_{ϕ,i,j}|_U`。 -/
noncomputable def nullMatrix {I : Type} (d : I → X → X → ENNReal) (ϕ : Operator X bot)
    (U : Set X) : I → I → ENNReal :=
  fun i j => uniformNullRadius (d i) (d j) ϕ U U

/-- 全域零化矩阵 `N(ϕ)`。 -/
noncomputable abbrev nullMatrixDom {I : Type} (d : I → X → X → ENNReal) (ϕ : Operator X bot) :
    I → I → ENNReal :=
  nullMatrix d ϕ (dom ϕ)

variable {din dout : X → X → ENNReal} {ϕ : Operator X bot} {U : Set X} {y : X}

theorem isNullRad_iff {r : ENNReal} :
    IsNullRad din dout ϕ U y r ↔
      0 < r ∧ r < ⊤ ∧ ∀ x ∈ U, din x y < r → dout (ϕ.func x) (ϕ.func y) = 0 := by
  unfold IsNullRad
  constructor
  · rintro ⟨h0, h1, h2⟩; exact ⟨h0, h1, fun x hx hxr => (h2 ⟨hxr, hx⟩).2⟩
  · rintro ⟨h0, h1, h2⟩; exact ⟨h0, h1, fun x ⟨hxr, hx⟩ => ⟨hx, h2 x hx hxr⟩⟩

theorem le_nullRadius {r : ENNReal} (h : IsNullRad din dout ϕ U y r) :
    r ≤ nullRadius din dout ϕ U y := le_sSup h

theorem isNullRad_mono {r r' : ENNReal} (h : IsNullRad din dout ϕ U y r) (h0 : 0 < r')
    (hle : r' ≤ r) : IsNullRad din dout ϕ U y r' := by
  rw [isNullRad_iff] at h ⊢
  exact ⟨h0, lt_of_le_of_lt hle h.2.1, fun x hx hxr => h.2.2 x hx (lt_of_lt_of_le hxr hle)⟩

theorem le_nullRadius_of_forall_lt {M : ENNReal}
    (h : ∀ r, 0 < r → r < ⊤ → r < M → IsNullRad din dout ϕ U y r) :
    M ≤ nullRadius din dout ϕ U y := by
  refine le_of_forall_lt fun c hc => ?_
  obtain ⟨r, hcr, hrM⟩ := exists_between hc
  exact lt_of_lt_of_le hcr
    (le_nullRadius (h r (lt_of_le_of_lt (zero_le c) hcr) (lt_of_lt_of_le hrM le_top) hrM))

theorem nullRadius_le_of_forall {c : ENNReal}
    (h : ∀ r, IsNullRad din dout ϕ U y r → r ≤ c) : nullRadius din dout ϕ U y ≤ c :=
  sSup_le h

theorem nullRadius_eq_top_of_forall
    (h : ∀ r, 0 < r → r < ⊤ → IsNullRad din dout ϕ U y r) :
    nullRadius din dout ϕ U y = ⊤ :=
  top_unique (le_nullRadius_of_forall_lt fun r h0 h1 _ => h r h0 h1)

theorem nullRadius_eq_top_of_const (h : ∀ x ∈ U, dout (ϕ.func x) (ϕ.func y) = 0) :
    nullRadius din dout ϕ U y = ⊤ :=
  nullRadius_eq_top_of_forall fun r h0 h1 => isNullRad_iff.2 ⟨h0, h1, fun x hx _ => h x hx⟩

/-- 命题 3.11 的核心：严格小于点态上确界时取出合法有限正半径，推出零距。 -/
theorem zero_of_lt_nullRadius (hsin : ∀ a b, din a b = din b a)
    (hsout : ∀ a b, dout a b = dout b a) {y' : X} (hy' : y' ∈ U)
    (h : din y y' < nullRadius din dout ϕ U y) : dout (ϕ.func y) (ϕ.func y') = 0 := by
  obtain ⟨ρ, hρ, hlt⟩ := lt_sSup_iff.mp h
  have hρ' : IsNullRad din dout ϕ U y ρ := hρ
  rw [isNullRad_iff] at hρ'
  rw [hsout]
  exact hρ'.2.2 y' hy' (by rw [hsin]; exact hlt)

theorem uniform_le_nullRadius {A : Set X} (hy : y ∈ A) :
    uniformNullRadius din dout ϕ A U ≤ nullRadius din dout ϕ U y :=
  biInf_le _ hy

theorem le_uniform {A : Set X} {c : ENNReal} (h : ∀ y ∈ A, c ≤ nullRadius din dout ϕ U y) :
    c ≤ uniformNullRadius din dout ϕ A U :=
  le_iInf₂ h

  have h2 := hpres _ (Set.mem_image_of_mem _ hx) _ (Set.mem_image_of_mem _ hx') h1

  rw [(hd j).symm]; exact h2

    (U : Set X) (hU0 : U.Nonempty) (hU : U ⊆ dom (ϕ2 * ϕ1)) (i j l : I)

    (hlip : ∀ x ∈ U, ∀ x' ∈ U, d l (ϕ1.func x) (ϕ1.func x') ≤ L * d i x x') :
    nullMatrix d ϕ2 (ϕ1.func '' U) l j / L ≤ nullMatrix d (ϕ2 * ϕ1) U i j := by
  refine le_uniform fun x hx => le_nullRadius_of_forall_lt fun r h0 htop hr => ?_
  refine isNullRad_iff.2 ⟨h0, htop, fun x' hx' hdist => ?_⟩
  have hrL : r * L < nullMatrix d ϕ2 (ϕ1.func '' U) l j :=
    (ENNReal.lt_div_iff_mul_lt (Or.inl hL0.ne') (Or.inl hLtop.ne)).1 hr
  have hmul : L * d i x x' < L * r := by
    rw [(hd i).symm]
    exact ENNReal.mul_lt_mul_right hL0.ne' hLtop.ne hdist
  have hlt : d l (ϕ1.func x) (ϕ1.func x') <
      nullRadius (d l) (d j) ϕ2 (ϕ1.func '' U) (ϕ1.func x) := by
    calc d l (ϕ1.func x) (ϕ1.func x') ≤ L * d i x x' := hlip x hx x' hx'
      _ < L * r := hmul
      _ = r * L := mul_comm _ _
      _ < nullMatrix d ϕ2 (ϕ1.func '' U) l j := hrL
      _ ≤ _ := uniform_le_nullRadius (Set.mem_image_of_mem _ hx)
  have h1 := zero_of_lt_nullRadius (hd l).symm (hd j).symm (Set.mem_image_of_mem _ hx') hlt
  simp only [mul_func]
  rw [(hd j).symm]; exact h1

/-- 输入尺度函数族：定义在非空子集上、关于包含单调（空集上的取值不参与任何条件）。 -/
structure InputScale (X I : Type) where
  σ : I → Set X → ENNReal
  mono : ∀ (i : I) (A1 A2 : Set X), A1.Nonempty → A1 ⊆ A2 → σ i A1 ≤ σ i A2

/-- 直径 `diam_e(A) = sup_{x,y∈A} e(x,y)`。 -/
noncomputable def diam (e : X → X → ENNReal) (A : Set X) : ENNReal :=
  ⨆ x ∈ A, ⨆ y ∈ A, e x y

theorem diam_mono (e : X → X → ENNReal) {A B : Set X} (h : A ⊆ B) : diam e A ≤ diam e B :=
  iSup₂_le fun x hx => iSup₂_le fun y hy =>
    le_iSup₂_of_le (f := fun x _ => ⨆ y ∈ B, e x y) x (h hx)
      (le_iSup₂_of_le (f := fun y _ => e x y) y (h hy) le_rfl)

theorem le_diam (e : X → X → ENNReal) {A : Set X} {x y : X} (hx : x ∈ A) (hy : y ∈ A) :
    e x y ≤ diam e A :=
  le_iSup₂_of_le (f := fun x _ => ⨆ y ∈ A, e x y) x hx
    (le_iSup₂_of_le (f := fun y _ => e x y) y hy le_rfl)

/-- 默认直径尺度。 -/

  σ i A := diam (d i) A

/-- 引理 3.13：四种组合互斥且穷尽，并与表格条件一一对应。 -/
theorem lemma313_classify (S : InputScale X I) (τσ τN : ENNReal) (d : I → X → X → ENNReal)

    thresholdType S τσ τN d f A i j ∈ ({(1, 1), (1, 0), (0, 1), (0, 0)} : Set (Fin 2 × Fin 2)) ∧

    (thresholdType S τσ τN d f A i j = (0, 0) ↔ τσ ≤ S.σ i A ∧ nullMatrix d f A i j < τN) := by
  unfold thresholdType
  by_cases h1 : S.σ i A < τσ <;> by_cases h2 : τN ≤ nullMatrix d f A i j <;>

/-- 命题 3.14 (ii-a)：基点在 `B` 内时缩小评估域使点态半径不减。 -/
theorem prop314_iia (din dout : X → X → ENNReal) (f : Operator X bot) (A B : Set X)
    (hBA : B ⊆ A) (x : X) (hx : x ∈ B) :
    nullRadius din dout f A x ≤ nullRadius din dout f B x := by
  apply sSup_le_sSup
  intro r hr
  have hr' : IsNullRad din dout f A x r := hr
  rw [isNullRad_iff] at hr'
  exact isNullRad_iff.2 ⟨hr'.1, hr'.2.1, fun z hz hzr => hr'.2.2 z (hBA hz) hzr⟩

    nullMatrix d f A i j ≤ nullMatrix d f B i j :=
  le_uniform fun x hx =>
    (uniform_le_nullRadius (hBA hx)).trans (prop314_iia (d i) (d j) f A B hBA x hx)

section Separation
variable {X I : Type} {bot : X} (d : I → X → X → ENNReal)

/-- 命题 3.15 (i)：严格低于统一半径的输入点对输出零距。 -/
theorem prop315_i (hd : ∀ i, ExtendedPseudoMetric X bot (d i)) (f : Operator X bot)
    (A : Set X) (hA0 : A.Nonempty) (hA : A ⊆ dom f) (i j : I) :
    ∀ x ∈ A, ∀ y ∈ A, d i x y < nullMatrix d f A i j → d j (f.func x) (f.func y) = 0 :=
  fun x hx y hy h => zero_of_lt_nullRadius (hd i).symm (hd j).symm hy
    (lt_of_lt_of_le h (uniform_le_nullRadius hx))

/-- 命题 3.15 (ii)：输出正分离推出输入距离不小于统一半径。 -/
theorem prop315_ii (hd : ∀ i, ExtendedPseudoMetric X bot (d i)) (f : Operator X bot)
    (A : Set X) (hA0 : A.Nonempty) (hA : A ⊆ dom f) (i j : I) :
    ∀ x ∈ A, ∀ y ∈ A, 0 < d j (f.func x) (f.func y) → nullMatrix d f A i j ≤ d i x y := by
  intro x hx y hy hpos
  by_contra h
  rw [not_le] at h
  rw [prop315_i d hd f A hA0 hA i j x hx y hy h] at hpos
  exact lt_irrefl _ hpos

/-- 命题 3.15 (iii)：零化半径不足时，对每个规定的有限正 `α` 给出分离见证（含输出距离为 `⊤`）。 -/
theorem prop315_iii (hd : ∀ i, ExtendedPseudoMetric X bot (d i)) (f : Operator X bot)
    (A : Set X) (hA0 : A.Nonempty) (hA : A ⊆ dom f) (i j : I) (τN : ENNReal) (hτ0 : 0 < τN)
    (hτ : τN < ⊤) (hlt : nullMatrix d f A i j < τN) (α : ENNReal) (hα0 : 0 < α) (hαt : α < ⊤)
    (hα1 : nullMatrix d f A i j < α) (hα2 : α < τN) :
    ∃ x ∈ A, ∃ y ∈ A, d i x y < α ∧
      ∃ ε : ENNReal, 0 < ε ∧ ε < ⊤ ∧ ε ≤ d j (f.func x) (f.func y) := by
  have : ∃ y ∈ A, nullRadius (d i) (d j) f A y < α := by
    have h := hα1
    unfold nullMatrix uniformNullRadius at h
    rw [iInf_lt_iff] at h
    obtain ⟨y, hy⟩ := h
    rw [iInf_lt_iff] at hy
    obtain ⟨hyA, hy⟩ := hy
    exact ⟨y, hyA, hy⟩
  obtain ⟨y, hyA, hyα⟩ := this
  have hnot : ¬ IsNullRad (d i) (d j) f A y α := fun h => absurd (le_nullRadius h) (not_le.mpr hyα)
  rw [isNullRad_iff] at hnot
  push Not at hnot
  obtain ⟨x, hxA, hxα, hne⟩ := hnot hα0 hαt
  refine ⟨x, hxA, y, hyA, hxα, min (d j (f.func x) (f.func y)) 1, ?_, ?_, min_le_left _ _⟩
  · exact lt_min (pos_iff_ne_zero.mpr hne) one_pos
  · exact lt_of_le_of_lt (min_le_right _ _) ENNReal.one_lt_top

end Separation

theorem covering_number_le_card {e : X → X → ENNReal} {ε : ENNReal} {U : Set X}
    (S : Finset X) (h : InternalEpsilonNet e ε U S) : covering_number e ε U ≤ S.card :=
  iInf_le_of_le S (iInf_le_of_le h le_rfl)

theorem le_covering_number_of {e : X → X → ENNReal} {ε : ENNReal} {U : Set X} {c : ENNReal}
    (h : ∀ S : Finset X, InternalEpsilonNet e ε U S → c ≤ S.card) : c ≤ covering_number e ε U :=
  le_iInf fun S => le_iInf fun hS => h S hS

/-- 下界工具：若同一中心不能同时服务 `W` 中两个不同点，则覆盖数至少为 `|W|`。 -/
theorem card_le_covering_number {e : X → X → ENNReal} {ε : ENNReal} {U : Set X}
    (W : Finset X) (hW : ↑W ⊆ U)
    (hsep : ∀ c ∈ U, ∀ w1 ∈ W, ∀ w2 ∈ W, e w1 c < ε → e w2 c < ε → w1 = w2) :
    (W.card : ENNReal) ≤ covering_number e ε U := by
  classical
  refine le_covering_number_of fun S hS => ?_
  have : ∀ w ∈ W, ∃ c ∈ S, e w c < ε := fun w hw => hS.2 w (hW hw)
  choose! g hgS hgd using this
  have hinj : Set.InjOn g W := by
    intro w1 h1 w2 h2 heq
    exact hsep (g w1) (hS.1 (hgS w1 h1)) w1 h1 w2 h2 (hgd w1 h1) (by rw [heq]; exact hgd w2 h2)
  exact_mod_cast Finset.card_le_card_of_injOn g (fun w hw => hgS w hw) hinj

/-- 单中心覆盖非空集合时覆盖数恰为 1。 -/
theorem covering_number_eq_one {e : X → X → ENNReal} {ε : ENNReal} {U : Set X}
    (c : X) (hc : c ∈ U) (h : ∀ x ∈ U, e x c < ε) : covering_number e ε U = 1 := by
  classical
  apply le_antisymm
  · have : InternalEpsilonNet e ε U {c} := by
      refine ⟨by simpa using hc, fun x hx => ⟨c, Finset.mem_singleton_self c, h x hx⟩⟩
    simpa using covering_number_le_card {c} this
  · exact one_le_covering_number_of_nonempty e ε U ⟨c, hc⟩

/-- 非空有限集以自身为中心集，覆盖数有限。 -/
theorem covering_number_finite {e : X → X → ENNReal} (h0 : ∀ x, e x x = 0) {ε : ENNReal}
    (hε : 0 < ε) {U : Set X} (hU : U.Finite) : covering_number e ε U ≤ hU.toFinset.card :=
  covering_number_le_card _ ⟨by simp, fun x hx => ⟨x, by simpa using hx, by rw [h0]; exact hε⟩⟩

/-- 全有界：每个有限正精度下的内覆盖数有限。 -/
def TotallyBoundedIn (e : X → X → ENNReal) (U : Set X) : Prop :=
  ∀ r : ENNReal, 0 < r → r < ⊤ → covering_number e r U < ⊤

theorem log2_ereal_mono {a b : ENNReal} (h : a ≤ b) : log2_ereal a ≤ log2_ereal b := by
  unfold log2_ereal
  by_cases ha0 : a = 0
  · simp [ha0]
  have hb0 : b ≠ 0 := fun hb => ha0 (le_antisymm (hb ▸ h) (zero_le _))
  by_cases hbt : b = ⊤
  · simp [hbt]
  have hat : a ≠ ⊤ := ne_top_of_le_ne_top hbt h
  simp only [ha0, hbt, hat, hb0, if_false]
  exact EReal.coe_le_coe_iff.mpr (Real.logb_le_logb_of_le one_lt_two
    (ENNReal.toReal_pos ha0 hat) (ENNReal.toReal_mono hbt h))

theorem log2_ereal_eq_top_iff (a : ENNReal) : log2_ereal a = ⊤ ↔ a = ⊤ := by
  unfold log2_ereal
  by_cases ha0 : a = 0
  · simp [ha0]
  by_cases hat : a = ⊤
  · simp [hat]
  simp [ha0, hat]

theorem log2_ereal_finite {a : ENNReal} (h1 : 1 ≤ a) (h2 : a < ⊤) :
    log2_ereal a = ((Real.logb 2 a.toReal : ℝ) : EReal) := by
  have ha0 : a ≠ 0 := ne_of_gt (lt_of_lt_of_le zero_lt_one h1)
  simp [log2_ereal, ha0, h2.ne]

@[simp] theorem log2_ereal_one : log2_ereal 1 = 0 := by
  simp [log2_ereal]

section EntropyDiff
variable {X I : Type} {bot : X} (d : I → X → X → ENNReal)

/-- 映射熵差良定条件：输入或输出覆盖数至少一个有限。 -/
def EntropyDiffDefined (Jin Jout : Set I) (ε : ENNReal) (ϕ : Operator X bot) (U : Set X) : Prop :=
  covering_number (subfamily_pseudometric d Jin) ε U < ⊤ ∨
    covering_number (subfamily_pseudometric d Jout) ε (ϕ.func '' U) < ⊤

open Classical in
/-- 映射熵差 `ΔE_{ε,Jin,Jout}(ϕ | U)`：良定时为 `some (E_in − E_out)`，否则 `none`（不定义 `⊤ − ⊤`）。 -/
noncomputable def entropyDiff (Jin Jout : Set I) (ε : ENNReal) (ϕ : Operator X bot)
    (U : Set X) : Option EReal :=
  if EntropyDiffDefined d Jin Jout ε ϕ U then
    some (subfamily_metric_entropy d Jin ε U - subfamily_metric_entropy d Jout ε (ϕ.func '' U))
  else none

theorem entropyDiff_of_defined {Jin Jout : Set I} {ε : ENNReal} {ϕ : Operator X bot}
    {U : Set X} (h : EntropyDiffDefined d Jin Jout ε ϕ U) :
    entropyDiff d Jin Jout ε ϕ U =
      some (subfamily_metric_entropy d Jin ε U - subfamily_metric_entropy d Jout ε (ϕ.func '' U)) := by
  simp [entropyDiff, h]

theorem entropyDiff_of_not_defined {Jin Jout : Set I} {ε : ENNReal} {ϕ : Operator X bot}
    {U : Set X} (h : ¬ EntropyDiffDefined d Jin Jout ε ϕ U) : entropyDiff d Jin Jout ε ϕ U = none := by
  simp [entropyDiff, h]

theorem subfamily_self_zero (hd : ∀ i, ExtendedPseudoMetric X bot (d i)) (J : Set I) (x : X) :
    subfamily_pseudometric d J x x = 0 :=
  (subfamily_eq_zero_iff d J x x).2 fun j _ => (hd j).dist_self x

/-- D08：良定条件排除两端熵同为 `⊤`；各种无穷情形的取值。 -/
theorem entropyDiff_cases (Jin Jout : Set I) (ε : ENNReal) (ϕ : Operator X bot) (U : Set X) :
    (EntropyDiffDefined d Jin Jout ε ϕ U ↔
      ¬ (subfamily_metric_entropy d Jin ε U = ⊤ ∧
         subfamily_metric_entropy d Jout ε (ϕ.func '' U) = ⊤)) ∧
    (subfamily_metric_entropy d Jin ε U = ⊤ → subfamily_metric_entropy d Jout ε (ϕ.func '' U) ≠ ⊤ →
      (ϕ.func '' U).Nonempty → entropyDiff d Jin Jout ε ϕ U = some ⊤) ∧
    (subfamily_metric_entropy d Jin ε U ≠ ⊤ → U.Nonempty →
      subfamily_metric_entropy d Jout ε (ϕ.func '' U) = ⊤ → entropyDiff d Jin Jout ε ϕ U = some ⊥) := by
  have hiff : EntropyDiffDefined d Jin Jout ε ϕ U ↔
      ¬ (subfamily_metric_entropy d Jin ε U = ⊤ ∧
         subfamily_metric_entropy d Jout ε (ϕ.func '' U) = ⊤) := by
    simp only [EntropyDiffDefined, subfamily_metric_entropy, log2_ereal_eq_top_iff, lt_top_iff_ne_top]
    tauto
  refine ⟨hiff, ?_, ?_⟩
  · intro hin hout hne
    rw [entropyDiff_of_defined d (hiff.2 fun h => hout h.2), hin]
    have h0 : (0 : EReal) ≤ subfamily_metric_entropy d Jout ε (ϕ.func '' U) :=
      metric_entropy_nonneg _ ε _ hne
    have hbot : subfamily_metric_entropy d Jout ε (ϕ.func '' U) ≠ ⊥ := by
      intro h; rw [h] at h0; exact absurd h0 (by simp)
    rw [EReal.top_sub hout]
  · intro hin hne hout
    rw [entropyDiff_of_defined d (hiff.2 fun h => hin h.1), hout]
    have h0 : (0 : EReal) ≤ subfamily_metric_entropy d Jin ε U := metric_entropy_nonneg _ ε _ hne
    have hbot : subfamily_metric_entropy d Jin ε U ≠ ⊥ := by
      intro h; rw [h] at h0; exact absurd h0 (by simp)
    exact congrArg some (EReal.sub_top _)

/-- D08：两端覆盖数均有限（集合非空）时映射熵差为普通实数。 -/
theorem entropyDiff_real (Jin Jout : Set I) (ε : ENNReal) (ϕ : Operator X bot) (U : Set X)

    (hin : covering_number (subfamily_pseudometric d Jin) ε U < ⊤)
    (hout : covering_number (subfamily_pseudometric d Jout) ε (ϕ.func '' U) < ⊤) :
    ∃ r : ℝ, entropyDiff d Jin Jout ε ϕ U = some (r : EReal) := by

  rw [log2_ereal_finite hin1 hin, log2_ereal_finite hout1 hout, ← EReal.coe_sub]

/-- 命题 3.19：单通道 Lip 上界与熵差下界（输入全有界给两端有限熵，熵差良定）。 -/
theorem prop319 (hd : ∀ i, ExtendedPseudoMetric X bot (d i)) (ϕ : Operator X bot)
    (U : Set X) (hU0 : U.Nonempty) (hU : U ⊆ dom ϕ) (i j : I)
    (hTB : TotallyBoundedIn (d i) U) (ε : ENNReal) (hε0 : 0 < ε) (hεt : ε < ⊤)
    (L : ENNReal) (hL0 : 0 < L) (hLt : L < ⊤)
    (hlip : ∀ x ∈ U, ∀ y ∈ U, d j (ϕ.func x) (ϕ.func y) ≤ L * d i x y) :
    covering_number (d j) ε (ϕ.func '' U) ≤ covering_number (d i) (ε / L) U ∧
    EntropyDiffDefined d {i} {j} ε ϕ U ∧
    metric_entropy (d i) ε U ≠ ⊤ ∧ metric_entropy (d j) ε (ϕ.func '' U) ≠ ⊤ ∧
    ∃ r : ℝ, entropyDiff d {i} {j} ε ϕ U = some (r : EReal) ∧
      metric_entropy (d i) ε U - metric_entropy (d i) (ε / L) U ≤ (r : EReal) := by
  have hcov := prop319_cover d ϕ U i j ε L hL0 hLt hlip
  have hεL0 : 0 < ε / L := ENNReal.div_pos hε0.ne' hLt.ne
  have hεLt : ε / L < ⊤ := ENNReal.div_lt_top hεt.ne hL0.ne'
  have hin := hTB ε hε0 hεt
  have hinL := hTB (ε / L) hεL0 hεLt
  have hout : covering_number (d j) ε (ϕ.func '' U) < ⊤ := lt_of_le_of_lt hcov hinL
  have hin1 := one_le_covering_number_of_nonempty (d i) ε U hU0
  have hout1 := one_le_covering_number_of_nonempty (d j) ε _ (hU0.image ϕ.func)
  have hinL1 := one_le_covering_number_of_nonempty (d i) (ε / L) U hU0
  have hdef : EntropyDiffDefined d {i} {j} ε ϕ U := by
    left; rw [subfamily_pseudometric_singleton]; exact hin
  have hEin := log2_ereal_finite hin1 hin
  have hEout := log2_ereal_finite hout1 hout
  have hEmono : metric_entropy (d j) ε (ϕ.func '' U) ≤ metric_entropy (d i) (ε / L) U :=
    log2_ereal_mono hcov
  refine ⟨hcov, hdef, ?_, ?_, ?_⟩
  · simp only [metric_entropy, hEin]; exact EReal.coe_ne_top _
  · simp only [metric_entropy, hEout]; exact EReal.coe_ne_top _
  · refine ⟨Real.logb 2 (covering_number (d i) ε U).toReal -
      Real.logb 2 (covering_number (d j) ε (ϕ.func '' U)).toReal, ?_, ?_⟩
    · rw [entropyDiff_of_defined d hdef, subfamily_metric_entropy_singleton,
        subfamily_metric_entropy_singleton]
      simp only [metric_entropy]
      rw [hEin, hEout, ← EReal.coe_sub]
    · rw [EReal.coe_sub, ← hEin, ← hEout]
      exact EReal.sub_le_sub le_rfl hEmono

/-- 命题 3.20（拉回）：正有限 co-Lip 单条目把输出内覆盖中心的原像拉回为 `ε/K` 输入内覆盖。 -/
theorem prop320_net [Fintype I] (ϕ : Operator X bot) (U : Set X) (i j : I) (ε : ENNReal)
    (K : I → I → ENNReal) (hK : IsCoLipMatrix d ϕ U K) (hK0 : 0 < K i j) (hKt : K i j < ⊤)
    [DecidableEq X] (T : Finset X) (hT : InternalEpsilonNet (d j) ε (ϕ.func '' U) T) :
    ∃ S : Finset X, InternalEpsilonNet (d i) (ε / K i j) U S ∧ S.card ≤ T.card := by

  let g : X → X := fun t => if h : t ∈ ϕ.func '' U then Classical.choose h else t

    intro t ht

  · intro z hz

  · intro x hx

        _ ≤ d j (ϕ.func x) (ϕ.func (g t)) := hco
        _ = d j (ϕ.func x) t := by rw [(hg t ht).2]
        _ < ε := hxt
    rw [ENNReal.lt_div_iff_mul_lt (Or.inl hK0.ne') (Or.inl hKt.ne), mul_comm]
    exact hlt

  have hinK1 := one_le_covering_number_of_nonempty (d i) (ε / K i j) U hU0
  have hdef : EntropyDiffDefined d {i} {j} ε ϕ U := by
    left; rw [subfamily_pseudometric_singleton]; exact hin
  refine ⟨hcov, hdef, ?_, ?_, ?_⟩
  · simp only [metric_entropy, log2_ereal_finite hin1 hin]; exact EReal.coe_ne_top _

section Absorb
variable {α : Type}

/-- 按吸收元隔离公理把 `α` 上的距离扩展到 `Option α`（`none` 为吸收元）。 -/
def absorbExt (b : α → α → ENNReal) : Option α → Option α → ENNReal
  | none, none => 0
  | none, some _ => ⊤
  | some _, none => ⊤
  | some x, some y => b x y

@[simp] theorem absorbExt_some (b : α → α → ENNReal) (x y : α) :
    absorbExt b (some x) (some y) = b x y := rfl

theorem absorbExt_epm [Nonempty α] (b : α → α → ENNReal) (h0 : ∀ x, b x x = 0)
    (hs : ∀ x y, b x y = b y x) (ht : ∀ x y z, b x z ≤ b x y + b y z) :
    ExtendedPseudoMetric (Option α) none (absorbExt b) where
  nonabsorbing := ⟨some (Classical.arbitrary α), by simp⟩
  dist_self x := by cases x <;> simp [absorbExt, h0]
  symm x y := by cases x <;> cases y <;> simp [absorbExt, hs]
  triangle x y z := by cases x <;> cases y <;> cases z <;> simp [absorbExt, ht]
  isolate_bot x hx := by cases x <;> simp_all [absorbExt]

/-- 全函数提升为保持吸收元的算子。 -/
def liftOp (g : α → α) : Operator (Option α) none := ⟨Option.map g, rfl⟩

/-- 偏函数（`none` 表示未定义）提升为算子。 -/
def partialOp (g : α → Option α) : Operator (Option α) none := ⟨fun o => o.bind g, rfl⟩

@[simp] theorem liftOp_some (g : α → α) (x : α) : (liftOp g).func (some x) = some (g x) := rfl
@[simp] theorem partialOp_some (g : α → Option α) (x : α) :
    (partialOp g).func (some x) = g x := rfl

  simp [dom]

/-- 两值距离：相等为 0，不等为 `c`。 -/
def boolDist (c : ENNReal) : Bool → Bool → ENNReal := fun x y => if x = y then 0 else c

theorem boolDist_epm (c : ENNReal) : ExtendedPseudoMetric (Option Bool) none (absorbExt (boolDist c)) :=
  absorbExt_epm _ (fun x => by simp [boolDist]) (fun x y => by cases x <;> cases y <;> simp [boolDist])

theorem diam_Icc (a b : ℝ) (hab : a ≤ b) :

  apply le_antisymm
  · refine iSup₂_le fun x hx => iSup₂_le fun y hy => ?_
    obtain ⟨x, ⟨hx1, hx2⟩, rfl⟩ := hx
    obtain ⟨y, ⟨hy1, hy2⟩, rfl⟩ := hy

    exact ENNReal.ofReal_le_ofReal (abs_sub_le_iff.2 ⟨by linarith, by linarith⟩)
  · have := le_diam dRx (A := some '' Set.Icc a b) (x := some b) (y := some a)
      ⟨b, ⟨hab, le_rfl⟩, rfl⟩ ⟨a, ⟨le_rfl, hab⟩, rfl⟩
    have e : dRx (some b) (some a) = ENNReal.ofReal (b - a) := by
      show ENNReal.ofReal |b - a| = _; rw [abs_of_nonneg (by linarith)]
    rwa [e] at this

/-- 跳点 `1` 左侧整段在评估域内时，`1` 处的零化半径为 0。 -/
theorem floor_jump_radius_zero (D : Set (Option ℝ)) (η : ℝ) (hη0 : 0 < η) (hη1 : η ≤ 1)
    (hsub : ∀ x : ℝ, 1 - η ≤ x → x < 1 → some x ∈ D) :
    nullRadius dRx dRx floorOp D (some 1) = 0 := by
  apply le_antisymm _ (zero_le _)
  apply nullRadius_le_of_forall
  intro r hr
  exfalso
  rw [isNullRad_iff] at hr
  obtain ⟨h0, hrt, h⟩ := hr
  have ht : 0 < r.toReal := ENNReal.toReal_pos h0.ne' hrt.ne
  set x := max (1 - η) (1 - r.toReal / 2) with hxdef
  have hx1 : 1 - η ≤ x := le_max_left _ _
  have hx1' : 1 - r.toReal / 2 ≤ x := le_max_right _ _
  have hx2 : x < 1 := max_lt (by linarith) (by linarith)
  have hx0 : 0 ≤ x := le_trans (by linarith) hx1
  have hd : dRx (some x) (some 1) < r := by
    show dR x 1 < r
    rw [dR_lt_iff _ _ hrt.ne, abs_of_neg (by linarith)]
    linarith
  have := h (some x) (hsub x hx1 hx2) hd
  have hfx : ⌊x⌋ = 0 := Int.floor_eq_zero_iff.mpr ⟨hx0, hx2⟩
  norm_num [floorOp, dRx, dR, hfx] at this
